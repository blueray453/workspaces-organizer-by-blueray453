import { createLogger } from '../logger.js';
const journal = createLogger(import.meta.url);

const _ = ExtensionModule.gettext;

let _instance = null;
let _settleWindow = null;

const MIN_RENDER_INTERVAL_MS = 100;

// Internal window drag: dwell before the flyout opens.
const DWELL_MS = 150;
// External (file/text) drag: pointer poll, dwell, preview.
const EXT_TICK_MS = 40;
// Hover-to-preview delay. Shorter than the flyout dwell so moving into the
// popup feels responsive; matches the overlay's _scheduleReveal timing.
const EXT_PREVIEW_MS = 150;
const EXT_WS_DWELL_MS = 500;

const SLACK_PX = 32;
const MAX_SOLVE_STEPS = 6;

const SIZE_LADDER = [1, 0.9, 0.78, 0.66, 0.55];
const SIZE_FLOOR = 20;

// get_transformed_position() etc. return NaN for not-yet-allocated
// actors; keep the logs readable either way.
const fmtNum = v => Number.isFinite(v) ? String(Math.round(v)) : 'NaN';

export function thumbnailAt(x, y) { return _instance?._thumbnailAt(x, y) ?? null; }
export function settleOnNextRender(window) { _settleWindow = window; }

export function toggleGroupFlyout(workspaceIndex, wmClass = '') {
    return _instance?._toggleGroupFlyout(workspaceIndex, wmClass) ?? false;
}

export function closeGroupFlyout() {
    return _instance?._closeGroupFlyout() ?? false;
}

export class WorkspaceIndicator extends PanelMenu.Button {
    static { GObject.registerClass(this); }

    constructor(settings, commands) {
        super(0.0, _('Workspace Indicator'));
        this.reactive = false;
        this._settings = settings;
        this._commands = commands;
        _instance = this;

        this._controllers = new Map();
        this._lastIconSize = undefined;
        this._lastRenderTime = undefined;
        this._toolbarKey = null;
        this._staticBox = null;
        this._nameLabel = null;
        this._thumbsBox = null;
        this._entries = [];

        this._collapsed = new Set();
        this._denseStep = 0;

        this._scope = new Scheduling.Scope();
        this._renderCoalescer = new Scheduling.Coalescer(() => this._render());
        this._leavePollId = 0;

        this._dwellId = 0;
        this._dwellKey = null;

        this._groupIcons = new Map();

        this._flyout = new GroupFlyout(settings);

        // External drag state
        this._extPollId = 0;
        this._extRowKey = null;
        this._extRowTimer = 0;
        this._extWsKey = null;
        this._extWsTimer = 0;
        this._extSawButton = false;
        this._extGroupKey = null;   // last group the external poll hovered
        this._geoRelogId = 0;       // deferred geometry re-log source

        // Last sticky (DBus) flyout request; re-applied after re-renders.
        this._stickyRequest = null;

        this._lastHintWs = Model.intent.hintWs;
        this._lastDragActive = !!Model.intent.drag;
        this._lastMenuOpen = !!Model.intent.menuOpen;

        this._ghost = Interactions.createReorderGhost({
            iconSize: () => settings.get_int('icon-size'),
        });

        this._mainBox = new St.BoxLayout({
            style_class: 'workspace-indicator-main-box',
            y_expand: true, x_expand: true, reactive: true,
        });
        this.add_child(this._mainBox);

        Model.onIntentChange(() => this._onIntentChanged());
        Model.onWorldChange(() => this._queueRender());

        this._extUnsub = externalDrag.onChanged(state => this._onExternalChanged(state));

        this._render();
    }

    // ==================== INTENT ====================
    _dragActive() { return !!Model.intent.drag || externalDrag.active; }

    _onIntentChanged() {
        const hintChanged = Model.intent.hintWs !== this._lastHintWs;
        const isDragging = !!Model.intent.drag;
        const isMenu = !!Model.intent.menuOpen;
        const dragChanged = isDragging !== this._lastDragActive;
        const menuChanged = isMenu !== this._lastMenuOpen;

        this._lastHintWs = Model.intent.hintWs;
        this._lastDragActive = isDragging;
        this._lastMenuOpen = isMenu;

        // A sticky (DBus-opened) flyout has no drag lifetime — it survives
        // intent churn and is closed explicitly or by a re-render (which
        // re-anchors it afterwards).
        if (!this._dragActive() && !this._flyout.sticky) {
            this._cancelDwell();
            this._flyout.close(true);
        }
        if (hintChanged) this._syncHint();
        if (dragChanged || menuChanged) this._queueRender();
    }

    _queueRender(delay = 0) {
        const now = GLib.get_monotonic_time() / 1000;
        const since = now - (this._lastRenderTime ?? 0);
        if (since < MIN_RENDER_INTERVAL_MS)
            delay = Math.max(delay, MIN_RENDER_INTERVAL_MS - since);
        this._renderCoalescer.queue(delay);
    }

    // ==================== RENDER ====================
    _render() {
        if (Model.intent.menuOpen) return;
        if (Model.intent.drag) return;
        if (externalDrag.active) return;

        // A rebuild replaces every group icon; an anchored flyout would be
        // left pointing at destroyed actors (NaN geometry, dead hit tests).
        // Sticky flyouts are re-anchored after the solve below.
        const hadFlyout = this._flyout.isOpen;
        if (hadFlyout) {
            journal('[flyout] re-render -> closing flyout (sticky restore if requested)');
            this._flyout.close(true);
        }

        this._ghost.clear();

        const model = Model.buildModel();
        Model.logModel(model);
        const s = this._settings;

        const iconSize = s.get_int('icon-size');
        if (this._lastIconSize !== undefined && this._lastIconSize !== iconSize) {
            for (const ctrl of this._controllers.values()) ctrl.destroy();
            this._controllers.clear();
        }
        this._lastIconSize = iconSize;

        for (const ctrl of this._controllers.values()) {
            const parent = ctrl.actor.get_parent();
            if (parent) parent.remove_child(ctrl.actor);
        }

        const toolbarKey = [
            s.get_boolean('show-drun-button'),
            s.get_boolean('show-window-button'),
            s.get_boolean('show-workspace-names'),
            s.get_int('toolbar-icon-size'),
            s.get_int('workspace-name-font-size'),
        ].join('|');
        if (this._toolbarKey !== toolbarKey) {
            this._toolbarKey = toolbarKey;
            this._buildToolbar(s);
        }
        if (this._nameLabel) {
            this._nameLabel.set_text(Meta.prefs_get_workspace_name(
                Model.WorkspaceManager.get_active_workspace_index()));
        }

        this._thumbsBox?.destroy();
        this._settleActor = null;
        this._entries = [];
        this._groupIcons.clear();

        this._thumbsBox = new St.BoxLayout({
            style_class: 'workspace-indicator-class',
            y_expand: true, reactive: true,
            clip_to_allocation: true,
        });
        this._mainBox.add_child(this._thumbsBox);

        const seen = new Set();
        for (const entry of model.workspaces) {
            this._entries.push(this._buildThumbShell(entry));
            this._thumbsBox.add_child(this._entries[this._entries.length - 1].actor);
        }
        for (const entry of model.workspaces)
            for (const w of entry.windows) seen.add(`${w.get_id()}:${entry.index}`);

        this._solveGrouping(model);

        for (const [key, ctrl] of this._controllers) {
            if (!seen.has(key)) {
                ctrl.destroy();
                this._controllers.delete(key);
            }
        }

        // A sticky (DBus-opened) flyout was closed at the top of this
        // render because the rebuild replaced its anchor actor; re-open it
        // on the freshly built group icon (open() retries until the new
        // actor's allocation lands).
        if (hadFlyout && this._stickyRequest && !this._flyout.isOpen) {
            const req = this._stickyRequest;
            journal(`[flyout] re-render -> restoring sticky flyout ws=${req.workspaceIndex} wmclass="${req.wmClass}"`);
            this._toggleGroupFlyout(req.workspaceIndex, req.wmClass);
        }

        if (this._settleActor) Interactions.settleIcon(this._settleActor);
        this._syncHint();

        this._lastRenderTime = GLib.get_monotonic_time() / 1000;
    }

    _buildToolbar(s) {
        if (!this._staticBox) {
            this._staticBox = new St.BoxLayout({
                y_expand: true, y_align: Clutter.ActorAlign.CENTER,
            });
            this._mainBox.insert_child_at_index(this._staticBox, 0);
        }
        this._staticBox.destroy_all_children();
        this._nameLabel = null;

        const iconSize = s.get_int('toolbar-icon-size');
        if (s.get_boolean('show-window-button')) {
            const b = UI.createToolbarButton('focus-windows-symbolic', iconSize, 'All windows');
            b.connect('clicked', () => this._commands.showAllWindows());
            this._staticBox.add_child(b);
        }
        if (s.get_boolean('show-drun-button')) {
            const b = UI.createToolbarButton('view-app-grid-symbolic', iconSize, 'Show apps');
            b.connect('clicked', () => this._commands.showAllApps());
            this._staticBox.add_child(b);
        }
        if (s.get_boolean('show-workspace-names')) {
            this._nameLabel = UI.createWorkspaceNameLabel(
                Meta.prefs_get_workspace_name(Model.WorkspaceManager.get_active_workspace_index()),
                s.get_int('workspace-name-font-size'));
            this._staticBox.add_child(this._nameLabel);
        }
    }

    // ==================== THUMBNAIL SHELL ====================
    _buildThumbShell(entry) {
        const s = this._settings;

        const { actor: thumb, box } = UI.createThumbnail({
            minWidth: s.get_int('thumbnail-min-width'),
            active: entry.index === Model.WorkspaceManager.get_active_workspace_index(),
        });
        thumb._wsEntry = { workspace: entry.workspace, index: entry.index };

        const content = new St.BoxLayout({
            style_class: 'ws-contents',
            y_align: Clutter.ActorAlign.CENTER,
        });
        box.add_child(content);

        const shell = { entry, actor: thumb, content, strip: null };

        const strip = Interactions.createDropStrip({
            container: content,
            ghost: this._ghost,
            acceptSource: source => source?._dragPayload?.kind === 'window',
            externalResult: Interactions.DragMotionResult.MOVE_DROP,
            onDragOver: () => {
                this._ensureLeavePoll();
                Model.setIntent({ hintWs: entry.index });
            },
            onExternalDragOver: () => {
                if (externalDrag.active) return;
                if (entry.workspace !== Model.WorkspaceManager.get_active_workspace())
                    entry.workspace.activate(global.get_current_time());
            },
            onDrop: (source, dropIndex, dragActor) => {
                this._cancelDwell();
                this._flyout.close(true);
                const grouped = this._isGrouped(entry);
                return this._dropOnWorkspace(thumb, entry, source,
                    grouped ? null : dropIndex, dragActor);
            },
        });
        shell.strip = strip;
        thumb._delegate = strip.delegate();

        thumb.connect('button-press-event', (actor, event) => {
            const b = event.get_button();
            if (b === Clutter.BUTTON_PRIMARY) entry.workspace.activate(0);
            else if (b === Clutter.BUTTON_SECONDARY) this._openThumbnailMenu(thumb, entry.workspace);
            return Clutter.EVENT_STOP;
        });

        return shell;
    }

    // ==================== GROUPING ====================
    _isCollapsed(entry, group) {
        if (group.count < 2) return false;
        if (this._settings.get_boolean('group-apps-always')) return true;
        if (!entry.direct) return true;
        return this._collapsed.has(group.key);
    }

    _isGrouped(entry) {
        return entry.groups.some(g => this._isCollapsed(entry, g));
    }

    _iconSize() {
        const base = this._settings.get_int('icon-size');
        const dense = this._entries.some(e => this._isGrouped(e.entry));
        if (!dense) return base;

        const rung = Math.min(this._denseStep, SIZE_LADDER.length - 1);
        return Math.max(SIZE_FLOOR, Math.round(base * SIZE_LADDER[rung]));
    }

    _widthBudget() {
        const panelW = Main.panel.width > 0 ? Main.panel.width : Model.screenWidth;
        const [x] = this.get_transformed_position();
        const left = Number.isFinite(x) ? Math.max(0, x) : 0;

        let reserve = 28;
        const rb = Main.panel._rightBox;
        if (rb && rb.width > 0) reserve += rb.width + 12;
        else reserve += Math.round(panelW * 0.30);

        return Math.max(200, Math.min(panelW - left - reserve, Math.round(panelW * 0.62)));
    }

    _naturalWidth() {
        try { return this._mainBox.get_preferred_width(-1)[1]; }
        catch (e) { return 0; }
    }

    _pruneCollapsed(entries) {
        const valid = new Set();
        for (const e of entries)
            for (const g of e.entry.groups)
                if (g.count > 1) valid.add(g.key);
        for (const key of [...this._collapsed])
            if (!valid.has(key)) this._collapsed.delete(key);
    }

    _collapseCandidates(entries) {
        if (this._settings.get_boolean('group-apps-always')) return [];

        const activeIdx = Model.WorkspaceManager.get_active_workspace_index();
        const out = [];
        for (const e of entries) {
            if (!e.entry.direct) continue;
            for (const g of e.entry.groups)
                if (g.count > 1) out.push({ key: g.key, count: g.count, wsIndex: e.entry.index });
        }
        out.sort((a, b) =>
            b.count - a.count ||
            (Number(b.wsIndex === activeIdx) - Number(a.wsIndex === activeIdx)) ||
            a.wsIndex - b.wsIndex);
        return out;
    }

    _solveGrouping() {
        const budget = this._widthBudget();
        this._pruneCollapsed(this._entries);
        const candidates = this._collapseCandidates(this._entries);

        this._denseStep = 0;
        this._renderContents();

        let steps = 0;

        if (this._naturalWidth() > budget) {
            for (const c of candidates) {
                if (steps++ >= MAX_SOLVE_STEPS) break;
                if (this._collapsed.has(c.key)) continue;
                this._collapsed.add(c.key);
                this._renderContents();
                if (this._naturalWidth() <= budget) break;
            }
        } else {
            const reverted = [...candidates].reverse();
            for (const c of reverted) {
                if (steps++ >= MAX_SOLVE_STEPS) break;
                if (!this._collapsed.has(c.key)) continue;
                this._collapsed.delete(c.key);
                this._renderContents();
                if (budget - this._naturalWidth() < SLACK_PX) {
                    this._collapsed.add(c.key);
                    this._renderContents();
                    break;
                }
            }
        }

        let rungs = 0;
        while (this._naturalWidth() > budget &&
            this._denseStep < SIZE_LADDER.length - 1 &&
            rungs++ < SIZE_LADDER.length) {
            this._denseStep++;
            this._renderContents();
        }

        this._thumbsBox.style_class = this._collapsed.size > 0
            ? 'workspace-indicator-class grouped-layout'
            : 'workspace-indicator-class';
    }

    // ==================== CONTENTS ====================
    _renderContents() {
        for (const ctrl of this._controllers.values()) {
            const parent = ctrl.actor.get_parent();
            if (parent) parent.remove_child(ctrl.actor);
        }
        this._groupIcons.clear();

        const sz = this._iconSize();
        const dense = this._entries.some(e => this._isGrouped(e.entry));

        for (const shell of this._entries) {
            const { entry, content } = shell;
            content.destroy_all_children();
            content.spacing = dense ? 3 : 6;

            if (!entry.windows.length) {
                content.add_child(new St.Label({ style_class: 'ws-empty', text: 'empty' }));
                continue;
            }

            for (const group of entry.groups) {
                if (this._isCollapsed(entry, group)) {
                    content.add_child(this._buildGroupIcon(shell, group, sz));
                    continue;
                }
                group.windows.forEach(w =>
                    content.add_child(this._windowActor(w, entry.index, sz, shell.strip)));
            }
        }
    }

    _windowActor(w, wsIndex, iconSize, strip) {
        const key = `${w.get_id()}:${wsIndex}`;
        let ctrl = this._controllers.get(key);

        if (ctrl?.stale) {
            ctrl.destroy();
            this._controllers.delete(key);
            ctrl = undefined;
        }

        if (ctrl) {
            ctrl.setStrip(strip);
        } else {
            const { actor, icon } = UI.createWindowIconActor(w, iconSize);
            ctrl = new Interactions.WindowIconController({
                window: w, actor, icon,
                settings: this._settings,
                strip,
                onRequestRender: () => { ctrl.stale = true; this._queueRender(); },
            });
            this._controllers.set(key, ctrl);
        }

        if (w === _settleWindow) { _settleWindow = null; this._settleActor = ctrl.actor; }
        return ctrl.actor;
    }

    _buildGroupIcon(shell, group, iconSize) {
        const { entry } = shell;
        const { actor } = UI.createGroupButton({
            gicon: UI.getWindowAppIcon(group.windows[0]),
            count: group.count,
            size: iconSize,
        });
        actor.add_style_class_name('window-preview-icon');
        actor.accessible_name = `${group.appName}, ${group.count} windows`;
        actor.set_name(`group-${group.key}`);

        this._groupIcons.set(actor, { shell, entry, group });

        actor.connect('clicked', () => this._openGroupSelector(entry, group));

        actor._delegate = {
            handleDragOver: source => {
                this._dwellGroup(shell, group, actor);
                return shell.strip.handleDragOver(source);
            },
            acceptDrop: (source, dragActor) => {
                this._cancelDwell();
                this._flyout.close(true);
                return shell.strip.acceptDrop(source, dragActor);
            },
        };

        return actor;
    }

    _openGroupSelector(entry, group) {
        if (this._dragActive()) return;
        this._flyout.close(true);
        this._commands.showGroupWindows?.(entry.index, group.appId);
    }

    // ==================== FLYOUT ====================
    _dwellGroup(shell, group, anchor) {
        if (group.count < 2) {
            journal(`[flyout] dwell skipped: ${group.appName} count=${group.count}`);
            return;
        }
        if (this._flyout.isOpen && this._flyout.anchor === anchor) return;
        if (this._dwellKey === group.key) return;

        journal(`[flyout] dwell armed: ${group.appName} x${group.count} key=${group.key}`);
        this._cancelDwell();
        this._dwellKey = group.key;
        this._dwellId = this._scope.timeout(DWELL_MS, () => {
            this._dwellId = 0;
            this._dwellKey = null;
            const active = this._dragActive();
            journal(`[flyout] dwell fired: dragActive=${active}`);
            if (!active) return GLib.SOURCE_REMOVE;
            this._openFlyout(shell, group, anchor);
            return GLib.SOURCE_REMOVE;
        });
    }

    _cancelDwell() {
        if (this._dwellId) { this._scope.cancel(this._dwellId); this._dwellId = 0; }
        this._dwellKey = null;
    }

    _openFlyout(shell, group, anchor, { sticky = false, anchorKey = null, attempt = 0 } = {}) {
        try {
            const { entry } = shell;
            let anchorOnStage = false;
            try { anchorOnStage = !!anchor.get_stage(); } catch (e) { anchorOnStage = false; }
            journal(`[flyout] _openFlyout ${group.appName} ws=${entry.index} ` +
                `sticky=${sticky} attempt=${attempt} ` +
                `hasStrip=${!!shell.strip} anchorOnStage=${anchorOnStage}`);
            shell.strip.cancel();

            const items = group.windows.map((w, k) => ({
                window: w,
                index: group.indices[k],
                icon: UI.getWindowAppIcon(w),
                title: w.get_title() || 'Untitled Window',
                sub: `${group.appName}`.toUpperCase(),
                color: UI.colorForApp(group.appName),
            }));
            journal(`[flyout] items=${items.length}`);

            const opened = this._flyout.open({
                anchor,
                title: `${group.appName.toUpperCase()} · ${group.count} WINDOWS · WS ${entry.index + 1}`,
                items,
                onDrop: (source, item) => this._dropIntoGroup(shell, source, item),
                sticky,
                anchorKey,
            });

            // open() refuses anchors whose allocation has not landed yet
            // (NaN geometry). Retry a few times; a re-render in between
            // replaces the anchor and the retry aborts itself.
            if (!opened) {
                let anchorAlive = false;
                try { anchorAlive = !!anchor.get_stage(); } catch (e) { anchorAlive = false; }
                if (attempt < 5 && anchorAlive) {
                    journal(`[flyout] deferring open — retry ${attempt + 1}/5 in 60ms`);
                    this._scope.timeout(60, () => {
                        if (!this._flyout.isOpen)
                            this._openFlyout(shell, group, anchor,
                                { sticky, anchorKey, attempt: attempt + 1 });
                        return GLib.SOURCE_REMOVE;
                    });
                } else {
                    journal(`[flyout] giving up after ${attempt + 1} attempt(s)`);
                }
            }

            journal(`[flyout] open() returned, isOpen=${this._flyout.isOpen}`);
        } catch (e) {
            journal(`[flyout] _openFlyout THREW: ${e.message}\n${e.stack}`);
        }
    }

    /**
     * DBus entry point: toggle the flyout for the badged group whose
     * windows have the given WM_CLASS. Exact match, no fallback. Sticky:
     * no pointer watchdog — stays open until toggled again, CloseFlyout,
     * a row click, or it is re-anchored on the next re-render.
     */
    _toggleGroupFlyout(workspaceIndex, wmClass = '') {
        if (this._dragActive()) return false;

        const entry = Model.buildModel().workspaces[workspaceIndex];
        if (!entry) return false;

        const group = entry.groups
            .find(g => g.windows.some(w => w.get_wm_class() === wmClass));
        if (!group || group.count < 2) {
            journal(`[flyout.dbus] no badged group with wmclass "${wmClass}" on ws ${workspaceIndex}`);
            return false;
        }

        const key = `${workspaceIndex}::${group.appId}`;
        if (this._flyout.isOpen && this._flyout.anchorKey === key) {
            this._closeGroupFlyout();
            return false;
        }

        let anchor = null, shell = null;
        for (const [actor, info] of this._groupIcons)
            if (info.group.key === group.key) { anchor = actor; shell = info.shell; break; }
        if (!anchor) {
            // Group rendered expanded (one icon per window): anchor to the
            // app's first window icon instead of refusing.
            const ctrl = this._controllers.get(`${group.windows[0].get_id()}:${entry.index}`);
            if (ctrl) {
                anchor = ctrl.actor;
                shell = this._entries.find(s => s.entry.index === entry.index) ?? null;
            }
        }
        if (!anchor || !shell) {
            journal(`[flyout.dbus] no anchor actor for ${group.key}`);
            return false;
        }

        this._openFlyout(shell, group, anchor, { sticky: true, anchorKey: key });
        if (this._flyout.isOpen) {
            this._stickyRequest = { workspaceIndex, wmClass };
            return true;
        }
        return false;
    }

    _closeGroupFlyout() {
        this._stickyRequest = null;
        if (!this._flyout.isOpen) return false;
        journal('[flyout.dbus] close');
        this._flyout.close(true);
        return true;
    }

    _dropIntoGroup(shell, source, item) {
        const { entry } = shell;
        const win = source?._dragPayload?.window;
        if (!win) return false;

        const sameWs = win.get_workspace() === entry.workspace;
        Model.placeWindow(win, entry.workspace, item ? item.index : null);

        if (!sameWs) {
            const monitorIndex = Main.layoutManager.findIndexForActor(shell.actor);
            if (monitorIndex !== win.get_monitor()) win.move_to_monitor(monitorIndex);
            win.change_workspace(entry.workspace);
        }

        _settleWindow = win;
        Model.setIntent({ drag: null, hintWs: null });
        return true;
    }

    // ==================== EXTERNAL DRAG ====================
    _onExternalChanged(state) {
        if (state) {
            if (this._extPollId) return;
            this._startExternalPoll();
        } else {
            this._stopExternalPoll(true);
        }
    }

    _stopExternalPoll(teardown) {
        if (this._extPollId) { this._scope.cancel(this._extPollId); this._extPollId = 0; }
        if (this._geoRelogId) { this._scope.cancel(this._geoRelogId); this._geoRelogId = 0; }
        if (!teardown) return;

        this._cancelExtRow();
        this._cancelExtWorkspace();
        this._cancelDwell();
        this._flyout?.close(true);

        this._extRowKey = null;
        this._extWsKey = null;
        this._extGroupKey = null;

        Model.setIntent({ hintWs: null });
        this._queueRender();
    }

    _startExternalPoll() {
        this._cancelDwell();
        this._flyout.close(true);

        this._extRowKey = null;
        this._extSawButton = false;
        this._extWsKey = null;
        this._extGroupKey = null;
        this._extLogAt = 0;
        if (this._geoRelogId) { this._scope.cancel(this._geoRelogId); this._geoRelogId = 0; }

        journal('[ext] poll START');
        this._logGroupIcons();

        if (this._extPollId) return;
        this._extPollId = this._scope.timeout(EXT_TICK_MS, () => this._externalTick());
    }

    _logGroupIcons(attempt = 0) {
        journal(`[ext] groupIcons=${this._groupIcons.size}`);

        let pending = false;
        for (const [actor, info] of this._groupIcons) {
            const [x, y] = actor.get_transformed_position();
            const [w, h] = actor.get_transformed_size();
            const ready = [x, y, w, h].every(Number.isFinite);
            if (!ready) pending = true;

            // get_transformed_*() return NaN until the actor has been
            // allocated (next stage paint) — visible/onStage stay true,
            // which is why the old line looked contradictory.
            journal(`[ext]   group ${info.group.appName} x${info.group.count} ` +
                `pos=${fmtNum(x)},${fmtNum(y)} size=${fmtNum(w)}x${fmtNum(h)} ` +
                `visible=${actor.visible} mapped=${actor.is_mapped()} onStage=${!!actor.get_stage()}`);

            if (!ready) continue;   // ancestor chain is noise until allocation lands

            let a = actor;
            let depth = 0;
            while (a && depth++ < 12) {
                let line;
                try {
                    const [px, py] = a.get_transformed_position();
                    const [aw, ah] = a.get_size();
                    line = `[ext]     ${a.constructor.name}` +
                        `${a.name ? `(${a.name})` : ''} ` +
                        `pos=${fmtNum(px)},${fmtNum(py)} size=${fmtNum(aw)}x${fmtNum(ah)} ` +
                        `mapped=${a.is_mapped()} scale=${a.scale_x},${a.scale_y}`;
                } catch (e) {
                    line = `[ext]     ${a.constructor?.name} threw: ${e.message}`;
                }
                journal(line);
                a = a.get_parent();
            }
        }

        if (pending && attempt < 3) {
            // Re-log once the frame has been allocated so the journal gets
            // real geometry instead of a NaN line.
            journal(`[ext] geometry pending — re-logging in 120ms (${attempt + 1}/3)`);
            if (this._geoRelogId) this._scope.cancel(this._geoRelogId);
            this._geoRelogId = this._scope.timeout(120, () => {
                this._geoRelogId = 0;
                if (externalDrag.active) this._logGroupIcons(attempt + 1);
                return GLib.SOURCE_REMOVE;
            });
        }
    }

    _externalTick() {
        try {
            return this._externalTickInner();
        } catch (e) {
            journal(`[ext] tick THREW: ${e.message}\n${e.stack}`);
            return GLib.SOURCE_CONTINUE;
        }
    }

    _shellForWindow(window) {
        const ws = window.get_workspace();
        const idx = ws ? ws.index()
            : Model.WorkspaceManager.get_active_workspace_index();
        return this._entries.find(s => s.entry.index === idx)
            ?? this._entries[0] ?? null;
    }

    _externalTickInner() {
        if (!externalDrag.active) {
            this._extPollId = 0;
            this._stopExternalPoll(true);
            return GLib.SOURCE_REMOVE;
        }

        const [x, y, mods] = global.get_pointer();

        if (y < Main.panel.height + 4) {
            journal(`[ext] PANEL x=${x} y=${y} ` +
                `overGroup=${!!this._groupIconAt(x, y)} ` +
                `overWinIcon=${!!this._windowIconAt(x, y)} ` +
                `overThumb=${!!this._thumbnailAt(x, y)}`);
        }

        const nowMs = GLib.get_monotonic_time() / 1000;
        if (nowMs - (this._extLogAt ?? 0) > 500) {
            this._extLogAt = nowMs;

            const picked = global.stage.get_actor_at_pos(
                Clutter.PickMode.REACTIVE, x, y);

            journal(`[ext] tick x=${x} y=${y} mods=${mods} held=${buttonHeld(mods)} ` +
                `sawButton=${this._extSawButton} ` +
                `flyoutOpen=${this._flyout.isOpen} sticky=${this._flyout.sticky} ` +
                `overGroup=${!!this._groupIconAt(x, y)} ` +
                `overWinIcon=${!!this._windowIconAt(x, y)} ` +
                `overThumb=${!!this._thumbnailAt(x, y)} ` +
                `picked=${picked?.constructor?.name}:${picked?.name ?? ''}`);
        }

        if (buttonHeld(mods)) this._extSawButton = true;
        else if (this._extSawButton) {
            journal('[ext] button released -> end');
            externalDrag.end('button-released');
            return GLib.SOURCE_REMOVE;
        }

        // (1) Flyout is up: hover a row to preview that window, leave to
        // restore. The flyout is NOT closed on preview — the user can
        // move between rows to compare, or drag back out to cancel.
        if (this._flyout.isOpen) {
            const hit = this._flyout.pointerAt(x, y);
            if (hit?.item?.window) this._trackExternalRow(hit.item.window);
            else this._cancelExtRow();
            return GLib.SOURCE_CONTINUE;
        }

        // (2) A badged app group — geometric hit test, so this works even
        // when stage picking is unreliable during an external drag.
        const group = this._groupIconAt(x, y);
        if (group) {
            if (this._extGroupKey !== group.group.key) {
                this._extGroupKey = group.group.key;
                const [gx, gy] = group.actor.get_transformed_position();
                journal(`[ext] group hit ${group.group.appName} x${group.group.count} ` +
                    `key=${group.group.key} anchorPos=${fmtNum(gx)},${fmtNum(gy)}`);
            }
            this._dwellGroup(group.shell, group.group, group.actor);
            this._cancelExtRow();
            return GLib.SOURCE_CONTINUE;
        }
        if (this._extGroupKey) {
            this._extGroupKey = null;
            journal('[ext] group left');
        }

        // (3) A plain window icon — dwell to open the flyout when the app
        // has siblings on this workspace, otherwise fall back to the
        // classic auto-activate reveal.
        const ctrl = this._windowIconAt(x, y);
        if (ctrl) {
            const shell = this._shellForWindow(ctrl.window);
            const ownerGroup = shell?.entry.groups
                .find(g => g.windows.includes(ctrl.window));
            if (shell && ownerGroup && ownerGroup.count >= 2) {
                this._dwellGroup(shell, ownerGroup, ctrl.actor);
            } else {
                ctrl.notifyExternalDrag();
                this._cancelDwell();
            }
            this._cancelExtRow();
            return GLib.SOURCE_CONTINUE;
        }

        this._cancelDwell();

        // (4) A workspace thumbnail
        this._maybeDwellWorkspace(x, y);
        return GLib.SOURCE_CONTINUE;
    }

    _trackExternalRow(win) {
        const key = String(win.get_id());
        if (this._extRowKey === key) return;

        this._extRowKey = key;
        this._cancelExtRowTimer();

        this._extRowTimer = this._scope.timeout(EXT_PREVIEW_MS, () => {
            this._extRowTimer = 0;
            this._previewExternalWindow(win);
            return GLib.SOURCE_REMOVE;
        });
    }

    _cancelExtRow() {
        this._extRowKey = null;
        this._cancelExtRowTimer();
    }

    _cancelExtRowTimer() {
        if (this._extRowTimer) { this._scope.cancel(this._extRowTimer); this._extRowTimer = 0; }
    }

    /**
     * Hover-to-preview: maximize and focus the target so the foreign drag
     * has an obvious drop target. Per request the window STAYS maximized
     * afterwards — nothing unmaximizes it on row switch, leave, or drag
     * end.
     */
    _previewExternalWindow(win) {
        try {
            const time = global.get_current_time();
            if (win.minimized) win.unminimize();
            if (win.get_maximized() !== Meta.MaximizeFlags.BOTH)
                win.maximize(Meta.MaximizeFlags.BOTH);
            const ws = win.get_workspace();
            if (ws) ws.activate_with_focus(win, time);
            else win.activate(time);
            journal(`[ext] preview ${win.get_id()} "${win.get_title()}" (kept maximized)`);
        } catch (e) {
            journal(`[WorkspaceIndicator] preview failed: ${e.message}`);
        }
    }

    _maybeDwellWorkspace(x, y) {
        const thumb = this._thumbnailAt(x, y);
        if (!thumb || thumb.workspace === Model.WorkspaceManager.get_active_workspace()) {
            this._cancelExtWorkspace();
            return;
        }

        const key = `ws:${thumb.index}`;
        if (this._extWsKey !== key) {
            this._extWsKey = key;
            this._cancelExtWorkspace();
            this._extWsKey = key;
            this._extWsTimer = this._scope.timeout(EXT_WS_DWELL_MS, () => {
                this._extWsTimer = 0;
                if (!externalDrag.active) return GLib.SOURCE_REMOVE;
                const ws = Model.WorkspaceManager.get_workspace_by_index(thumb.index);
                try { ws?.activate(global.get_current_time()); } catch (e) { /* gone */ }
                return GLib.SOURCE_REMOVE;
            });
        }

        Model.setIntent({ hintWs: thumb.index });
    }

    _cancelExtWorkspace() {
        if (this._extWsTimer) { this._scope.cancel(this._extWsTimer); this._extWsTimer = 0; }
        this._extWsKey = null;
    }

    // ---------- hit testing ----------
    // Geometric tests over our own tracked actors instead of
    // global.stage.get_actor_at_pos(): picking can be intercepted by
    // drag-feedback chrome during an external drag, and unallocated actors
    // pick as nothing. `_actorHit` also rejects NaN geometry for free —
    // every comparison against NaN is false.
    _actorHit(x, y, actor) {
        if (!actor || !actor.get_stage() || !actor.visible) return false;
        const [ax, ay] = actor.get_transformed_position();
        const [aw, ah] = actor.get_transformed_size();
        return x >= ax && x <= ax + aw && y >= ay && y <= ay + ah;
    }

    _groupIconAt(x, y) {
        for (const [actor, info] of this._groupIcons)
            if (this._actorHit(x, y, actor)) return { ...info, actor };
        return null;
    }

    _windowIconAt(x, y) {
        for (const ctrl of this._controllers.values())
            if (this._actorHit(x, y, ctrl.actor)) return ctrl;
        return null;
    }

    // ==================== HINTS / MENUS / DROPS ====================
    _syncHint() {
        const s = this._settings;
        const wsIndex = Model.intent.hintWs;
        const target = (wsIndex !== null && this._thumbsBox)
            ? this._thumbsBox.get_child_at_index(wsIndex)
            : null;
        if (!target || !s.get_boolean('show-workspace-names')) {
            Interactions.getTooltip().hide();
            return;
        }
        Interactions.getTooltip().showAbove(
            target,
            Meta.prefs_get_workspace_name(wsIndex),
            s.get_int('tooltip-font-size'));
    }

    _openThumbnailMenu(anchor, workspace) {
        const windows = Model.getModelWindows(workspace);
        Interactions.openContextMenu(anchor, menu => {
            menu.addAction('Close all windows on all workspaces', () => {
                const now = global.get_current_time();
                for (const w of Model.getModelWindows(null)) w.delete(now);
            });
            if (windows.length > 0) {
                menu.addAction(`Close all windows except workspace ${workspace.index()}`, () => {
                    const now = global.get_current_time();
                    for (const w of Model.getModelWindows(null).filter(w => w.get_workspace() !== workspace))
                        w.delete(now);
                });
                menu.addAction(`Close all windows on workspace ${workspace.index()}`, () => {
                    const now = global.get_current_time();
                    for (const w of windows) w.delete(now);
                });
            }
        });
    }

    _dropOnWorkspace(thumb, entry, source, index, dragActor) {
        Model.setIntent({ hintWs: null });

        const win = source?._dragPayload?.window ?? null;
        if (!win) return true;

        const sameWs = win.get_workspace() === entry.workspace;

        if (dragActor && dragActor !== source._actor) {
            dragActor.get_parent()?.remove_child(dragActor);
            try { dragActor.destroy(); } catch (e) { /* ignore */ }
        }

        if (sameWs && index === null) { Model.setIntent({ drag: null }); return true; }

        Model.placeWindow(win, entry.workspace, index);
        if (!sameWs) {
            const monitorIndex = Main.layoutManager.findIndexForActor(thumb);
            if (monitorIndex !== win.get_monitor()) win.move_to_monitor(monitorIndex);
            win.change_workspace(entry.workspace);
        }
        _settleWindow = win;
        Model.setIntent({ drag: null });
        return true;
    }

    _ensureLeavePoll() {
        if (this._leavePollId) return;
        this._leavePollId = this._scope.timeout(100, () => {
            const [px, py, mods] = global.get_pointer();
            const held = buttonHeld(mods) || externalDrag.active;
            if (!held || !this._thumbnailAt(px, py)) {
                this._leavePollId = 0;
                this._cancelDwell();
                Model.setIntent({ hintWs: null });
                this._ghost.clear();
                return GLib.SOURCE_REMOVE;
            }
            return GLib.SOURCE_CONTINUE;
        });
    }

    _thumbnailAt(x, y) {
        if (!this._thumbsBox) return null;
        for (const thumb of this._thumbsBox.get_children()) {
            if (!thumb.get_stage() || !thumb._wsEntry) continue;
            const [tx, ty] = thumb.get_transformed_position();
            const [tw, th] = thumb.get_transformed_size();
            if (x >= tx && x <= tx + tw && y >= ty && y <= ty + th)
                return { ...thumb._wsEntry, actor: thumb };
        }
        return null;
    }

    // ==================== LIFECYCLE ====================
    destroy() {
        this._cancelDwell();
        this._stopExternalPoll(true);
        this._extUnsub?.();
        this._extUnsub = null;

        this._renderCoalescer.cancel();
        Model.onIntentChange(null);
        Model.onWorldChange(null);
        Model.resetIntent();

        this._flyout?.destroy();
        this._flyout = null;

        for (const ctrl of this._controllers.values()) ctrl.destroy();
        this._controllers.clear();
        this._groupIcons.clear();

        this._ghost.clear();
        this._ghost = null;
        _instance = null;
        _settleWindow = null;

        this._scope.dispose();
        this._mainBox?.destroy();
        Main.panel.set_offscreen_redirect(Clutter.OffscreenRedirect.ALWAYS);
        super.destroy();
    }
}

// ==================== TITLE BAR MOVE MONITOR ====================
export class TitleBarMoveMonitor {
    constructor() {
        this._currentDragWindow = null;
        this._lastSwitchedWorkspace = null;
        this._cursorTracker = global.backend.get_cursor_tracker();
        this._cursorPositionId = 0;

        this._beginId = Model.Display.connect('grab-op-begin',
            (display, window, op) => this._onGrabOpBegin(window, op));
        this._endId = Model.Display.connect('grab-op-end',
            (display, window, op) => this._onGrabOpEnd(window, op));
        this._sessionModeId = Main.sessionMode.connect('updated', () => {
            if (Main.sessionMode.currentMode === 'unlock-dialog') {
                journal('[TitleBarMoveMonitor] session locked, resetting drag state');
                this.reset();
            }
        });
    }

    reset() {
        this._currentDragWindow = null;
        this._lastSwitchedWorkspace = null;
        this._stopGrabTracking();
        Model.setIntent({ hintWs: null });
    }

    _isMoveOp(op) { return op === Meta.GrabOp.MOVING || op === Meta.GrabOp.KEYBOARD_MOVING; }

    _onGrabOpBegin(window, op) {
        if (!this._isMoveOp(op)) return;
        this._currentDragWindow = window;
        this._lastSwitchedWorkspace = null;
        this._startGrabTracking();
    }

    _onGrabOpEnd(window, op) {
        const grabbed = this._currentDragWindow;
        this._currentDragWindow = null;
        this._lastSwitchedWorkspace = null;
        this._stopGrabTracking();
        Model.setIntent({ hintWs: null });

        if (!grabbed || grabbed !== window || !this._isMoveOp(op)) return;

        const [x, y] = this._getPointerPosition();
        const target = thumbnailAt(x, y);
        if (target) this._moveWindowTo(target, window);
    }

    _startGrabTracking() {
        this._stopGrabTracking();
        if (!this._cursorTracker) return;
        this._cursorPositionId = this._cursorTracker.connect('position-invalidated',
            () => this._onCursorPositionInvalidated());
    }

    _stopGrabTracking() {
        if (this._cursorPositionId && this._cursorTracker) {
            this._cursorTracker.disconnect(this._cursorPositionId);
            this._cursorPositionId = 0;
        }
    }

    _onCursorPositionInvalidated() {
        if (!this._currentDragWindow) return;

        const [x, y] = this._getPointerPosition();
        const thumb = thumbnailAt(x, y);
        Model.setIntent({ hintWs: thumb ? thumb.index : null });
        if (!thumb) return;

        const ws = thumb.workspace;
        if (ws === Model.WorkspaceManager.get_active_workspace() || ws === this._lastSwitchedWorkspace) return;
        this._moveWindowTo(thumb, this._currentDragWindow);
    }

    _getPointerPosition() {
        if (this._cursorTracker) {
            try {
                const [point] = this._cursorTracker.get_pointer();
                return [point.x, point.y];
            } catch (e) { journal(`[TitleBarMoveMonitor] get_pointer failed: ${e}`); }
        }
        return global.get_pointer();
    }

    _moveWindowTo(thumb, window) {
        const ws = thumb.workspace;
        const alreadyThere = window.get_workspace() === ws;

        if (!alreadyThere) {
            Model.placeWindow(window, ws, null);
            settleOnNextRender(window);
        }
        const monitorIndex = Main.layoutManager.findIndexForActor(thumb.actor);
        if (monitorIndex !== window.get_monitor()) window.move_to_monitor(monitorIndex);
        window.change_workspace(ws);
        ws.activate(global.get_current_time());
        this._lastSwitchedWorkspace = ws;
    }

    destroy() {
        this._stopGrabTracking();
        if (this._beginId) { Model.Display.disconnect(this._beginId); this._beginId = null; }
        if (this._endId) { Model.Display.disconnect(this._endId); this._endId = null; }
        if (this._sessionModeId) { Main.sessionMode.disconnect(this._sessionModeId); this._sessionModeId = null; }
        this._currentDragWindow = null;
    }
}

// ==================== DISCRETE FOCUS STRIP ====================
// Prototype neighborOpacity(): the selection is fully opaque, the first
// neighbour steps down, the second is gone by x = 2.8. No scaling — rows
// keep one uniform size so the strip reads as beads on a wire.
function neighborOpacity(x) {
    if (x <= 1) return 0.92;
    if (x <= 2) return 0.92 - 0.40 * (x - 1);          // x = 2 -> 0.52
    return Math.max(0, 0.52 * (1 - (x - 2) / 0.8));    // gone by x = 2.8
}

// ==================== LAYOUT ====================
// Three independent layout functions, one per region of the screen. Each
// region owns its geometry exclusively; no cross-references except through
// derived fields. The rail reserves a `gutter` on the right and the band is
// inset by that gutter on both sides, so the band never crosses the rail.
//
// The type scale is one number — `overlay.centerLabelSize` — and every text
// size in the selector derives from it.

function computeOverlayLayout(monitor) {
    const W = monitor.width, H = monitor.height;
    const centerLabelSize = Math.max(40, Math.min(92, Math.round(H * 0.072)));
    return {
        x: monitor.x, y: monitor.y, width: W, height: H,
        centerLabelSize,
    };
}

function computeRailLayout(overlay) {
    const width = 48;
    const margin = 14;
    const pad = 10;
    const height = Math.round(overlay.height * 0.80);
    return {
        x: overlay.width - margin - width,
        y: Math.round((overlay.height - height) / 2),
        width, height, margin, pad,
        gutter: margin + width + margin,
    };
}

function computeBandLayout(overlay, rail) {
    const selectedHeight = Math.round(overlay.centerLabelSize * 1.95);
    const searchHeight = Math.round(overlay.centerLabelSize * 1.1) + 24;
    const height = selectedHeight + searchHeight + 40;
    // Symmetric gutters: left inset equals right inset (= rail.gutter), so
    // the band is centered on the overlay but never touches the rail.
    const width = Math.max(400, overlay.width - rail.gutter * 2);
    return {
        x: Math.round((overlay.width - width) / 2),
        y: Math.round((overlay.height - height) / 2),
        width, height,
        selectedHeight, searchHeight,
        searchWidth: Math.min(1100, Math.round(width * 0.85)),
    };
}

function computeRowLayout(overlay, band) {
    const rowLabelSize = Math.round(overlay.centerLabelSize * 0.88);
    return {
        rowLabelSize,
        rowIconSize: Math.round(rowLabelSize * 1.15),
        slotH: Math.round(rowLabelSize * 1.55),
        // Row text column stays inside the band's horizontal extent so a
        // long title truncates before reaching the rail gutter.
        rowTextMax: Math.round(band.width * 0.85),
        // Rail tip max width — overlay-relative, not band-relative, because
        // the tip hangs off to the left of the rail.
        tipWidth: Math.min(700, Math.round(overlay.width * 0.45)),
    };
}

// ==================== launcher layout ====================
function computeLauncherLayout(monitor) {
    const margin = Math.max(24, Math.round(monitor.height * 0.04));
    const searchHeight = 160;
    const searchWidth = Math.min(1100, monitor.width - margin * 2);

    const searchY = margin;
    const pinnedY = searchY + searchHeight + 16;
    const pinnedHeight = 240;

    const resultsY = pinnedY + pinnedHeight + 16;
    const resultsHeadingHeight = 70;
    const resultsHeight = Math.max(240, monitor.height - resultsY - margin);

    const fullWidth = monitor.width - margin * 2;
    const innerWidth = fullWidth - 40;
    const columns = Math.max(2, Math.min(8, Math.floor(innerWidth / 240)));
    const tileWidth = Math.floor(innerWidth / columns);

    return {
        margin, searchWidth,
        searchX: Math.round((monitor.width - searchWidth) / 2), searchY,
        fullWidth, fullX: margin,
        pinnedY, pinnedHeight,
        resultsY, resultsHeadingHeight, resultsHeight,
        columns, tileWidth, tileHeight: 340, iconSize: 144,
        pinnedTileWidth: 200, pinnedTileHeight: 200, pinnedIconSize: 160,
    };
}

// ==================== shared overlay base ====================
export class SearchOverlayBase {
    static _current = null;

    static get current() {
        if (SearchOverlayBase._current?._closed) SearchOverlayBase._current = null;
        return SearchOverlayBase._current;
    }

    static closeCurrent() {
        const overlay = SearchOverlayBase.current;
        if (!overlay) return false;
        overlay._close();
        return true;
    }

    constructor(settings) {
        this._settings = settings;
        this._results = [];
        this._selectedIndex = -1;
        this._query = '';
        this._initialKey = null;
        this._modalGrab = null;
        this._closed = false;
        this._closing = false;
        this._focusGuard = false;

        this._scope = new Scheduling.Scope();
        this._rowsScope = new Scheduling.Scope();
    }

    _trackFullscreen() { return true; }
    _computeResults(query) { return []; }
    _beforeResults(query, unchanged) { }
    _renderResults(results) { }
    _onSelectionChanged(index) { }
    _getRestoreKey(item) { return null; }
    _activateResult(index) { }
    _activateFocused() { if (this._selectedIndex >= 0) this._activateResult(this._selectedIndex); }
    _initialKeyFocus() { this._entry.grab_key_focus(); }
    _handleEscape() { this._close(); }
    _handleExtraKeys(event, symbol) { return undefined; }
    _onBeforeClose() { }
    _shouldCloseOnWindowRemoved(window) { return true; }

    _wireEntry(entry) {
        this._entry = entry;
        this._scope.connect(entry.clutter_text, 'text-changed', () => this._onSearchChanged());
        this._scope.connect(entry.clutter_text, 'key-press-event', (a, e) => this._onKeyPress(e));
    }

    _open() {
        Main.layoutManager.addChrome(this._container, {
            affectsInputRegion: true, trackFullscreen: this._trackFullscreen(),
        });
        Main.layoutManager.uiGroup.set_child_above_sibling(this._container, null);
        this._modalGrab = Main.pushModal(this._container, { actionMode: Shell.ActionMode.NORMAL });
        this._initialKeyFocus();
        SearchOverlayBase._current = this;

        this._scope.connect(this._container, 'key-press-event', (a, e) => this._onKeyPress(e));

        this._scope.connect(Model.Display, 'notify::focus-window', () => {
            if (this._closed || this._closing || this._focusGuard) return;
            this._close();
        });
        this._scope.connect(Model.Display, 'window-created', () => {
            if (!this._closed && !this._closing) this._close();
        });
        this._scope.connect(global.window_manager, 'destroy', (wm, actor) => {
            const window = actor.meta_window;
            if (!window || this._closed || this._closing || !this._shouldCloseOnWindowRemoved(window)) return;
            this._close();
        });
    }

    _close() {
        if (this._closed) return;
        this._closed = true;
        Interactions.getTooltip().hide();
        this._onBeforeClose();

        if (this._modalGrab) { Main.popModal(this._modalGrab); this._modalGrab = null; }
        this._rowsScope.dispose();
        this._scope.dispose();

        Main.layoutManager.removeChrome(this._container);
        this._container.destroy();
    }

    _onSearchChanged() {
        if (this._closed) return;
        const query = this._entry.get_text().trim();
        const unchanged = query === this._query;
        this._query = query;
        this._beforeResults(query, unchanged);
        this._setResults(this._computeResults(query), unchanged);
    }

    _setResults(results, keepSelection = true) {
        const prev = this._results[this._selectedIndex];
        let previousKey = keepSelection && prev ? this._getRestoreKey(prev) : null;

        if (previousKey === null && this._initialKey !== null) previousKey = this._initialKey;
        this._initialKey = null;

        this._results = results;
        this._rowsScope.dispose();
        this._rowsScope = new Scheduling.Scope();
        this._selectedIndex = -1;
        this._renderResults(results);

        if (results.length === 0) { this._onSelectionChanged(-1); return; }
        const restored = previousKey !== null
            ? results.findIndex(item => this._getRestoreKey(item) === previousKey) : -1;
        this._selectIndex(restored >= 0 ? restored : 0);
    }

    _selectIndex(index) {
        if (this._closed || index < 0 || index >= this._results.length) return;
        if (index === this._selectedIndex) return;
        this._selectedIndex = index;
        this._onSelectionChanged(index);
    }

    _onKeyPress(event) {
        if (this._closed) return Clutter.EVENT_PROPAGATE;
        const symbol = event.get_key_symbol();
        const handled = this._handleExtraKeys(event, symbol);
        if (handled !== undefined) return handled;

        switch (symbol) {
            case Clutter.KEY_Escape:
                this._handleEscape();
                return Clutter.EVENT_STOP;
            case Clutter.KEY_Down:
                this._selectIndex(this._selectedIndex + 1);
                return Clutter.EVENT_STOP;
            case Clutter.KEY_Up:
                this._selectIndex(this._selectedIndex - 1);
                return Clutter.EVENT_STOP;
            case Clutter.KEY_Return:
            case Clutter.KEY_KP_Enter:
                this._activateFocused();
                return Clutter.EVENT_STOP;
            default:
                return Clutter.EVENT_PROPAGATE;
        }
    }
}

        this._buildUI();
        this._initialKey = this._focusedWindow ?? null;
        this._onSearchChanged();
        this._open();
    }

    _trackFullscreen() { return false; }
    _getRestoreKey(item) { return item.window; }

    _shouldCloseOnWindowRemoved(window) {
        return this._results.some(item => item.window === window);
    }

    _matchItem(query, item) {
        const r = Fuzzy.fuzzyMatch(query, `${item.title} ${item.appName}`);
        if (!r.matched) return r;
        return { ...r, indices: new Set([...r.indices].filter(i => i < item.title.length)) };
    }

    _computeResults(query) {
        if (!query) return this._items.map(item => ({ ...item, indices: null }));

        const scored = [];
        for (const item of this._items) {
            const r = this._matchItem(query, item);
            if (r.matched) scored.push({ ...item, score: r.score, indices: r.indices });
        }
        return scored.sort((a, b) => b.score - a.score);
    }

    _getAppName(window) {
        const app = Shell.WindowTracker.get_default().get_window_app(window);
        return app ? app.get_name() : (window.get_wm_class() || 'Unknown');
    }

    _buildUI() {
        const monitor = Main.layoutManager.primaryMonitor;
        this._monitor = monitor;
        this._overlayL = computeOverlayLayout(monitor);
        this._railL = computeRailLayout(this._overlayL);
        this._bandL = computeBandLayout(this._overlayL, this._railL);
        this._rowL = computeRowLayout(this._overlayL, this._bandL);

        const OV = this._overlayL;
        const BAND = this._bandL;

        this._container = UI.createFullscreenContainer(monitor, 'window-selector-overlay');
        this._container.add_style_class_name('search-mode');

        const size = { width: OV.width, height: OV.height };
        this._tint = new St.Widget({ style_class: 'selector-tint', reactive: false, opacity: 255, ...size });
        this._listLayer = new St.Widget(size);

        const { band, titleBin } = UI.createSelectionBand(BAND);
        this._bandBin = titleBin;
        this._band = band;

        const search = UI.createSearchField({ width: BAND.searchWidth, hint: 'Type to search…' });
        this._wireEntry(search.entry);

        // Prototype: bare underline field sized off the band, no magnifier.
        search.entry.set_style(`font-size: ${Math.round(OV.centerLabelSize * 0.55)}px;`);
        if (search.entry.primary_icon) search.entry.primary_icon.visible = false;
        this._searchRoot = search.actor;

        band.add_child(new St.Bin({
            x_expand: true, x_align: Clutter.ActorAlign.CENTER, child: search.actor,
        }));

        const hint = UI.createSelectorHint(OV);
        this._hintLabel = hint.label;

        for (const actor of [this._tint, this._listLayer, band, hint.actor])
            this._container.add_child(actor);

        this._buildRail();

        this._scope.connect(this._container, 'scroll-event', (a, e) => this._onScroll(e));
        this._scope.connect(this._container, 'button-release-event', (a, e) => this._onClick(e));

        this._scope.connect(global.stage, 'notify::key-focus', () => {
            if (this._closed || this._closing) return;
            if (global.stage.get_key_focus() === this._entry.clutter_text) return;
            this._refocusEntry();
        });
    }

    // ----- rail widget tree -----
    _buildRail() {
        const RL = this._railL;
        const OV = this._overlayL;
        const ROW = this._rowL;

        this._railH = RL.height;
        this._railPad = RL.pad;

        this._rail = new St.Widget({
            style_class: 'selector-rail',
            reactive: true, track_hover: true,
            x: RL.x, y: RL.y,
            width: RL.width, height: RL.height,
        });

        const track = new St.Widget({
            style_class: 'selector-rail-track',
            x: Math.round((RL.width - 2) / 2),
            y: 0, width: 2, height: RL.height,
        });
        this._rail.add_child(track);

        this._railTicksBox = new St.Widget({ x: 0, y: 0, width: RL.width, height: RL.height });
        this._rail.add_child(this._railTicksBox);

        this._railMarker = new St.Widget({
            style_class: 'selector-rail-marker',
            x: Math.round((RL.width - 32) / 2),
            y: 0, width: 32, height: 8,
        });
        this._rail.add_child(this._railMarker);

        const tipFontSize = Math.round(OV.centerLabelSize * 0.45);
        this._railTip = new St.Label({
            style_class: 'selector-rail-tip', text: '',
            x: 0, y: 0, width: ROW.tipWidth, opacity: 0,
        });
        this._railTip.set_style(`font-size: ${tipFontSize}px; font-weight: 600;`);
        this._railTip.clutter_text.set_ellipsize(Pango.EllipsizeMode.START);
        this._railTip.clutter_text.set_line_wrap(false);

        this._container.add_child(this._rail);
        this._container.add_child(this._railTip);

        this._wireRail();
    }

    _wireRail() {
        const rail = this._rail;
        const scope = this._scope;

        scope.connect(rail, 'button-press-event', (a, e) => {
            if (e.get_button() !== Clutter.BUTTON_PRIMARY) return Clutter.EVENT_PROPAGATE;
            this._railDragging = true;
            this._acquireRailGrab();
            this._railPointerFromEvent(e);
            return Clutter.EVENT_STOP;
        });

        scope.connect(rail, 'motion-event', (a, e) => {
            if (this._railDragging) {
                this._railPointerFromEvent(e);
                return Clutter.EVENT_STOP;
            }
            const [, y] = e.get_coords();
            const [, ry] = rail.get_transformed_position();
            const idx = this._railIndexFromY(y - ry);
            const item = this._results[idx];
            if (item) this._showRailTip(idx, item);
            return Clutter.EVENT_PROPAGATE;
        });

        scope.connect(rail, 'button-release-event', (a, e) => {
            if (!this._railDragging) return Clutter.EVENT_PROPAGATE;
            this._railDragging = false;
            this._releaseRailGrab();
            this._hideRailTip();
            return Clutter.EVENT_STOP;
        });

        scope.connect(rail, 'leave-event', () => {
            if (!this._railDragging) this._hideRailTip();
            return Clutter.EVENT_PROPAGATE;
        });
    }

    _acquireRailGrab() {
        try {
            const grab = Clutter.grab?.(this._rail);
            if (grab) { this._railGrab = { dismiss: () => grab.dismiss() }; return; }
        } catch (e) {
            journal(`[rail] Clutter.grab failed: ${e.message}`);
        }
        try {
            const modal = Main.pushModal(this._rail, { actionMode: Shell.ActionMode.NORMAL });
            this._railGrab = { modal };
        } catch (e) {
            journal(`[rail] pushModal fallback failed: ${e.message}`);
            this._railGrab = null;
            this._railDragging = false;
        }
    }

    _releaseRailGrab() {
        const g = this._railGrab;
        this._railGrab = null;
        if (!g) return;
        if (g.modal) { try { Main.popModal(g.modal); } catch (e) { /* already popped */ } }
        else { try { g.dismiss(); } catch (e) { /* already dismissed */ } }
    }

    _railY(i) {
        const n = this._results.length;
        const span = Math.max(1, this._railH - this._railPad * 2);
        if (n <= 1) return this._railH / 2;
        return this._railPad + (i / (n - 1)) * span;
    }

    _railIndexFromY(py) {
        const n = this._results.length;
        if (n <= 1) return 0;
        const span = Math.max(1, this._railH - this._railPad * 2);
        return clamp(Math.round(((py - this._railPad) / span) * (n - 1)), 0, n - 1);
    }

    _buildRailTicks() {
        if (!this._railTicksBox || !this._rail) return;
        this._railTicksBox.destroy_all_children();
        this._railTicks = [];

        const n = this._results.length;
        this._rail.visible = n > 0;
        if (n === 0) return;

        const W = this._railL.width;
        for (let i = 0; i < n; i++) {
            const item = this._results[i];
            const tick = new St.Widget({
                style_class: 'selector-rail-tick',
                x: Math.round((W - 12) / 2),
                y: Math.round(this._railY(i) - 2),
                width: 12, height: 4,
            });
            tick.set_style(`background-color: ${appColor(item.appName || '')};`);
            this._railTicksBox.add_child(tick);
            this._railTicks.push(tick);
        }
        this._updateRailMarker();
    }

    _updateRailMarker() {
        if (!this._railMarker || !this._results.length) return;
        const p = clamp(this._scrollPos, 0, this._results.length - 1);
        const lo = Math.floor(p);
        const hi = Math.min(this._results.length - 1, lo + 1);
        const y = this._railY(lo) + (this._railY(hi) - this._railY(lo)) * (p - lo);
        this._railMarker.set_position(this._railMarker.x, Math.round(y - this._railMarker.height / 2));
    }

    _railPointerFromEvent(event) {
        const [, y] = event.get_coords();
        const [, ry] = this._rail.get_transformed_position();
        const idx = this._railIndexFromY(y - ry);
        const item = this._results[idx];
        if (!item) return;
        this._selectIndex(idx);
        this._showRailTip(idx, item);
    }

    _showRailTip(idx, item) {
        if (!this._railTip) return;
        const RL = this._railL, OV = this._overlayL, BAND = this._bandL;
        const text = `${item.appName} — ${item.title}`;
        this._railTip.set_text(text);

        const h = this._railTip.height > 0 ? this._railTip.height : 58;
        const tipX = Math.max(8, RL.x - this._railTip.width - 8);
        let tipY = RL.y + this._railY(idx) - Math.round(h / 2);

        // If the natural position would land inside the band, park the tip
        // above it (or below, if there is no room above).
        if (tipY + h > BAND.y && tipY < BAND.y + BAND.height) {
            const above = BAND.y - h - 8;
            tipY = above >= 8 ? above : Math.min(BAND.y + BAND.height + 8, OV.height - h - 8);
        }
        tipY = clamp(tipY, 8, OV.height - h - 8);

        this._railTip.set_position(tipX, tipY);
        this._railTip.opacity = 255;
    }

    _hideRailTip() {
        if (this._railTip) this._railTip.opacity = 0;
    }

    // ----- rows -----
    _renderResults(results) {
        const OV = this._overlayL, ROW = this._rowL;

        this._releaseRailGrab();
        this._railDragging = false;
        this._hideRailTip();

        this._listLayer.destroy_all_children();
        this._rows = results.map((item, index) => {
            const row = UI.createWindowTitleRow({
                icon: item.icon, title: item.title, indices: item.indices,
                width: OV.width, height: ROW.slotH,
                iconSize: ROW.rowIconSize,
                labelSize: ROW.rowLabelSize,
                maxWidth: ROW.rowTextMax,
            });
            this._rowsScope.connect(row, 'button-press-event', (a, e) => {
                if (e.get_button() !== Clutter.BUTTON_PRIMARY) return Clutter.EVENT_PROPAGATE;
                this._selectIndex(index);
                return Clutter.EVENT_STOP;
            });
            this._rowsScope.connect(row, 'button-release-event', (a, e) => {
                if (e.get_button() !== Clutter.BUTTON_PRIMARY) return Clutter.EVENT_PROPAGATE;
                return Clutter.EVENT_STOP;
            });
            this._listLayer.add_child(row);
            return row;
        });
        this._snap = true;
        this._buildRailTicks();
    }

    // ----- discrete focus strip layout -----
    _layoutRows() {
        const BAND = this._bandL, ROW = this._rowL;
        const pos = this._scrollPos;
        const a = BAND.y - ROW.slotH / 2;
        const b = BAND.y + BAND.height + ROW.slotH / 2;
        const mid = (a + b) / 2;
        const half = (b - a) / 2;

        this._rows.forEach((row, i) => {
            const d = i - pos;
            const x = Math.abs(d);

            if (x >= 2.8) { row.hide(); return; }

            let c, op;
            if (x < 1) {
                // Sliding into the middle: drifts toward the big title and
                // fades out as it arrives.
                c = mid + d * half;
                op = neighborOpacity(1) * ease((x - 0.2) / 0.8);
            } else {
                c = d > 0 ? b + (x - 1) * ROW.slotH : a - (x - 1) * ROW.slotH;
                op = neighborOpacity(x);
            }
            if (op < 0.01) { row.hide(); return; }

            row.show();
            row.set_position(0, Math.round(c - ROW.slotH / 2));
            row.set_opacity(Math.round(op * 255));
            // Uniform size: no fisheye, no per-row scaling.
            row.set_pivot_point(0.5, 0.5);
            row.set_scale(1, 1);
            row.set_reactive(x >= 1);
        });

        this._bandBin.opacity = Math.round(255 * ease(1 - Math.abs(this._selectedIndex - pos)));
        this._updateRailMarker();
    }

    // ----- selection -----
    _onSelectionChanged(index) {
        if (index < 0) {
            this._bandBin.set_child(null);
            this._cancelReveal();
            this._updateHint();
            return;
        }
        const item = this._results[index];
        const OV = this._overlayL, BAND = this._bandL;

        this._bandBin.set_child(UI.createBandTitle({
            icon: item.icon,
            title: item.title,
            appName: (item.appName || '').toUpperCase(),
            width: BAND.width,
            height: BAND.selectedHeight,
            iconSize: Math.round(OV.centerLabelSize * 1.55),
            labelSize: OV.centerLabelSize,
            appSize: Math.max(12, Math.round(OV.centerLabelSize * 0.30)),
        }));

        this._scrollTarget = index;
        if (this._snap) { this._scrollPos = index; this._layoutRows(); this._snap = false; }
        else this._startTween();

        this._scheduleReveal(item);
        this._updateHint();
    }

    _updateHint() {
        if (!this._hintLabel) return;
        const total = this._results.length;
        if (total === 0) {
            this._hintLabel.set_text('No matching windows   •   Esc to Close');
            return;
        }
        this._hintLabel.set_text(
            `${this._selectedIndex + 1} / ${total}   •   ↑↓ Navigate   •   Drag the right rail to jump   •   Enter / Click to Activate   •   Esc to Close`);
    }

    _startTween() {
        if (this._tweenId) return;
        this._lastFrame = GLib.get_monotonic_time() / 1000;
        this._tweenId = this._scope.timeout(16, () => {
            const now = GLib.get_monotonic_time() / 1000;
            const dt = Math.min(now - this._lastFrame, 50);
            this._lastFrame = now;
            const diff = this._scrollTarget - this._scrollPos;
            if (Math.abs(diff) < 0.002) {
                this._scrollPos = this._scrollTarget;
                this._layoutRows();
                this._tweenId = 0;
                return GLib.SOURCE_REMOVE;
            }
            this._scrollPos += diff * (1 - Math.exp(-dt / 70));
            this._layoutRows();
            return GLib.SOURCE_CONTINUE;
        });
    }

    _onScroll(event) {
        let step = 0;
        const dir = event.get_scroll_direction();
        if (dir === Clutter.ScrollDirection.UP) step = -1;
        else if (dir === Clutter.ScrollDirection.DOWN) step = 1;
        else if (dir === Clutter.ScrollDirection.SMOOTH) {
            const [, dy] = event.get_scroll_delta();
            this._scrollAccum += dy;
            if (Math.abs(this._scrollAccum) >= 1) {
                step = Math.sign(this._scrollAccum);
                this._scrollAccum = 0;
            }
        }
        if (!step) return Clutter.EVENT_STOP;

        const now = GLib.get_monotonic_time() / 1000;
        if (now - this._lastScrollStep < 50) return Clutter.EVENT_STOP;
        this._lastScrollStep = now;

        this._selectIndex(clamp(this._selectedIndex + step, 0, this._results.length - 1));
        return Clutter.EVENT_STOP;
    }

    _guardFocus(ms = 1200) {
        this._focusGuard = true;
        if (this._guardTimer) this._scope.cancel(this._guardTimer);
        this._guardTimer = this._scope.timeout(ms, () => {
            this._guardTimer = 0;
            this._focusGuard = false;
            return GLib.SOURCE_REMOVE;
        });
    }

    _scheduleReveal(item) {
        this._cancelReveal();
        this._revealTimer = this._scope.timeout(120, () => {
            this._revealTimer = 0;
            this._revealWindow(item.window);
            return GLib.SOURCE_REMOVE;
        });
    }

    _cancelReveal() {
        if (!this._revealTimer) return;
        this._scope.cancel(this._revealTimer);
        this._revealTimer = 0;
    }

    _revealWindow(win) {
        if (this._closed || this._closing || win === this._revealed) return;
        this._guardFocus();
        this._releaseRevealed();
        try {
            const time = global.get_current_time();
            if (win.minimized) win.unminimize();
            const alreadyMax = win.get_maximized() === Meta.MaximizeFlags.BOTH;
            if (!alreadyMax) { win.maximize(Meta.MaximizeFlags.BOTH); this._maximizedWindow = win; }
            const ws = win.get_workspace();
            if (ws) ws.activate_with_focus(win, time);
            else win.activate(time);
            this._revealed = win;
            this._revealedAny = true;
        } catch (e) {
            journal(`[WindowSearchOverlay] reveal failed: ${e.message}`, true);
        }
        this._refocusEntry();
    }

    _refocusEntry() {
        GLib.idle_add(GLib.PRIORITY_DEFAULT, () => {
            if (this._closed || this._closing) return GLib.SOURCE_REMOVE;
            this._entry.grab_key_focus();
            return GLib.SOURCE_REMOVE;
        });
    }

    _releaseRevealed() {
        const win = this._maximizedWindow;
        this._maximizedWindow = null;
        if (!win) return;
        try { win.unmaximize(Meta.MaximizeFlags.BOTH); } catch (e) { /* window gone */ }
    }

    _restoreOrigin() {
        if (!this._revealedAny) return;
        this._revealedAny = false;
        this._guardFocus();
        const { window: win, workspace } = this._origin;
        const time = global.get_current_time();
        try {
            const ws = win?.get_workspace();
            if (ws) ws.activate_with_focus(win, time);
            else workspace?.activate(time);
        } catch (e) { /* window gone */ }
    }

    // Prototype: first Esc clears the query, second cancels the overlay.
    _handleEscape() {
        if (this._closed || this._closing) return;
        if (this._entry.get_text().length > 0) {
            this._entry.set_text('');
            return;
        }
        this._cancelled = true;
        this._container.add_style_class_name('selector-cancelling');
        GLib.timeout_add(GLib.PRIORITY_DEFAULT, 180, () => {
            if (!this._closed) this._close();
            return GLib.SOURCE_REMOVE;
        });
    }

    _onBeforeClose() {
        this._releaseRailGrab();
        this._railDragging = false;
        this._cancelReveal();
        this._releaseRevealed();
        if (this._cancelled) this._restoreOrigin();
    }

    _handleExtraKeys(event, symbol) {
        const inEntry = global.stage.get_key_focus() === this._entry.clutter_text;
        const state = event.get_state();

        // Prototype: typing anywhere focuses the field and types into it.
        if (!inEntry && !(state & (Clutter.ModifierType.CONTROL_MASK |
            Clutter.ModifierType.MOD1_MASK | Clutter.ModifierType.SUPER_MASK))) {
            if (symbol === Clutter.KEY_BackSpace && this._entry.get_text().length > 0) {
                this._entry.grab_key_focus();
                this._entry.clutter_text.delete_text(this._entry.get_text().length - 1, -1);
                return Clutter.EVENT_STOP;
            }
            const uc = event.get_key_unicode();
            const code = typeof uc === 'string' ? uc.codePointAt(0) : uc;
            if (code && code >= 0x20 && code !== 0x7f) {
                this._entry.grab_key_focus();
                this._entry.clutter_text.insert_unichar(code);
                return Clutter.EVENT_STOP;
            }
        }

        if (symbol === Clutter.KEY_Page_Down) {
            this._selectIndex(Math.min(this._selectedIndex + 5, this._results.length - 1));
            return Clutter.EVENT_STOP;
        }
        if (symbol === Clutter.KEY_Page_Up) {
            this._selectIndex(Math.max(this._selectedIndex - 5, 0));
            return Clutter.EVENT_STOP;
        }
        if (symbol === Clutter.KEY_Home && !inEntry) {
            this._selectIndex(0);
            return Clutter.EVENT_STOP;
        }
        if (symbol === Clutter.KEY_End && !inEntry) {
            this._selectIndex(this._results.length - 1);
            return Clutter.EVENT_STOP;
        }
        return undefined;
    }

    _onClick(event) {
        if (event.get_button() !== Clutter.BUTTON_PRIMARY) return Clutter.EVENT_PROPAGATE;
        if (this._closing) return Clutter.EVENT_STOP;
        const src = event.get_source();
        if (src && this._searchRoot.contains(src)) {
            this._entry.grab_key_focus();
            return Clutter.EVENT_PROPAGATE;
        }
        this._activateFocused();
        return Clutter.EVENT_STOP;
    }

    _activateResult(index) {
        if (this._closed || this._closing) return;
        const item = this._results[index];
        if (!item) return;
        const window = item.window;

        this._closing = true;
        this._cancelReveal();
        this._releaseRevealed();

        try {
            if (window.minimized) window.unminimize();
            const ws = window.get_workspace();
            if (ws) ws.activate_with_focus(window, global.get_current_time());
            else window.activate(global.get_current_time());
        } catch (e) {
            journal(`[WindowSearchOverlay] activate failed: ${e.message}`, true);
        }

        this._container.ease({
            opacity: 0, duration: 220, mode: Clutter.AnimationMode.EASE_OUT_QUAD,
        });
        GLib.timeout_add(GLib.PRIORITY_DEFAULT, 250, () => {
            if (!this._closed) this._close();
            return GLib.SOURCE_REMOVE;
        });
    }

    _close() {
        if (this._closed) return;
        journal('[WindowSearchOverlay] Closing');
        super._close();
    }
}

// ==================== app launcher ====================
const PINNED_ICON_SIZE = 160;

export class AppSearchOverlay extends SearchOverlayBase {
    constructor(settings) {
        super(settings);
        this._favorites = AppFavorites.getAppFavorites();
        this._menuManager = null;
        this._openMenu = null;

        this._justDroppedAppId = null;
        this._apps = null;
        this._pinnedScope = new Scheduling.Scope();
        this._ghost = Interactions.createReorderGhost({ iconSize: () => PINNED_ICON_SIZE });

        this._zone = 'grid';
        this._pinnedIndex = 0;
        this._pinnedItems = [];
        this._pinnedButtons = [];
        this._buttons = [];
        this._focusedGridBtn = null;
        this._focusedPinnedBtn = null;
        this._searching = false;
        this._resetScroll = true;
        this._lastWheel = 0;
        this._scrollTarget = 0;
        this._pinnedTarget = 0;
        this._lastPinnedWheel = 0;
        this._pinnedAdj = null;
        this._pinnedLeftFade = null;
        this._pinnedRightFade = null;

        this._buildUI();
        this._onSearchChanged();
        this._refreshFocus();
        this._open();
    }

    _initialKeyFocus() { this._container.grab_key_focus(); }
    _entryFocused() { return global.stage.get_key_focus() === this._entry.clutter_text; }
    _focusSearch() { this._entry.grab_key_focus(); }

    _getRestoreKey(item) { return item.app.get_id(); }
    _pinnedVisible() { return this._pinnedSection.visible && this._pinnedButtons.length > 0; }

    _shouldCloseOnWindowRemoved() { return false; }

    _getAllApps() {
        if (this._apps) return this._apps;

        const appSystem = Shell.AppSystem.get_default();
        const apps = [];
        for (const info of Gio.AppInfo.get_all()) {
            if (!info.should_show()) continue;
            const app = appSystem.lookup_app(info.get_id());
            if (app) apps.push({ app, name: app.get_name() || '', icon: app.get_icon() });
        }
        apps.sort((a, b) => a.name.localeCompare(b.name));
        return (this._apps = apps);
    }

    _launchApp(app) {
        const name = app.get_name();
        try {
            app.launch(global.get_current_time(), -1, 0);
            journal(`[AppSearchOverlay] ✅ Launched successfully: ${name}`);
            return true;
        } catch (e) {
            journal(`[AppSearchOverlay] ❌ Launch failed for ${name}: ${e.message}`, true);
            return false;
        }
    }

    _computeResults(query) {
        const all = this._getAllApps();
        if (!query) return all.map(item => ({ ...item, indices: null }));

        const scored = [];
        for (const item of all) {
            const r = Fuzzy.fuzzyMatch(query, item.name);
            if (r.matched) scored.push({ ...item, score: r.score, indices: r.indices });
        }
        scored.sort((a, b) => b.score - a.score);

        const seen = new Set(scored.map(i => i.app.get_id()));
        for (const item of this._shellSearch(query, all)) {
            const id = item.app.get_id();
            if (seen.has(id)) continue;
            seen.add(id);
            scored.push({ ...item, score: 0, indices: null });
        }
        return scored;
    }

    _shellSearch(query, allItems) {
        const appSystem = Shell.AppSystem.get_default();
        const usage = Shell.AppUsage.get_default();

        let groups;
        try {
            groups = Shell.AppSystem.search(query);
        } catch (e) {
            journal(`[AppSearchOverlay] Shell.AppSystem.search() threw: ${e.message}`);
            return [];
        }
        if (!groups || groups.length === 0) return [];

        const itemByApp = new Map();
        for (const item of allItems) itemByApp.set(item.app, item);

        const results = [];
        for (const group of groups) {
            if (!group || group.length === 0) continue;
            const resolved = [];
            for (const id of group) {
                const app = appSystem.lookup_app(id);
                if (!app) continue;
                const item = itemByApp.get(app);
                if (item) resolved.push({ id, item });
            }
            if (resolved.length === 0) continue;
            resolved.sort((a, b) => usage.compare(a.id, b.id));
            for (const { item } of resolved) results.push(item);
        }
        return results;
    }

    _beforeResults(query, unchanged) {
        const searching = query.length > 0;
        if (searching && this._zone === 'pinned') this._zone = 'grid';
        this._searching = searching;
        this._resetScroll = !unchanged;
        this._syncSectionVisibility();
    }

    _syncSectionVisibility() {
        this._pinnedSection.visible = this._pinnedItems.length > 0;
        this._allHeadingLabel.set_text(this._searching ? 'RESULTS' : 'ALL APPS');
    }

    _buildUI() {
        const monitor = Main.layoutManager.primaryMonitor;
        const L = this._layout = computeLauncherLayout(monitor);
        this._cols = L.columns;
        this._tileW = L.tileWidth;
        this._container = UI.createFullscreenContainer(monitor, 'launcher-overlay');

        const search = UI.createSearchField({
            width: L.searchWidth, hint: 'Search', keyHint: "Press '/' to search", x: L.searchX, y: L.searchY,
        });
        this._wireEntry(search.entry);
        this._container.add_child(search.actor);

        const pinned = UI.createPinnedRow();
        this._pinnedBarBox = pinned.box;
        this._pinnedLeftFade = pinned.leftFade;
        this._pinnedRightFade = pinned.rightFade;
        this._wirePinnedScroll(pinned);
        this._pinnedStrip = Interactions.createDropStrip({
            container: this._pinnedBarBox, ghost: this._ghost,
            acceptSource: source => source?._dragPayload?.kind === 'app',
            externalResult: Interactions.DragMotionResult.NO_DROP,
            hint: {
                show: (anchor, text) => Interactions.getTooltip().showAbove(anchor, text, this._settings.get_int('tooltip-font-size')),
                hide: () => Interactions.getTooltip().hide(),
            },
            hintText: insertion => {
                if (!insertion.neighbor) return 'Pin here';
                const name = insertion.neighbor._appName ?? '';
                return insertion.insertBefore ? `Move before ${name}` : `Move after ${name}`;
            },
            onDrop: (source, index, actor) => this._dropPinned(source, index, actor),
        });
        this._pinnedBarBox._delegate = this._pinnedStrip.delegate();

        this._pinnedSection = new St.BoxLayout({
            style_class: 'launcher-pinned-section', vertical: true,
            x: L.fullX, y: L.pinnedY, width: L.fullWidth, height: L.pinnedHeight,
        });
        const pinnedHeading = UI.createDividerHeading('PINNED APPS');
        this._pinnedSection.add_child(pinnedHeading.actor);
        this._pinnedSection.add_child(pinned.actor);
        this._container.add_child(this._pinnedSection);

        const heading = UI.createDividerHeading('ALL APPS');
        this._allHeadingLabel = heading.label;
        heading.actor.set_position(L.fullX, L.resultsY);
        heading.actor.set_size(L.fullWidth, L.resultsHeadingHeight);
        this._container.add_child(heading.actor);

        const grid = UI.createGridScroll();
        this._gridScroll = grid.scroll;
        this._gridBox = grid.box;
        grid.scroll.set_position(L.fullX, L.resultsY + L.resultsHeadingHeight);
        grid.scroll.set_size(L.fullWidth, L.resultsHeight - L.resultsHeadingHeight);
        this._container.add_child(grid.scroll);

        this._scope.connect(this._gridScroll, 'scroll-event', (a, e) => this._onGridScroll(e));
        this._scope.connect(this._favorites, 'changed', () => this._onFavoritesChanged());
        this._scope.connect(Shell.AppSystem.get_default(), 'installed-changed', () => { this._apps = null; });

        this._renderPinnedBar();
    }

    _wirePinnedScroll(pinned) {
        this._pinnedAdj = pinned.scroll.hadjustment;
        const update = () => this._updatePinnedFades();
        this._scope.connect(this._pinnedAdj, 'changed', update);
        this._scope.connect(this._pinnedAdj, 'notify::value', update);
        this._scope.connect(pinned.scroll, 'scroll-event', (a, e) => this._onPinnedWheel(e));
    }

    _updatePinnedFades() {
        const adj = this._pinnedAdj;
        if (!adj) return;
        const eps = 1;
        this._pinnedLeftFade.visible = adj.value > eps;
        this._pinnedRightFade.visible = adj.value + adj.page_size < adj.upper - eps;
    }

    _easePinned(delta) {
        const adj = this._pinnedAdj;
        if (!adj) return;
        const now = GLib.get_monotonic_time() / 1000;
        if (now - this._lastPinnedWheel > 250) this._pinnedTarget = adj.value;
        this._lastPinnedWheel = now;
        const max = Math.max(0, adj.upper - adj.page_size);
        this._pinnedTarget = clamp(this._pinnedTarget + delta, 0, max);
        adj.ease(this._pinnedTarget, { duration: 160, mode: Clutter.AnimationMode.EASE_OUT_QUAD });
    }

    _onPinnedWheel(event) {
        const dir = event.get_scroll_direction();
        let d = 0;
        if (dir === Clutter.ScrollDirection.UP || dir === Clutter.ScrollDirection.LEFT) d = -1;
        else if (dir === Clutter.ScrollDirection.DOWN || dir === Clutter.ScrollDirection.RIGHT) d = 1;
        else if (dir === Clutter.ScrollDirection.SMOOTH) {
            const [dx, dy] = event.get_scroll_delta();
            d = Math.abs(dx) > Math.abs(dy) ? dx : dy;
        } else return Clutter.EVENT_PROPAGATE;
        this._easePinned(d * this._layout.pinnedTileWidth * 0.8);
        return Clutter.EVENT_STOP;
    }

    _renderPinnedBar() {
        this._pinnedScope.dispose();
        this._pinnedScope = new Scheduling.Scope();
        this._focusedPinnedBtn = null;
        this._pinnedBarBox.destroy_all_children();
        this._pinnedButtons = [];
        this._pinnedItems = [];

        const L = this._layout;

        this._pinnedBarBox.add_child(new St.Widget({ x_expand: true }));
        for (const app of this._favorites.getFavorites()) {
            const id = app.get_id();
            const name = app.get_name() || '';
            const icon = app.get_icon();
            const index = this._pinnedItems.length;
            this._pinnedItems.push({ app, name, icon });

            const button = UI.createAppTile({
                icon, name, iconSize: L.pinnedIconSize,
                width: L.pinnedTileWidth, height: L.pinnedTileHeight,
                showLabel: false,
            });
            button._appName = name;

            this._pinnedScope.connect(button, 'clicked', () => {
                this._zone = 'pinned'; this._pinnedIndex = index; this._refreshFocus();
                if (this._launchApp(app)) this._close();
            });
            this._pinnedScope.connect(button, 'button-press-event', (a, e) => {
                if (e.get_button() === Clutter.BUTTON_SECONDARY) {
                    this._openContextMenuForApp(app, button);
                    return Clutter.EVENT_STOP;
                }
                return Clutter.EVENT_PROPAGATE;
            });
            this._pinnedScope.connect(button, 'notify::hover', () => {
                if (this._closed) return;
                if (button.hover) {
                    Interactions.getTooltip().showBelow(button, name, this._settings.get_int('tooltip-font-size'));
                } else {
                    Interactions.getTooltip().hide();
                }
            });

            Interactions.makeDraggable(button, {
                iconSize: PINNED_ICON_SIZE,
                payload: { kind: 'app', appId: id, gicon: icon, name },
                delegate: this._pinnedStrip.delegate(),
                onDragBegin: () => { Interactions.getTooltip().hide(); },
                onDragEnd: () => { this._ghost.clear(); Interactions.getTooltip().hide(); },
            });

            this._pinnedBarBox.add_child(button);
            this._pinnedButtons.push(button);

            if (id === this._justDroppedAppId) { this._justDroppedAppId = null; Interactions.settleIcon(button); }
        }
        this._pinnedBarBox.add_child(new St.Widget({ x_expand: true }));

        this._pinnedIndex = clamp(this._pinnedIndex, 0, Math.max(0, this._pinnedItems.length - 1));
        this._syncSectionVisibility();
        this._scope.idle(() => { this._updatePinnedFades(); return GLib.SOURCE_REMOVE; });
    }

    _renderResults(results) {
        const L = this._layout;
        this._focusedGridBtn = null;
        this._gridBox.destroy_all_children();
        this._buttons = [];

        let row = null;
        results.forEach((item, index) => {
            if (index % this._cols === 0) {
                row = new St.BoxLayout({ x_align: Clutter.ActorAlign.CENTER, width: this._cols * this._tileW });
                this._gridBox.add_child(row);
            }
            const button = UI.createAppTile({
                icon: item.icon, name: item.name, indices: item.indices,
                iconSize: L.iconSize, width: L.tileWidth, height: L.tileHeight,
            });
            this._rowsScope.connect(button, 'clicked', () => {
                this._zone = 'grid'; this._selectedIndex = index; this._refreshFocus();
                this._activateResult(index);
            });
            this._rowsScope.connect(button, 'button-press-event', (a, e) => {
                if (e.get_button() === Clutter.BUTTON_SECONDARY) {
                    this._openContextMenuForApp(item.app, button);
                    return Clutter.EVENT_STOP;
                }
                return Clutter.EVENT_PROPAGATE;
            });
            row.add_child(button);
            this._buttons.push(button);
        });

        if (this._resetScroll) {
            this._gridScroll.vadjustment.value = 0;
            this._scrollTarget = 0;
        }
    }

    _onSelectionChanged() { this._refreshFocus(); }

    _refreshFocus() {
        this._focusedGridBtn?.remove_style_class_name('focused');
        this._focusedPinnedBtn?.remove_style_class_name('focused');
        this._focusedGridBtn = this._focusedPinnedBtn = null;

        if (this._zone === 'pinned') {
            this._focusedPinnedBtn = this._pinnedButtons[this._pinnedIndex] ?? null;
            this._focusedPinnedBtn?.add_style_class_name('focused');
        } else {
            this._focusedGridBtn = this._buttons[this._selectedIndex] ?? null;
            this._focusedGridBtn?.add_style_class_name('focused');
        }
    }

    _scrollToFocus() {
        if (this._zone === 'grid' && this._focusedGridBtn) {
            Util.ensureActorVisibleInScrollView(this._gridScroll, this._focusedGridBtn);
        } else if (this._zone === 'pinned' && this._focusedPinnedBtn && this._pinnedAdj) {
            const adj = this._pinnedAdj;
            const [value, , , , , page] = adj.get_values();
            const box = this._focusedPinnedBtn.get_allocation_box();
            if (box.x1 < value) adj.ease(box.x1 - 16, { duration: 150, mode: Clutter.AnimationMode.EASE_OUT_QUAD });
            else if (box.x2 > value + page) adj.ease(box.x2 - page + 16, { duration: 150, mode: Clutter.AnimationMode.EASE_OUT_QUAD });
        }
    }

    _afterFocusMove() { this._refreshFocus(); this._scrollToFocus(); }

    _centerX(button) {
        const [x] = button.get_transformed_position();
        return x + button.width / 2;
    }

    _nearestIndex(buttons, cx) {
        let best = 0, bestDist = Infinity;
        buttons.forEach((b, i) => {
            const d = Math.abs(this._centerX(b) - cx);
            if (d < bestDist) { best = i; bestDist = d; }
        });
        return best;
    }

    _moveFocus(dx, dy) {
        if (this._zone === 'pinned') {
            const n = this._pinnedButtons.length;
            if (dx) this._pinnedIndex = clamp(this._pinnedIndex + dx, 0, n - 1);
            else if (dy > 0 && this._buttons.length) {
                const cx = this._centerX(this._pinnedButtons[this._pinnedIndex]);
                this._zone = 'grid';
                this._selectedIndex = this._nearestIndex(this._buttons.slice(0, this._cols), cx);
            }
            this._afterFocusMove();
            return;
        }

        const n = this._buttons.length;
        if (!n) return;
        const cols = this._cols;
        let i = this._selectedIndex;
        if (dx) i = clamp(i + dx, 0, n - 1);
        if (dy) {
            const next = i + dy * cols;
            if (next < 0) {
                if (this._pinnedVisible()) {
                    this._pinnedIndex = this._nearestIndex(this._pinnedButtons, this._centerX(this._buttons[i]));
                    this._zone = 'pinned';
                    this._afterFocusMove();
                }
                return;
            }
            if (next >= n) { if (Math.floor(i / cols) < Math.floor((n - 1) / cols)) i = n - 1; }
            else i = next;
        }
        this._selectedIndex = i;
        this._afterFocusMove();
    }

    _onGridScroll(event) {
        const adj = this._gridScroll.vadjustment;
        const dir = event.get_scroll_direction();
        let delta = 0;
        if (dir === Clutter.ScrollDirection.UP) delta = -1;
        else if (dir === Clutter.ScrollDirection.DOWN) delta = 1;
        else if (dir === Clutter.ScrollDirection.SMOOTH) delta = event.get_scroll_delta()[1];
        else return Clutter.EVENT_PROPAGATE;

        const now = GLib.get_monotonic_time() / 1000;
        if (now - this._lastWheel > 250) this._scrollTarget = adj.value;
        this._lastWheel = now;

        const max = adj.upper - adj.page_size;
        this._scrollTarget = clamp(this._scrollTarget + delta * this._layout.tileHeight * 0.75, 0, Math.max(0, max));
        adj.ease(this._scrollTarget, { duration: 140, mode: Clutter.AnimationMode.EASE_OUT_QUAD });
        return Clutter.EVENT_STOP;
    }

    _handleExtraKeys(event, symbol) {
        const state = event.get_state();
        if (symbol === Clutter.KEY_p && (state & Clutter.ModifierType.CONTROL_MASK)) {
            const item = this._zone === 'pinned' ? this._pinnedItems[this._pinnedIndex] : this._results[this._selectedIndex];
            if (item) this._togglePin(item);
            return Clutter.EVENT_STOP;
        }

        const inEntry = this._entryFocused();
        switch (symbol) {
            case Clutter.KEY_slash:
                if (inEntry) return undefined;
                this._focusSearch();
                return Clutter.EVENT_STOP;
            case Clutter.KEY_Left: this._moveFocus(-1, 0); return Clutter.EVENT_STOP;
            case Clutter.KEY_Right: this._moveFocus(1, 0); return Clutter.EVENT_STOP;
            case Clutter.KEY_Up: this._moveFocus(0, -1); return Clutter.EVENT_STOP;
            case Clutter.KEY_Down: this._moveFocus(0, 1); return Clutter.EVENT_STOP;
        }

        if (!inEntry && !(state & (Clutter.ModifierType.CONTROL_MASK | Clutter.ModifierType.MOD1_MASK))) {
            const uc = event.get_key_unicode();
            const code = typeof uc === 'string' ? uc.codePointAt(0) : uc;
            if (code && code >= 0x20 && code !== 0x7f) {
                this._focusSearch();
                this._entry.clutter_text.insert_unichar(code);
                return Clutter.EVENT_STOP;
            }
        }
        return undefined;
    }

    _handleEscape() {
        if (this._entry.get_text().length > 0) {
            this._entry.set_text('');
            this._container.grab_key_focus();
            return;
        }
        this._close();
    }

    _activateFocused() {
        if (this._zone === 'pinned') {
            const item = this._pinnedItems[this._pinnedIndex];
            if (item && this._launchApp(item.app)) this._close();
            return;
        }
        if (this._selectedIndex >= 0) this._activateResult(this._selectedIndex);
    }

    _activateResult(index) {
        if (this._closed) return;
        const item = this._results[index];
        if (item && this._launchApp(item.app)) this._close();
    }

    _dropPinned(source, index, actor) {
        const appId = source?._dragPayload?.appId;
        if (!appId) return false;

        const pos = index ?? this._pinnedBarBox.get_children().length;
        this._justDroppedAppId = appId;

        try { this._favorites.moveFavoriteToPos(appId, pos); }
        catch (e) { journal(`[AppSearchOverlay] Reorder failed: ${e.message}`); }

        if (actor) { actor.get_parent()?.remove_child(actor); actor.destroy(); }
        return true;
    }

    _openContextMenuForApp(app, sourceActor) {
        this._closeContextMenu();
        if (!this._menuManager)
            this._menuManager = new PopupMenu.PopupMenuManager(sourceActor);

        const menu = new AppMenuModule.AppMenu(sourceActor, St.Side.BOTTOM, { favoritesSection: true, showSingleWindows: true });
        Main.layoutManager.addChrome(menu.actor);
        menu.actor.hide();
        this._menuManager.addMenu(menu);
        menu.setApp(app);
        this._openMenu = menu;

        menu.connect('open-state-changed', (o, isOpen) => {
            if (isOpen) return;
            if (menu === this._openMenu) this._openMenu = null;
            GLib.idle_add(GLib.PRIORITY_DEFAULT, () => {
                if (!menu.is_destroyed) menu.destroy();
                return GLib.SOURCE_REMOVE;
            });
        });
        menu.open(true);
    }

    _closeContextMenu() {
        if (this._openMenu) {
            try { this._openMenu.close(); } catch (e) { }
            this._openMenu = null;
        }
    }

    _togglePin(item) {
        if (this._closed) return;
        const id = item.app.get_id();
        if (this._favorites.isFavorite(id)) this._favorites.removeFavorite(id);
        else this._favorites.addFavorite(id);
    }

    _onFavoritesChanged() {
        if (this._closed) return;
        this._renderPinnedBar();
        this._onSearchChanged();
        this._refreshFocus();
    }

    _onBeforeClose() {
        this._closeContextMenu();
        this._ghost.clear();
        this._pinnedScope.dispose();
        Interactions.getTooltip().hide();
        this._apps = null;
    }

    _close() {
        if (this._closed) return;
        super._close();
    }
}

export function fadeInActor(actor, duration = Model.TimeoutDelay, mode = Clutter.AnimationMode.EASE_OUT_QUAD) {
    if (!actor) { journal('[fadeInActor] called with null actor'); return; }
    try {
        if (!actor.get_stage()) { journal('[fadeInActor] actor not on stage, skipping'); return; }
        actor.remove_all_transitions();
        actor.opacity = 0;
        actor.ease({ opacity: 255, duration, mode });
    } catch (e) {
        journal(`[fadeInActor] error: ${e}`);
    }
}

export function settleIcon(actor, duration = Model.SettleAnimationDuration, mode = Clutter.AnimationMode.EASE_OUT_QUAD) {
    if (!actor) { journal('[settleIcon] called with null actor'); return; }
    try {
        if (!actor.get_stage()) { journal('[settleIcon] actor not on stage, skipping'); return; }
        actor.remove_all_transitions();
        actor.set_pivot_point(0.5, 0.5);
        actor.set_scale(0.85, 0.85);
        actor.opacity = 160;
        actor.ease({ scale_x: 1, scale_y: 1, opacity: 255, duration, mode });
    } catch (e) {
        journal(`[settleIcon] error: ${e}`);
    }
}

function createTooltip() {
    const label = new St.Label({
        style_class: 'dmenu-tooltip',
        visible: false,
        reactive: true,
        track_hover: true,
    });
    Main.layoutManager.addChrome(label);

    function apply(text, fontSize) {
        if (fontSize != null) label.set_style(`font-size: ${fontSize}pt;`);
        label.set_text(text);
        label.show();
        try { Main.layoutManager.uiGroup.set_child_above_sibling(label, null); }
        catch (e) { /* nothing else in uiGroup yet */ }
    }

    function show(actor, text, fontSize, prefer) {
        apply(text, fontSize);

        const [width, height] = label.get_size();
        const [actorX, actorY] = actor.get_transformed_position();
        const actorWidth = actor.get_width();
        const actorHeight = actor.get_height();

        const monitorIndex = Main.layoutManager.findIndexForActor(actor);
        const monitor = Main.layoutManager.monitors[monitorIndex] ?? Main.layoutManager.primaryMonitor;
        const pad = 6;

        const minX = monitor.x + pad;
        const maxX = monitor.x + monitor.width - width - pad;
        const minY = monitor.y + pad;
        const maxY = monitor.y + monitor.height - height - pad;

        let x = Math.round(actorX + (actorWidth - width) / 2);
        x = Math.max(minX, Math.min(x, maxX));

        const above = Math.round(actorY - height - 8);
        const below = Math.round(actorY + actorHeight + 8);
        let y;
        if (prefer === 'below') { y = below; if (y > maxY) y = above; }
        else { y = above; if (y < minY) y = below; }
        y = Math.max(minY, Math.min(y, maxY));

        label.set_position(x, y);
    }

    return {
        get hover() { return label.hover; },

        showAbove(actor, text, fontSize = null) { show(actor, text, fontSize, 'above'); },
        showBelow(actor, text, fontSize = null) { show(actor, text, fontSize, 'below'); },

        showCentered(actor, text, y, fontSize = null, padding = 10) {
            apply(text, fontSize);
            const width = label.width;
            const [actorX] = actor.get_transformed_position();
            let x = actorX + (actor.get_width() - width) / 2;
            x = Math.max(padding, Math.min(x, Model.screenWidth - width - padding));
            label.set_position(Math.round(x), Math.round(y));
        },

        hide() { label.hide(); },

        destroy() {
            Main.layoutManager.removeChrome(label);
            label.destroy();
        },
    };
}

let _sharedTooltip = null;

export function getTooltip() {
    if (!_sharedTooltip) _sharedTooltip = createTooltip();
    return _sharedTooltip;
}

export function destroyTooltip() {
    _sharedTooltip?.destroy();
    _sharedTooltip = null;
}

// ==================== reorder ghost ====================
export function createReorderGhost({ iconSize, giconFor = s => s._dragPayload?.gicon }) {
    let container = null;
    let source = null;
    let ghost = null;
    let ghostIndex = null;
    let snap = null;

    function takeSnapshot() {
        const src = source._actor ?? source;
        return container.get_children()
            .filter(b => b !== src && b._dragItem && b.get_stage())
            .map(b => {
                const [x] = b.get_transformed_position();
                const [w] = b.get_transformed_size();
                return { item: b, x, width: w, mid: x + w / 2, right: x + w };
            });
    }

    function place(insertion) {
        if (ghostIndex === insertion.index && ghost?.get_parent()) return;

        if (!ghost)
            ghost = UI.createGhostIcon(giconFor(source), iconSize());

        ghost.get_parent()?.remove_child(ghost);

        const { neighbor, insertBefore } = insertion;
        if (!neighbor) container.add_child(ghost);
        else if (insertBefore) container.insert_child_below(ghost, neighbor);
        else container.insert_child_above(ghost, neighbor);

        ghostIndex = insertion.index;
    }

    function begin(newContainer, newSource) {
        if (container !== newContainer) {
            ghost?.get_parent()?.remove_child(ghost);
            ghostIndex = null;
            snap = null;
        }
        container = newContainer;
        if (source !== newSource) { source = newSource; snap = null; }
    }

    function update(pointerX) {
        if (!container || !source) return null;
        if (!snap) snap = takeSnapshot();
        const insertion = UI.computeInsertion(snap, pointerX);
        place(insertion);
        return insertion;
    }

    function clear() {
        const g = ghost;
        ghost = null;
        ghostIndex = null;
        snap = null;
        container = null;
        source = null;
        if (!g) return;
        try { g.get_parent()?.remove_child(g); g.destroy(); }
        catch (e) { /* already gone */ }
    }

    return {
        get index() { return ghostIndex; },
        begin, update, clear,
    };
}

// ==================== draggable ====================
export function makeDraggable(actor, {
    iconSize, payload, gicon, delegate = {}, onDragBegin = null, onDragEnd = null,
}) {
    actor._dragItem = true;
    actor._dragPayload = payload;
    actor._delegate = Object.assign(delegate, {
        _dragPayload: payload,
        _actor: actor,
        getDragActorSource: () => actor,
        getDragActor: () => UI.createDragActor(gicon ?? payload?.gicon, iconSize),
    });

    const draggable = DND.makeDraggable(actor, {
        restoreOnSuccess: false,
        dragActorMaxSize: iconSize,
        dragActorOpacity: 178,
    });
    draggable.connect('drag-begin', () => { actor.opacity = 80; onDragBegin?.(); });
    draggable.connect('drag-end', () => {
        try { actor.opacity = 255; } catch (e) { /* source may be gone */ }
        onDragEnd?.();
    });
    return draggable;
}

// ==================== drop strip ====================
export function createDropStrip({
    container, ghost, acceptSource,
    externalResult = DragMotionResult.MOVE_DROP,
    onDragOver = null, onExternalDragOver = null, onDrop,
    hint = null, hintText = null,
}) {
    function handleDragOver(src) {
        onDragOver?.(src);

        if (!acceptSource(src)) {
            cancel();
            onExternalDragOver?.(src);
            return externalResult;
        }

        const [pointerX] = global.get_pointer();
        ghost.begin(container, src);
        const insertion = ghost.update(pointerX);
        if (insertion && hint && hintText)
            hint.show(insertion.neighbor ?? container, hintText(insertion));
        return DragMotionResult.MOVE_DROP;
    }

    function acceptDrop(src, actor) {
        const accepted = acceptSource(src);
        const index = accepted ? ghost.index : null;
        cancel();
        return onDrop(src, index, actor);
    }

    function cancel() {
        ghost.clear();
        hint?.hide();
    }

    function delegate() {
        return {
            handleDragOver: s => handleDragOver(s),
            acceptDrop: (s, a) => acceptDrop(s, a),
        };
    }

    return { handleDragOver, acceptDrop, delegate, cancel };
}

}

// ==================== active hover ====================
const ActiveHover = (() => {
    let current = null;
    let ctrlPressed = false;
    let pollId = 0;

    function readCtrl() {
        const [, , mods] = global.get_pointer();
        return (mods & Clutter.ModifierType.CONTROL_MASK) !== 0;
    }

    function stopPoll() {
        if (pollId) { GLib.Source.remove(pollId); pollId = 0; }
    }

    function startPoll() {
        if (pollId) return;
        ctrlPressed = readCtrl();
        pollId = GLib.timeout_add(GLib.PRIORITY_DEFAULT, Model.TimeoutDelay, () => {
            if (!current) { pollId = 0; return GLib.SOURCE_REMOVE; }
            const ctrl = readCtrl();
            if (ctrl !== ctrlPressed) {
                ctrlPressed = ctrl;
                current.onCtrlChanged(ctrl);
            }
            return GLib.SOURCE_CONTINUE;
        });
    }

    return {
        get current() { return current; },
        get ctrlPressed() { return ctrlPressed; },
        activate(controller) {
            if (current && current !== controller) current.forceHide();
            current = controller;
            startPoll();
        },
        deactivate(controller) {
            if (current === controller) { current = null; stopPoll(); }
        },
    };
})();

// ==================== external drag auto-activate ====================
function createExternalDragAutoActivate(window) {
    const scope = new Scheduling.Scope();
    let id = 0;
    let lastHover = 0;

    function notify() {
        lastHover = GLib.get_monotonic_time();
        if (id) return;
        id = scope.timeout(400, () => {
            id = 0;
            if ((GLib.get_monotonic_time() - lastHover) / 1000 > 500)
                return GLib.SOURCE_REMOVE;

            const win = window;
            try {
                if (win.minimized) win.unminimize();
                if (win.get_maximized() !== Meta.MaximizeFlags.BOTH)
                    win.maximize(Meta.MaximizeFlags.BOTH);
                const ws = win.get_workspace();
                if (ws) ws.activate_with_focus(win, global.get_current_time());
                else win.activate(global.get_current_time());
            } catch (e) {
                journal(`[externalAutoActivate] failed: ${e.message}`);
            }
            return GLib.SOURCE_REMOVE;
        });
    }

    function destroy() { scope.dispose(); }

    return { notify, destroy };
}

// ==================== window icon controller ====================
export class WindowIconController {
    static DEBOUNCE = 30;

    constructor({ window, actor, icon, settings, strip, onRequestRender }) {
        this.window = window;
        this.actor = actor;

        this._settings = settings;
        this._strip = strip;
        this._onRequestRender = onRequestRender;
        this._scope = new Scheduling.Scope();
        this._external = createExternalDragAutoActivate(window);

        this._preview = null;
        this._showingTitle = false;
        this._hoverId = 0;
        this._cleanupId = 0;

        const iconSize = settings.get_int('icon-size');

        this._scope.connect(icon, 'stage-views-changed', () => {
            const rect = new Mtk.Rectangle();
            [rect.x, rect.y] = icon.get_transformed_position();
            [rect.width, rect.height] = icon.get_transformed_size();
            window.set_icon_geometry(rect);
        });

        this._scope.connect(window, 'notify::wm-class', onRequestRender);
        this._scope.connect(window, 'notify::mapped', onRequestRender);
        this._scope.connect(window, 'notify::appears-focused', () => this._syncFocus());

        makeDraggable(actor, {
            iconSize,
            payload: { kind: 'window', window, gicon: UI.getWindowAppIcon(window) },
            delegate: {
                handleDragOver: (source) => {
                    if (source === actor._delegate) return DragMotionResult.CONTINUE;
                    if (!source?._dragPayload?.window) this._external.notify();
                    return this._strip?.handleDragOver(source) ?? DragMotionResult.CONTINUE;
                },
                acceptDrop: (source, dragActor) =>
                    this._strip?.acceptDrop(source, dragActor) ?? false,
            },
            onDragBegin: () => Model.setIntent({ drag: true }),
            onDragEnd: () => Model.setIntent({ drag: null, hintWs: null }),
        });

        this._scope.connect(actor, 'notify::hover', () => this._onHover());
        this._scope.connect(actor, 'button-press-event', (a, e) => this._onPressed(e));
        this._scope.connect(actor, 'button-release-event', (a, e) => this._onReleased(e));

        this._syncFocus();
    }

    setStrip(strip) { this._strip = strip; }

    notifyExternalDrag() { this._external.notify(); }

    destroy() {
        this.forceHide();
        this._external.destroy();
        this._scope.dispose();
        this._strip = null;
        this._onRequestRender = null;
        try { this.actor?.destroy(); } catch (e) { /* already gone */ }
    }

    _onHover() {
        if (this._hoverId) { this._scope.cancel(this._hoverId); this._hoverId = 0; }
        if (this.actor.hover) {
            this._cancelCleanup();
            if (this._isShowing()) { this.onCtrlChanged(ActiveHover.ctrlPressed); return; }
            this._hoverId = this._scope.timeout(WindowIconController.DEBOUNCE, () => {
                this._hoverId = 0;
                if (!this.actor.hover && !this._isShowing()) return GLib.SOURCE_REMOVE;
                ActiveHover.activate(this);
                this._render();
                return GLib.SOURCE_REMOVE;
            });
        } else if (this._isShowing()) {
            this._startCleanup();
        }
    }

    onCtrlChanged(ctrl) { this._render(ctrl); }

    forceHide() {
        if (this._hoverId) { this._scope.cancel(this._hoverId); this._hoverId = 0; }
        this._cancelCleanup();
        this._hide();
        ActiveHover.deactivate(this);
    }

    _render(ctrl = ActiveHover.ctrlPressed) {
        if (ctrl) { this._hidePreview(); this._showTitle(); }
        else { this._hideTitle(); this._showPreview(); }
    }

    _isShowing() { return this._showingTitle || !!this._preview; }

    _onPreviewHover(isHovered) {
        if (isHovered) this._cancelCleanup();
        else if (!this.actor.hover) this._startCleanup();
    }

    _startCleanup() {
        this._cancelCleanup();
        this._cleanupId = this._scope.timeout(Model.TimeoutDelay, () => {
            this._cleanupId = 0;
            const tip = getTooltip();
            if (this.actor.hover || this._preview?.hover || (this._showingTitle && tip.hover))
                return GLib.SOURCE_REMOVE;
            this._hide();
            ActiveHover.deactivate(this);
            return GLib.SOURCE_REMOVE;
        });
    }

    _cancelCleanup() {
        if (this._cleanupId) { this._scope.cancel(this._cleanupId); this._cleanupId = 0; }
    }

    _hide() {
        this._hidePreview();
        this._hideTitle();
    }

    _showTitle() {
        this._showingTitle = true;
        getTooltip().showCentered(
            this.actor,
            this.window.get_title() || 'Untitled Window',
            Model.screenHeight - 200,
            this._settings.get_int('title-popup-font-size'));
    }

    _hideTitle() {
        if (!this._showingTitle) return;
        this._showingTitle = false;
        getTooltip().hide();
    }

    _showPreview() {
        if (this._preview) return;
        const win = this.window;
        const previewHeight = this._settings.get_int('hover-preview-height');
        const built = UI.createClonePreview(win, previewHeight, {
            wrapperStyleClass: 'hover-preview-wrapper',
            showTitle: false,
            closeButtonSize: this._settings.get_int('close-button-size'),
            closeButtonOffsetX: 60,
            closeButtonOffsetY: 10,
        });
        if (!built) return;

        if (built.wrapper)
            this._scope.connect(built.wrapper, 'notify::hover',
                () => this._onPreviewHover(built.wrapper.hover));
        if (built.wrapper)
            this._scope.connect(built.wrapper, 'button-press-event', (a, e) => {
                if (e.get_button() === Clutter.BUTTON_PRIMARY) {
                    win.get_workspace().activate_with_focus(win, 0);
                    this._hide();
                    return Clutter.EVENT_STOP;
                }
                return Clutter.EVENT_PROPAGATE;
            });
        if (built.closeButton)
            this._scope.connect(built.closeButton, 'clicked', () => {
                win.delete(global.get_current_time());
                this._hide();
                return Clutter.EVENT_STOP;
            });

        const [anchorX] = this.actor.get_transformed_position();
        const x = Math.max(0, anchorX + (this.actor.get_width() - built.width) / 2);
        const y = Model.screenHeight - built.height - 200 + 55;
        built.actor.set_position(x, y);

        this._preview = built.actor;
        Main.layoutManager.addChrome(this._preview);
        fadeInActor(this._preview);
    }

    _hidePreview() {
        if (!this._preview) return;
        const actor = this._preview;
        this._preview = null;
        Main.layoutManager.removeChrome(actor);
        actor.destroy();
    }

    // Prototype `.ws-ico.focused`: ring + 10px halo + a small lift. St CSS
    // has no `transform`, so the lift is an actor scale on a centred pivot.
    _syncFocus() {
        const on = !!this.window.appears_focused;
        if (on) this.actor.add_style_class_name('focused');
        else this.actor.remove_style_class_name('focused');

        this.actor.set_pivot_point(0.5, 0.5);
        this.actor.set_scale(on ? 1.06 : 1, on ? 1.06 : 1);
    }

    _onPressed(event) {
        const button = event.get_button();
        if (button === Clutter.BUTTON_SECONDARY) {
            this.forceHide();
            this._openMenu();
            return Clutter.EVENT_STOP;
        }
        return button === Clutter.BUTTON_PRIMARY ? Clutter.EVENT_STOP : Clutter.EVENT_PROPAGATE;
    }

    _onReleased(event) {
        if (event.get_button() !== Clutter.BUTTON_PRIMARY) return Clutter.EVENT_PROPAGATE;

        this.forceHide();
        const win = this.window;
        const winWs = win.get_workspace();
        if (winWs === Model.WorkspaceManager.get_active_workspace()) {
            if (win.minimized) { win.unminimize(); win.activate_with_workspace(0, winWs); }
            else if (!win.appears_focused || this._isCovered(win)) win.activate_with_workspace(0, winWs);
            else win.minimize();
        } else {
            winWs.activate_with_focus(win, 0);
        }
        return Clutter.EVENT_STOP;
    }

    _openMenu() {
        const win = this.window;
        openContextMenu(this.actor, m => {
            m.addAction(`Activate ${win.title}`, () => win.get_workspace().activate_with_focus(win, 0));
            m.addAction(`Close ${win.title}`, () => win.delete(0));
            m.addAction(`Close Except ${win.title}`, () => {
                const wmClass = win.get_wm_class();
                const others = Model.getModelWindows(win.get_workspace())
                    .filter(w => w !== win && w.get_wm_class() === wmClass
                        && w.get_wm_class_instance() !== 'file_progress');
                const now = global.get_current_time();
                for (const w of others) w.delete(now);
            });

            const app = UI.getWindowApp(win);
            if (!app) return;
            m.addMenuItem(new PopupMenu.PopupSeparatorMenuItem());

            const favorites = AppFavorites.getAppFavorites();
            const isFav = favorites.isFavorite(app.get_id());
            m.addAction(isFav ? `Unpin ${app.get_name()}` : `Pin ${app.get_name()}`, () => {
                if (isFav) favorites.removeFavorite(app.get_id());
                else favorites.addFavorite(app.get_id());
            });

            const appInfo = app.get_app_info();
            const actions = appInfo?.list_actions();
            if (actions?.length) {
                m.addMenuItem(new PopupMenu.PopupSeparatorMenuItem());
                for (const action of actions)
                    m.addAction(`${appInfo.get_action_name(action)}`, () => app.launch_action(action, 0, -1));
            }
        }, { translationY: -35 });
    }

    _isCovered(window) {
        if (window.minimized) return false;
        const workspace = Model.WorkspaceManager.get_active_workspace();
        const stacked = Model.Display.sort_windows_by_stacking(
            Model.Display.list_all_windows().filter(w =>
                !w.minimized &&
                (w.get_window_type() === Meta.WindowType.NORMAL || w.get_window_type() === Meta.WindowType.DIALOG) &&
                w.get_workspace() === workspace));

        const index = stacked.indexOf(window);
        if (index === -1) return false;

        const area = workspace.get_work_area_for_monitor(window.get_monitor());
        const clip = r => ({
            l: Math.max(r.x, area.x), t: Math.max(r.y, area.y),
            r: Math.min(r.x + r.width, area.x + area.width),
            b: Math.min(r.y + r.height, area.y + area.height),
        });

        const t = clip(window.get_frame_rect());
        if (t.r <= t.l || t.b <= t.t) return false;

        for (let i = index + 1; i < stacked.length; i++) {
            const other = stacked[i];
            if (other.get_transient_for() === window) continue;
            const o = clip(other.get_frame_rect());
            if (o.r <= o.l || o.b <= o.t) continue;
            if (t.l < o.r && t.r > o.l && t.t < o.b && t.b > o.t) return true;
        }
        return false;
    }
}

// ===== FILE: ./lib/externalDrag.js =====
// Drags that originate OUTSIDE the shell (Nautilus, Nemo, Firefox, …) never
// enter the internal St DND chain, so no `handleDragOver`/`acceptDrop` ever
// fires on our actors. The only reliable early signal is Meta.Selection's
// SELECTION_DND owner: an external client claims it when the drag starts and
// replaces it with a Memory source when the drag ends.
//
// This is the same approach as the ExternalDragNotify diagnostic extension:
// BEGIN/CLEAR come exclusively from the Selection watch. Meta.Dnd's
// dnd-enter / dnd-leave are late and fire per-stage-entry, not per-drag, so
// they are hooked as DIAGNOSTICS ONLY — a STAGE-ENTER without a preceding
// BEGIN means the early signal missed that drag.

import Clutter from 'gi://Clutter';
import GLib from 'gi://GLib';
import Meta from 'gi://Meta';

import { createLogger } from '../logger.js';
const journal = createLogger(import.meta.url);

const DRAG_MAX_MS = 30000;

const TEXT_TYPES = [
    'text/plain;charset=utf-8', 'UTF8_STRING', 'text/plain', 'STRING', 'TEXT',
];

function typeName(obj) {
    if (!obj) return 'null';
    return obj.constructor?.$gtype?.name ?? obj.constructor?.name ?? 'unknown';
}

function isMemory(source) {
    return source instanceof Meta.SelectionSourceMemory || typeName(source).includes('Memory');
}

function mimesOf(source) {
    try { return source?.get_mimetypes() ?? []; } catch (e) { return []; }
}

function classify(mimes) {
    if (mimes.includes('text/uri-list')) return 'files';
    if (TEXT_TYPES.some(t => mimes.includes(t))) return 'text';
    if (mimes.length === 0) return 'unknown';
    return 'other';
}

class ExternalDragWatcher {
    constructor() {
        this._current = null;
        this._selection = null;
        this._selectionId = 0;
        this._dnd = null;
        this._dndIds = [];
        this._listeners = new Set();
        this._expiryId = 0;
        this._counter = 0;
        this._started = false;
    }

    /** { id, kind, mimes, app, startedAt } or null. */
    get current() { return this._current; }

    get active() { return this._current !== null; }

    /** cb(state|null) — fired on begin and on end, never in between. */
    onChanged(cb) {
        this._listeners.add(cb);
        return () => this._listeners.delete(cb);
    }

    init() {
        if (this._started) return;
        this._started = true;

        // ---- early signal (state transitions) ----
        this._selection = global.display.get_selection();
        if (!this._selection) {
            journal('[externalDrag] no Meta.Selection — external drags unsupported');
            return;
        }
        this._selectionId = this._selection.connect('owner-changed',
            (sel, type, source) => this._onOwnerChanged(type, source));

        // ---- late signals (diagnostics only, as in ExternalDragNotify) ----
        try {
            this._dnd = global.backend.get_dnd();
            this._dndIds.push(this._dnd.connect('dnd-enter', () => this._onStageEnter()));
            this._dndIds.push(this._dnd.connect('dnd-leave', () => {
                journal(`[externalDrag] STAGE-LEAVE${this._current ? ` #${this._current.id}` : ''}`);
            }));
        } catch (e) {
            journal(`[externalDrag] backend dnd hooks unavailable: ${e.message}`);
        }
    }

    _onStageEnter() {
        if (this._current) {
            const t = Math.round(GLib.get_monotonic_time() / 1000 - this._current.startedAt);
            journal(`[externalDrag] STAGE-ENTER #${this._current.id} t=${t}ms`);
        } else {
            const app = global.display.focus_window?.get_wm_class() ?? 'unknown';
            journal(`[externalDrag] STAGE-ENTER LATE-ONLY (no early begin) focus=${app}`);
        }
    }

    _onOwnerChanged(type, source) {
        if (type !== Meta.SelectionType.SELECTION_DND) return;

        if (!source || isMemory(source)) { this.end('owner-cleared'); return; }

        const mimes = mimesOf(source);
        if (mimes.length === 0) { this.end('no-mimes'); return; }

        const previous = this._current;
        this._current = {
            // A new begin without an end keeps the same id; otherwise allocate one.
            id: previous ? previous.id : ++this._counter,
            kind: classify(mimes),
            mimes,
            app: global.display.focus_window?.get_wm_class() ?? 'unknown',
            startedAt: GLib.get_monotonic_time() / 1000,
        };

        journal(`[externalDrag] begin #${this._current.id} kind=${this._current.kind} ` +
            `app=${this._current.app} src=${typeName(source)} n=${mimes.length}`);
        if (this._current.kind === 'other')
            journal(`[externalDrag]   mimes=${mimes.join(',')}`);

        if (this._expiryId) GLib.Source.remove(this._expiryId);
        this._expiryId = GLib.timeout_add(GLib.PRIORITY_DEFAULT, DRAG_MAX_MS, () => {
            this._expiryId = 0;
            this.end('expired');
            return GLib.SOURCE_REMOVE;
        });

        this._emit();
    }

    /** Public so the UI's pointer poll can add backstops. */
    end(reason) {
        const drag = this._current;
        if (!drag) return;
        this._current = null;
        if (this._expiryId) { GLib.Source.remove(this._expiryId); this._expiryId = 0; }
        journal(`[externalDrag] end #${drag.id} via=${reason} ` +
            `after=${Math.round(GLib.get_monotonic_time() / 1000 - drag.startedAt)}ms`);
        this._emit();
    }

    _emit() {
        for (const cb of [...this._listeners]) {
            try { cb(this._current); }
            catch (e) { journal(`[externalDrag] listener failed: ${e.message}`); }
        }
    }

    destroy() {
        this.end('destroyed');
        this._listeners.clear();
        if (this._selection && this._selectionId) {
            try { this._selection.disconnect(this._selectionId); } catch (e) { /* gone */ }
        }
        this._selection = null;
        this._selectionId = 0;
        if (this._dnd) {
            for (const id of this._dndIds) {
                try { this._dnd.disconnect(id); } catch (e) { /* gone */ }
            }
        }
        this._dnd = null;
        this._dndIds = [];
        this._started = false;
    }
}

    ToggleGroupFlyout(workspaceNum, wmClass) {
        return WorkspaceIndicatorModule.toggleGroupFlyout(workspaceNum, wmClass ?? '');
    },
    CloseFlyout() {
        return WorkspaceIndicatorModule.closeGroupFlyout();
    },
};

let _ownerId = 0;
let _exported = null;

function _onBusAcquired(connection) {
    try {
        _exported = Gio.DBusExportedObject.wrapJSObject(MR_DBUS_IFACE, dbusObject);
        _exported.export(connection, BUS_PATH);
        journal(`[dbusService] Exported on ${BUS_PATH}`);
    } catch (e) {
        journal(`[dbusService] Failed to export: ${e.message}`, true);
    }
}

function _unexport() {
    if (!_exported) return;
    try { _exported.flush(); _exported.unexport(); } catch (e) { /* not exported */ }
    _exported = null;
}

export function initDBusService() {
    _ownerId = Gio.bus_own_name(
        Gio.BusType.SESSION, BUS_NAME, Gio.BusNameOwnerFlags.NONE,
        connection => _onBusAcquired(connection),
        (connection, name) => journal(`[dbusService] ${name}: name acquired`),
        (connection, name) => { journal(`[dbusService] ${name}: name lost`, true); _unexport(); });
}

export function destroyDBusService() {
    _unexport();
    if (_ownerId) { Gio.bus_unown_name(_ownerId); _ownerId = 0; }
}

// The prototype's #fan.local-anchored: the same strip, detached from the
// HUD and parked next to a dock icon. Opened by a dwell during an internal
// window drag (delegate.handleDragOver) or by the external-drag poll
// (pointerAt) — both drive the same PagedStrip. Also openable without a
// drag ("sticky", e.g. from DBus): rows are then click-to-activate and the
// pointer watchdog is off.

import Clutter from 'gi://Clutter';
import GLib from 'gi://GLib';
import Pango from 'gi://Pango';
import St from 'gi://St';

import * as Main from 'resource:///org/gnome/shell/ui/main.js';

import * as Scheduling from './scheduling.js';
import * as Interactions from './interactions.js';
import * as UI from './ui.js';
import { PagedStrip, ROW_SLOT, VISIBLE_SLOTS } from './strip.js';

import { createLogger } from '../logger.js';
const journal = createLogger(import.meta.url);

const clamp = (v, a, b) => Math.max(a, Math.min(b, v));

// Panel metrics scaled ~2x versus the original 12/28/8 layout.
const PAD = 20;
const RAIL_W = 48;
const GAP = 12;
const HEAD_H = 30;
const HEAD_Y = 16;

// Large, legible row text: 38px titles over 72px icons in 112px slots.
const ROW_ICON_SIZE = 72;
const ROW_LABEL_SIZE = 38;

export class GroupFlyout {
    constructor(settings) {
        this._settings = settings;
        this._scope = new Scheduling.Scope();
        this._open = false;
        this._items = [];
        this._anchor = null;
        this._onDrop = null;
        this._misses = 0;
        this._opensDown = true;
        this._rowsW = 240;
        this._watchId = 0;
        this._watchLogAt = 0;
        this._sticky = false;      // DBus-opened flyout (no drag lifetime)
        this._anchorKey = null;    // identity used by ToggleGroupFlyout

        this._root = new St.Widget({
            style_class: 'ws-flyout',
            reactive: true, track_hover: true, clip_to_allocation: true,
        });
        this._root.opacity = 0;
        this._root.visible = false;

        this._head = new St.Label({ style_class: 'ws-flyout-head', text: '' });
        this._head.clutter_text.set_ellipsize(Pango.EllipsizeMode.END);
        this._head.clutter_text.set_line_wrap(false);
        this._head.set_style('font-size: 32px;');

        this._rowsEl = new St.Widget({ clip_to_allocation: true });
        this._upEl = new St.Bin({ style_class: 'ws-flyout-edge ws-flyout-edge-up' });
        this._downEl = new St.Bin({ style_class: 'ws-flyout-edge ws-flyout-edge-down' });
        this._rail = UI.createRail();

        this._root.add_child(this._head);
        this._root.add_child(this._upEl);
        this._root.add_child(this._downEl);
        this._root.add_child(this._rail.root);
        this._root.add_child(this._rowsEl);

        Main.layoutManager.addChrome(this._root);

        this._strip = new PagedStrip({
            rowsEl: this._rowsEl,
            upEl: this._upEl,
            downEl: this._downEl,
            rail: this._rail,
            makeRow: item => this._makeRow(item),
            onEdgeStart: () => { this._misses = 0; },
            alive: () => this._open,
        });

        this._root._delegate = {
            handleDragOver: () => {
                this._misses = 0;
                const [, py] = global.get_pointer();
                const [, ry] = this._root.get_transformed_position();
                this._strip.over(py - ry);
                return Interactions.DragMotionResult.MOVE_DROP;
            },
            acceptDrop: source => {
                const idx = this._strip.highlighted;
                const item = idx >= 0 ? this._items[idx] : null;
                const handled = this._onDrop ? this._onDrop(source, item) : false;
                this.close(true);
                return handled;
            },
        };

        // Hover highlight + edge auto-scroll when no drag owns the pointer
        // (DBus-opened flyout). During a drag the grab routes motion to the
        // drag actor and the external poll drives `over()` instead.
        this._scope.connect(this._root, 'captured-event', (a, event) => {
            if (!this._open || event.type() !== Clutter.EventType.MOTION)
                return Clutter.EVENT_PROPAGATE;
            const [, sy] = event.get_coords();
            const [, ry] = this._root.get_transformed_position();
            this._strip.over(sy - ry);
            return Clutter.EVENT_PROPAGATE;
        });
        this._scope.connect(this._root, 'leave-event', () => {
            if (this._open) this._strip.clearHover();
            return Clutter.EVENT_PROPAGATE;
        });
    }

    get isOpen() { return this._open; }
    get anchor() { return this._anchor; }
    get items() { return this._items; }
    get sticky() { return this._sticky; }
    get anchorKey() { return this._anchorKey; }

    /**
     * Dismissal watchdog. Also emits a throttled state line so the caller
     * can see whether the flyout is actually visible (opacity, scale,
     * mapped) while the drag is in flight.
     */
    _startWatch() {
        if (this._watchId) return;
        journal('[flyout.watch] started');
        this._watchLogAt = 0;
        this._watchId = this._scope.timeout(120, () => {
            if (!this._open) { this._watchId = 0; journal('[flyout.watch] stopped'); return GLib.SOURCE_REMOVE; }
            const [x, y] = global.get_pointer();
            const over = this._isOver(x, y);

            const nowMs = GLib.get_monotonic_time() / 1000;
            if (nowMs - (this._watchLogAt ?? 0) > 500) {
                this._watchLogAt = nowMs;
                let rx = NaN, ry = NaN, rw = NaN, rh = NaN;
                try {
                    [rx, ry] = this._root.get_transformed_position();
                    [rw, rh] = this._root.get_transformed_size();
                } catch (e) { /* actor may be mid-destroy */ }
                journal(`[flyout.watch] ptr=${x},${y} over=${over} ` +
                    `bounds=${Math.round(rx)},${Math.round(ry)} ${Math.round(rw)}x${Math.round(rh)} ` +
                    `opacity=${this._root.opacity} visible=${this._root.visible} ` +
                    `mapped=${this._root.is_mapped()} scale=${this._root.get_scale().join(',')}`);
            }

            if (over) this._misses = 0;
            else if (++this._misses >= 2) {
                journal(`[flyout.watch] pointer left (${x},${y}) -> close`);
                this.close();
            }
            return GLib.SOURCE_CONTINUE;
        });
    }

    _makeRow(item) {
        const row = UI.createFlyoutRow({
            gicon: item.icon,
            title: item.title,
            sub: item.sub || '',
            badge: item.badge || '',
            width: this._rowsW,
            height: ROW_SLOT,
            iconSize: ROW_ICON_SIZE,
            labelSize: ROW_LABEL_SIZE,
        });

        // Click-to-activate: harmless during a drag (button is held, no
        // click events are delivered) and the primary interaction for
        // DBus-opened flyouts, where there is no drag payload to drop.
        row.connect('button-press-event', (a, e) => {
            if (e.get_button() !== Clutter.BUTTON_PRIMARY) return Clutter.EVENT_PROPAGATE;
            this._activateItem(item);
            return Clutter.EVENT_STOP;
        });
        return row;
    }

    _activateItem(item) {
        if (!item?.window) return;
        try {
            const win = item.window;
            if (win.minimized) win.unminimize();
            const ws = win.get_workspace();
            if (ws) ws.activate_with_focus(win, global.get_current_time());
            else win.activate(global.get_current_time());
            journal(`[flyout] row activate "${item.title}"`);
        } catch (e) {
            journal(`[flyout] row activate failed: ${e.message}`);
        }
        this.close(true);
    }

    /** Inside the flyout OR back over the icon that opened it. */
    _isOver(x, y) {
        return this.contains(x, y) || this.isOverAnchor(x, y);
    }

    contains(x, y) {
        if (!this._open || !this._root.get_stage()) return false;
        const [rx, ry] = this._root.get_transformed_position();
        const [rw, rh] = this._root.get_transformed_size();
        return x >= rx && x <= rx + rw && y >= ry && y <= ry + rh;
    }

    isOverAnchor(x, y) {
        const a = this._anchor;
        if (!this._open || !a) return false;
        try {
            if (!a.get_stage()) return false;
            const [ax, ay] = a.get_transformed_position();
            const [aw, ah] = a.get_transformed_size();
            return x >= ax && x <= ax + aw && y >= ay && y <= ay + ah;
        } catch (e) {
            return false;   // anchor destroyed mid-drag by a re-render
        }
    }

    pointerAt(x, y) {
        if (!this.contains(x, y)) {
            if (this._strip.highlighted >= 0) this._strip.clearHover();
            return null;
        }
        const [, ry] = this._root.get_transformed_position();
        const idx = this._strip.over(y - ry);
        if (idx < 0) return null;
        return { index: idx, item: this._items[idx] ?? null };
    }

    open({ anchor, title, items, onDrop, sticky = false, anchorKey = null }) {
        let onStage = false;
        try { onStage = !!anchor?.get_stage(); } catch (e) { onStage = false; }
        if (!anchor || !onStage || !items?.length) {
            journal(`[flyout.open] rejected anchor=${!!anchor} onStage=${onStage} items=${items?.length}`);
            return false;
        }

        if (this._open && this._anchor === anchor) { this._misses = 0; return true; }

        const monitorIndex = Math.max(0, Main.layoutManager.findIndexForActor(anchor));
        const monitor = Main.layoutManager.monitors[monitorIndex] ?? Main.layoutManager.primaryMonitor;

        this._anchor = anchor;
        this._items = items;
        this._onDrop = onDrop;

        const W = clamp(Math.round(monitor.width * 0.26), 560, 1080);
        const rowsW = W - PAD * 2 - RAIL_W - GAP;
        const rowsH = Math.min(VISIBLE_SLOTS, items.length) * ROW_SLOT;
        const H = Math.min(monitor.height - 120, rowsH + HEAD_H + PAD * 2 + 40);
        const rowsY = Math.round((H - rowsH) / 2);
        const railH = Math.min(rowsH, 640);

        this._rowsW = rowsW;

        this._head.set_position(PAD, HEAD_Y);
        this._head.set_size(W - PAD * 2, HEAD_H);
        this._head.set_text(title || '');

        this._rowsEl.set_position(PAD, rowsY);
        this._rowsEl.set_size(rowsW, rowsH);

        this._rail.root.set_position(W - PAD - RAIL_W, Math.round((H - railH) / 2));
        this._rail.root.set_size(RAIL_W, railH);

        this._upEl.set_position(0, 0);
        this._upEl.set_size(W, Math.max(0, rowsY));
        this._downEl.set_position(0, rowsY + rowsH);
        this._downEl.set_size(W, Math.max(0, H - rowsY - rowsH));

        this._root.set_size(W, H);

        // ---------- anchor geometry ----------
        let ax, ay, aw, ah;
        try {
            [ax, ay] = anchor.get_transformed_position();
            [aw, ah] = anchor.get_transformed_size();
        } catch (e) {
            journal(`[flyout.open] anchor geometry threw: ${e.message}`);
            return false;
        }
        // A just-rebuilt anchor has no allocation yet and reports NaN;
        // placing the flyout at NaN makes it invisible and makes the
        // watchdog close it instantly. Refuse; the caller retries after
        // allocation.
        if (![ax, ay, aw, ah].every(Number.isFinite) || aw <= 0 || ah <= 0) {
            journal(`[flyout.open] anchor not allocated yet: ax=${ax} ay=${ay} aw=${aw} ah=${ah}`);
            return false;
        }

        const left = clamp(ax + aw / 2 - W / 2, monitor.x + 8, monitor.x + monitor.width - W - 8);
        let top = ay - H - 16;
        this._opensDown = false;
        if (top < monitor.y + 8) {
            top = Math.max(monitor.y + 8, ay + ah + 16);
            this._opensDown = true;
        }
        if (top + H > monitor.y + monitor.height - 8)
            top = Math.max(monitor.y + 8, monitor.y + monitor.height - H - 8);

        this._root.set_position(Math.round(left), Math.round(top));

        // Float above client windows so a maximized preview cannot hide it.
        try {
            Main.layoutManager.uiGroup.set_child_above_sibling(this._root, null);
            const children = Main.layoutManager.uiGroup.get_children();
            journal(`[flyout.open] uiGroup: idx=${children.indexOf(this._root)} total=${children.length}`);
        } catch (e) {
            journal(`[flyout.open] set_child_above_sibling failed: ${e.message}`);
        }

        journal(`[flyout.open] monitor=${monitor.x},${monitor.y} ${monitor.width}x${monitor.height} ` +
            `W=${W} H=${H} left=${Math.round(left)} top=${Math.round(top)} opensDown=${this._opensDown} ` +
            `anchor=${Math.round(ax)},${Math.round(ay)} ${Math.round(aw)}x${Math.round(ah)}`);

        const px = clamp((ax + aw / 2 - left) / W, 0.08, 0.92);
        this._root.set_pivot_point(px, this._opensDown ? 0 : 1);

        this._strip.setViewport(H);
        this._strip.setItems(items, 0);
        this._strip.clearHover();

        this._open = true;
        this._sticky = sticky;
        this._anchorKey = anchorKey;
        this._misses = 0;

        // ---------- show ----------
        //
        // Two independent changes:
        //
        //   1. Opacity is set to 255 immediately. The previous version eased
        //      it from 0 alongside translation_y, and translation_y is not
        //      an animatable Clutter ease target on all GNOME versions —
        //      which can make the whole transition fail silently and leave
        //      opacity pinned at 0. The flyout is then "open" and pickable
        //      but invisible.
        //
        //   2. Only scale and translation are animated, both of which are
        //      universally supported by Clutter.ease().
        //
        this._root.remove_all_transitions();
        this._root.show();
        this._root.opacity = 255;
        this._root.translation_y = this._opensDown ? -12 : 12;
        this._root.set_scale(0.96, 0.96);

        journal(`[flyout.open] shown: opacity=${this._root.opacity} visible=${this._root.visible} ` +
            `mapped=${this._root.is_mapped()} x=${this._root.x} y=${this._root.y} ` +
            `w=${this._root.width} h=${this._root.height} scale=${this._root.get_scale().join(',')}`);

        this._root.ease({
            translation_y: 0,
            scale_x: 1, scale_y: 1,
            duration: 190,
            mode: Clutter.AnimationMode.EASE_OUT_CUBIC,
            onComplete: () => {
                journal(`[flyout.open] ease complete: opacity=${this._root.opacity} ` +
                    `scale=${this._root.get_scale().join(',')} ` +
                    `pos=${this._root.x},${this._root.y} size=${this._root.width}x${this._root.height}`);
            },
        });

        // Safety net: if for any reason the ease did not bring opacity to
        // full within 300ms, force it. This is a no-op when the ease works.
        this._scope.timeout(300, () => {
            if (this._open && this._root && this._root.opacity < 255) {
                journal(`[flyout.open] SAFETY FORCE opacity 255 (was ${this._root.opacity})`);
                this._root.opacity = 255;
            }
            return GLib.SOURCE_REMOVE;
        });

        if (sticky) {
            journal('[flyout.open] sticky (no drag) — watchdog off, rows are click-to-activate');
        } else {
            this._startWatch();
        }

        return true;
    }

    close(instant = false) {
        if (!this._open) return;
        journal(`[flyout.close] instant=${instant} sticky=${this._sticky}`);
        this._open = false;
        this._sticky = false;
        this._anchorKey = null;
        this._strip.stopEdge();
        this._strip.clearHover();
        this._onDrop = null;
        this._root.remove_all_transitions();

        if (instant) {
            this._root.hide();
            this._anchor = null;
            return;
        }

        this._root.ease({
            opacity: 0,
            translation_y: this._opensDown ? -12 : 12,
            scale_x: 0.97, scale_y: 0.97,
            duration: 130,
            mode: Clutter.AnimationMode.EASE_IN_QUAD,
            onComplete: () => { if (!this._open) this._root.hide(); },
        });
        this._anchor = null;
    }

    destroy() {
        this._open = false;
        this._sticky = false;
        this._anchorKey = null;
        if (this._watchId) { this._scope.cancel(this._watchId); this._watchId = 0; }
        this._strip?.destroy();
        this._scope.dispose();
        try { Main.layoutManager.removeChrome(this._root); } catch (e) { /* already gone */ }
        this._root.destroy();
        this._root = null;
    }
}

import GLib from 'gi://GLib';

/**
 * Owns the signals and GLib sources created by one imperative object.
 */
export class Scope {
    constructor() {
        this._signals = [];
        this._sources = [];
        this._disposed = false;
    }

    get disposed() { return this._disposed; }

    connect(obj, signal, cb) {
        if (this._disposed) return 0;
        const id = obj.connect(signal, cb);
        this._signals.push([obj, id]);
        return id;
    }

    timeout(ms, cb, priority = GLib.PRIORITY_DEFAULT) {
        if (this._disposed) return 0;
        let id = 0;
        id = GLib.timeout_add(priority, ms, () => {
            const r = cb();
            if (r !== GLib.SOURCE_CONTINUE) this._forgetSource(id);
            return r ?? GLib.SOURCE_REMOVE;
        });
        this._sources.push(id);
        return id;
    }

    idle(cb, priority = GLib.PRIORITY_DEFAULT) {
        if (this._disposed) return 0;
        let id = 0;
        id = GLib.idle_add(priority, () => {
            const r = cb();
            if (r !== GLib.SOURCE_CONTINUE) this._forgetSource(id);
            return r ?? GLib.SOURCE_REMOVE;
        });
        this._sources.push(id);
        return id;
    }

    cancel(id) {
        if (!id) return;
        this._forgetSource(id);
        try { GLib.Source.remove(id); } catch (e) { /* already gone */ }
    }

    disconnectSignals() {
        for (const [obj, id] of this._signals) {
            try { obj.disconnect(id); } catch (e) { /* obj may be gone */ }
        }
        this._signals = [];
    }

    _forgetSource(id) {
        const i = this._sources.indexOf(id);
        if (i !== -1) this._sources.splice(i, 1);
    }

    dispose() {
        if (this._disposed) return;
        this._disposed = true;
        for (const [obj, id] of this._signals) {
            try { obj.disconnect(id); } catch (e) { /* obj may be gone */ }
        }
        for (const id of this._sources) {
            try { GLib.Source.remove(id); } catch (e) { /* already gone */ }
        }
        this._signals = [];
        this._sources = [];
    }
}

export class Coalescer {
    constructor(fn) {
        this._fn = fn;
        this._id = 0;
        this._delay = 0;
    }

    queue(delay = 0) {
        if (this._id) {
            if (delay >= this._delay) return;
            GLib.Source.remove(this._id);
        }
        this._delay = delay;
        const run = () => { this._id = 0; this._fn(); return GLib.SOURCE_REMOVE; };
        this._id = delay
            ? GLib.timeout_add(GLib.PRIORITY_DEFAULT, delay, run)
            : GLib.idle_add(GLib.PRIORITY_DEFAULT, run);
    }

    cancel() {
        if (this._id) { GLib.Source.remove(this._id); this._id = 0; }
    }
}

// position that scrolls by whole rows, edge hold zones with acceleration,
// and a rail with ticks + marker. Used by the anchored flyout.

import GLib from 'gi://GLib';
import St from 'gi://St';

const clamp = (v, a, b) => Math.max(a, Math.min(b, v));

// Row metrics sized for large legible text on a 4K display.
export const ROW_SLOT = 112;
export const VISIBLE_SLOTS = 8;
const HALF = Math.floor(VISIBLE_SLOTS / 2);
const RAIL_PAD = 16;
const DEAD = 0.06;

// Tick geometry — must match the marker sizing in ui.createRail().
const TICK_W = 18;
const TICK_H = 5;

const speedWord = t => (t < 0.25 ? 'SLOW' : t < 0.6 ? 'MEDIUM' : 'FAST');
const depthToT = d => Math.max(0, (d - DEAD) / (1 - DEAD));
const edgeSpeed = d => {
    const t = depthToT(d);
    return t <= 0 ? 0 : 0.8 + 30 * t * t;
};

export class PagedStrip {
    constructor({ rowsEl, upEl, downEl, rail, makeRow, onEdgeStart = null, alive = null }) {
        this._rowsEl = rowsEl;
        this._upEl = upEl;
        this._downEl = downEl;
        this._rail = rail;
        this._makeRow = makeRow;
        this._onEdgeStart = onEdgeStart;
        this._alive = alive ?? (() => true);

        const upLabel = new St.Label({ style_class: 'ws-flyout-edge-label' });
        upLabel.set_x_align(ClutterAlign.CENTER);
        upLabel.set_y_align(ClutterAlign.END);
        upEl.set_child(upLabel);
        this._upLabel = upLabel;

        const downLabel = new St.Label({ style_class: 'ws-flyout-edge-label' });
        downLabel.set_x_align(ClutterAlign.CENTER);
        downLabel.set_y_align(ClutterAlign.START);
        downEl.set_child(downLabel);
        this._downLabel = downLabel;

        this._items = [];
        this._rowEls = new Map();
        this._view = 0;
        this._highlight = -1;
        this._slots = 1;
        this._hostH = 0;
        this._edgeDir = 0;
        this._edgeDepth = 0;
        this._raf = 0;
        this._last = 0;
        this._zoneText = { up: '', down: '' };
        this._zoneAlpha = { up: -1, down: -1 };
    }

    get count() { return this._items.length; }
    get highlighted() { return this._highlight; }
    get selectedIndex() { return Math.max(0, this._highlight); }
    scrolling() { return this._edgeDir !== 0; }

    setViewport(height) { this._hostH = height; }

    setItems(items, startIndex = 0) {
        this.stopEdge();
        this._items = items;
        this._slots = Math.min(VISIBLE_SLOTS, Math.max(1, items.length));
        this._rowsEl.set_height(this._slots * ROW_SLOT);

        for (const [, el] of this._rowEls) el.destroy();
        this._rowEls.clear();

        this._view = this._clampV(Math.max(0, startIndex));
        this._highlight = Math.round(this._view);

        this._buildRail();
        this.layout();
    }

    _isStrip() { return this._items.length > VISIBLE_SLOTS; }
    _minV() { return this._isStrip() ? HALF : (this._items.length - 1) / 2; }
    _maxV() { return this._isStrip() ? this._items.length - 1 - HALF : (this._items.length - 1) / 2; }
    _clampV(v) { return clamp(v, this._minV(), this._maxV()); }

    // ---------- rail ----------
    _railY(p) {
        const h = this._rail.root.height;
        if (this._items.length <= 1) return h / 2;
        const span = Math.max(1, h - RAIL_PAD * 2);
        return RAIL_PAD + (p / (this._items.length - 1)) * span;
    }

    _buildRail() {
        this._rail.ticks.destroy_all_children();
        this._rail.root.visible = this._isStrip();
        if (!this._isStrip()) return;

        const w = this._rail.root.width;
        this._items.forEach((it, i) => {
            const tick = new St.Widget({
                style_class: 'ws-flyout-rail-tick',
                x: Math.round((w - TICK_W) / 2),
                y: Math.round(this._railY(i)) - Math.round(TICK_H / 2),
                width: TICK_W, height: TICK_H,
            });
            tick.set_style(`background-color: ${it.color || '#93a1a1'};`);
            this._rail.ticks.add_child(tick);
        });
        this._updateMarker();
    }

    _updateMarker() {
        const marker = this._rail.marker;
        if (!marker) return;
        if (!this._isStrip()) { marker.visible = false; return; }
        marker.visible = true;

        const p = clamp(this._view, 0, Math.max(0, this._items.length - 1));
        const lo = Math.floor(p);
        const hi = Math.min(this._items.length - 1, lo + 1);
        const y = this._railY(lo) + (this._railY(hi) - this._railY(lo)) * (p - lo);
        marker.y = Math.round(y - marker.height / 2);
    }

    // ---------- rows ----------
    layout() {
        const R = this._items.length;
        const vis = this._slots;
        const lo = this._isStrip() ? Math.max(0, Math.floor(this._view - HALF - 1)) : 0;
        const hi = this._isStrip() ? Math.min(R - 1, Math.ceil(this._view + HALF + 1)) : R - 1;

        for (const [r, el] of [...this._rowEls]) {
            if (r < lo || r > hi) { el.destroy(); this._rowEls.delete(r); }
        }

        for (let r = lo; r <= hi; r++) {
            let el = this._rowEls.get(r);
            if (!el) {
                el = this._makeRow(this._items[r], r);
                this._rowEls.set(r, el);
                this._rowsEl.add_child(el);
            }
            const d = r - this._view;
            const ad = Math.abs(d);
            el.translation_y = Math.round((d + (vis - 1) / 2) * ROW_SLOT);

            const sel = r === this._highlight;
            if (sel) el.add_style_class_name('sel');
            else el.remove_style_class_name('sel');

            const op = this._isStrip()
                ? Math.max(0, 0.9 - 0.17 * ad) * clamp(HALF + 1 - ad, 0, 1)
                : 0.85;
            el.opacity = Math.round((sel ? 1 : op) * 255);
            el.show();
        }

        this._updateEdges();
        this._updateMarker();
    }

    setHighlight(i) {
        const idx = clamp(i, -1, Math.max(-1, this._items.length - 1));
        if (idx === this._highlight) return;
        this._highlight = idx;
        this.layout();
    }

    center(i) {
        this._view = this._clampV(i);
        this.layout();
    }

    // ---------- edges ----------
    _updateEdges() {
        const now = Math.round(this._view);
        const t = depthToT(this._edgeDepth);
        const above = this._isStrip() ? Math.max(0, now - HALF) : 0;
        const below = this._isStrip() ? Math.max(0, this._items.length - 1 - HALF - now) : 0;

        const apply = (which, el, label, count, dir, arrow, word) => {
            const active = this._edgeDir === dir;
            el.visible = count > 0;
            if (!el.visible) return;
            if (active) el.add_style_class_name('active');
            else el.remove_style_class_name('active');

            const text = `${arrow} ${count} more ${word}${active && t > 0 ? ' · ' + speedWord(t) : ''}`;
            if (this._zoneText[which] !== text) {
                this._zoneText[which] = text;
                label.set_text(text);
            }
            const alpha = active ? Math.round(clamp(t * 0.42, 0, 1) * 100) / 100 : 0;
            if (this._zoneAlpha[which] !== alpha) {
                this._zoneAlpha[which] = alpha;
                el.set_style(`background-color: rgba(42, 161, 152, ${alpha});`);
            }
        };

        apply('up', this._upEl, this._upLabel, above, -1, '▲', 'above');
        apply('down', this._downEl, this._downLabel, below, 1, '▼', 'below');
    }

    setEdge(dir, depth) {
        this._edgeDepth = depth;
        if (this._edgeDir !== dir) {
            this._edgeDir = dir;
            this._onEdgeStart?.();
        }
        if (!this._raf) {
            this._last = GLib.get_monotonic_time() / 1000;
            this._raf = GLib.timeout_add(GLib.PRIORITY_DEFAULT, 16, () => this._step());
        }
        this._updateEdges();
    }

    stopEdge() {
        if (this._raf) { GLib.Source.remove(this._raf); this._raf = 0; }
        if (!this._edgeDir) return;
        this._edgeDir = 0;
        this._edgeDepth = 0;
        this._updateEdges();
        this._updateMarker();
    }

    _step() {
        this._raf = 0;
        if (!this._edgeDir) return GLib.SOURCE_REMOVE;
        if (!this._alive()) { this.stopEdge(); return GLib.SOURCE_REMOVE; }

        const now = GLib.get_monotonic_time() / 1000;
        const dt = Math.min(now - this._last, 60) / 1000;
        this._last = now;

        this._view = this._clampV(this._view + this._edgeDir * edgeSpeed(this._edgeDepth) * dt);
        const r = Math.round(this._view);
        if (r !== this._highlight) this._highlight = r;
        this.layout();

        if (this._view <= this._minV() || this._view >= this._maxV()) {
            this.stopEdge();
            return GLib.SOURCE_REMOVE;
        }
        this._raf = GLib.timeout_add(GLib.PRIORITY_DEFAULT, 16, () => this._step());
        return GLib.SOURCE_REMOVE;
    }

    // ---------- pointer ----------
    over(y) {
        const rowsTop = this._rowsEl.y;
        const rowsH = this._rowsEl.height;

        if (this._isStrip()) {
            if (y < rowsTop && this._view > this._minV() + 0.001) {
                this.setEdge(-1, clamp((rowsTop - y) / Math.max(1, rowsTop), 0, 1));
                return this._highlight;
            }
            if (y > rowsTop + rowsH && this._view < this._maxV() - 0.001) {
                const below = Math.max(1, this._hostH - rowsTop - rowsH);
                this.setEdge(1, clamp((y - rowsTop - rowsH) / below, 0, 1));
                return this._highlight;
            }
        }
        this.stopEdge();

        if (y < rowsTop - 2 || y > rowsTop + rowsH + 2) return -1;

        const idx = Math.round((y - rowsTop) / ROW_SLOT - (this._slots - 1) / 2 + this._view);
        if (idx < 0 || idx >= this._items.length) return -1;
        this.setHighlight(idx);
        return idx;
    }

    clearHover() { this.setHighlight(-1); }

    destroy() {
        this.stopEdge();
        for (const [, el] of this._rowEls) el.destroy();
        this._rowEls.clear();
        this._items = [];
    }
}

const SOLARIZED = [
    '#268bd2', '#cb4b16', '#859900', '#d33682',
    '#2aa198', '#b58900', '#6c71c4', '#dc322f',
];

export function colorForApp(appName) {
    const s = appName || '';
    let h = 0;
    for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) | 0;
    return SOLARIZED[Math.abs(h) % SOLARIZED.length];
}

// ==================== app / icon resolution ====================
function resolveApp(window) {
    return Shell.WindowTracker.get_default().get_window_app(window)
        || Shell.AppSystem.get_default().lookup_app(window.get_wm_class());
}

export function getWindowApp(window) { return resolveApp(window); }

export function getWindowAppIcon(window) {
    const app = resolveApp(window);
    if (app?.get_app_info()?.get_icon()) return app.get_icon();
    return window.get_gicon() || new Gio.ThemedIcon({ name: 'applications-system-symbolic' });
}

    const app = resolveApp(window);
    if (app?.get_app_info()?.get_icon()) return app.create_icon_texture(iconSize);
    return St.TextureCache.get_default().load_gicon(null, getWindowAppIcon(window), iconSize);
}

/**

 * top-right corner, sized at 45% of the icon (no small cap — the badge
 * must be clearly readable), and the wrapper is exactly icon-sized so a
 * badged group occupies the same cell as a plain window icon.

        style_class: 'ws-ico-wrap',
        width: size, height: size,
    });

            style_class: 'ws-badge-label',
            text: String(count),
            x_align: Clutter.ActorAlign.CENTER,
            y_align: Clutter.ActorAlign.CENTER,
        });
        label.set_style(`font-size: ${Math.max(10, Math.round(badgeSize * 0.58))}px;`);

            style_class: 'ws-badge',

            y_align: Clutter.ActorAlign.CENTER,

        badge.set_style(`border-radius: ${Math.round(badgeSize / 2)}px;`);
        wrap.add_child(badge);
        badge.set_position(size - badgeSize, 0);
    }

    return { actor: wrap, icon };
}

export function createRail() {
    const root = new St.Widget({ style_class: 'ws-flyout-rail', reactive: false });

    // Track and marker widths scale with the enlarged flyout, so they must
    // be re-centred whenever the rail is resized. Hook to the root's
    // allocation so both stay correct on any panel size.
    const TRACK_W = 4;
    const MARKER_W = 40;
    const MARKER_H = 12;

    const track = new St.Widget({ style_class: 'ws-flyout-rail-track' });
    root.add_child(track);

    const ticks = new St.Widget({ reactive: false });
    root.add_child(ticks);

    const marker = new St.Widget({
        style_class: 'ws-flyout-rail-marker',
        width: MARKER_W, height: MARKER_H,
    });
    root.add_child(marker);

    root.connect('notify::allocation', () => {
        const alloc = root.get_allocation_box();
        const w = Math.round(alloc.x2 - alloc.x1);
        const h = Math.round(alloc.y2 - alloc.y1);
        if (w <= 0 || h <= 0) return;
        track.set_position(Math.round((w - TRACK_W) / 2), 0);
        track.set_size(TRACK_W, h);
        marker.x = Math.round((w - MARKER_W) / 2);
    });

    return { root, track, ticks, marker };
}

const PREVIEW = { TITLE_MIN: 40, TITLE_MAX: 100, TITLE_FRACTION: 0.2 };

export function createClonePreview(window, targetHeight, options = {}) {
    if (!window) return null;
    const windowActor = window.get_compositor_private();
    if (!windowActor) return null;

    const frame = window.get_frame_rect();
    const buffer = window.get_buffer_rect();
    if (frame.height === 0) return null;

    const width = targetHeight * (frame.width / frame.height);
    const scale = targetHeight / frame.height;
    const left = (frame.x - buffer.x) * scale;
    const top = (frame.y - buffer.y) * scale;
    const right = ((buffer.x + buffer.width) - (frame.x + frame.width)) * scale;
    const bottom = ((buffer.y + buffer.height) - (frame.y + frame.height)) * scale;

    const inner = new St.BoxLayout({
        style_class: 'collection-preview-inner',
        width, height: targetHeight, clip_to_allocation: true,
    });

    const clone = new Clutter.Clone({
        source: windowActor,
        width: width + left + right,
        height: targetHeight + top + bottom,
    });
    clone.set_position(-left, -top);

    const overlay = new Clutter.Actor();
    overlay.add_child(clone);
    inner.add_child(overlay);

    if (options.showTitle !== false) {
        const raw = window.get_title();
        const text = raw && raw.trim() ? raw : 'Untitled';
        const titleHeight = Math.min(PREVIEW.TITLE_MAX,
            Math.max(PREVIEW.TITLE_MIN, targetHeight * PREVIEW.TITLE_FRACTION));
        const title = new St.Label({
            style_class: 'clone-title-overlay', text,
            x_align: Clutter.ActorAlign.FILL, y_align: Clutter.ActorAlign.CENTER,
        });
        title.clutter_text.set_x_align(Clutter.ActorAlign.CENTER);
        title.clutter_text.set_y_align(Clutter.ActorAlign.CENTER);
        title.clutter_text.set_line_wrap(true);
        title.set_size(width, titleHeight);
        title.set_position(0, (targetHeight - titleHeight) / 2);
        overlay.add_child(title);
    }

    let closeButton = null;
    if (options.closeButtonSize) {
        const size = options.closeButtonSize;
        closeButton = new St.Button({
            style_class: 'window-close-button',
            child: new St.Icon({ icon_name: 'window-close-symbolic', icon_size: size }),
            x_align: Clutter.ActorAlign.END, y_align: Clutter.ActorAlign.START,
            reactive: true,
        });
        closeButton.set_position(width - (options.closeButtonOffsetX ?? (size + 14)),
            options.closeButtonOffsetY ?? 10);
        overlay.add_child(closeButton);
    }

    if (!options.wrapperStyleClass)
        return { actor: inner, inner, overlay, closeButton, width, height: targetHeight };

    const wrapper = new St.BoxLayout({
        style_class: options.wrapperStyleClass,
        reactive: true, track_hover: true,
    });
    wrapper.add_child(inner);
    return { actor: wrapper, wrapper, inner, overlay, closeButton, width, height: targetHeight };
}

// ==================== insertion math ====================
export function computeInsertion(rects, pointerX) {
    if (rects.length === 0) return { index: 0, neighbor: null, insertBefore: true };

    const first = rects[0];
    const last = rects[rects.length - 1];
    let target = last, before = false;

    if (pointerX < first.x) { target = first; before = true; }
    else if (pointerX < last.right) {
        for (let i = 0; i < rects.length; i++) {
            const r = rects[i], next = rects[i + 1];
            if (pointerX >= r.x && pointerX < r.right) { target = r; before = pointerX < r.mid; break; }
            if (next && pointerX >= r.right && pointerX <= next.x) {
                before = (pointerX - r.right) > (next.x - pointerX);
                target = before ? next : r;
                break;
            }
        }
    }
    const i = rects.indexOf(target);
    return { index: before ? i : i + 1, neighbor: target.item, insertBefore: before };
}

// ==================== text helpers ====================
function escapeMarkup(s) {
    return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}

export function highlightMarkup(text, indices) {
    if (!indices || indices.size === 0) return escapeMarkup(text);
    let out = '', open = false, i = 0;
    for (const ch of text) {
        const hit = indices.has(i);
        if (hit && !open) { out += '<b>'; open = true; }
        else if (!hit && open) { out += '</b>'; open = false; }
        out += escapeMarkup(ch);
        i += ch.length;
    }
    return open ? out + '</b>' : out;
}

// ==================== USER ORDER ====================
const ranks = new Map();
let nextRank = 1;

function isEligible(w) {
    return !w.skip_taskbar && !w.is_override_redirect();
}

// ==================== LAST FOCUSED PER WORKSPACE ====================
// The shell only reports a focused window while one actually has focus.
// Opening a window overlay right after switching workspaces (or from a
// different workspace) therefore sees none, and "start at the focused
// window" degrades to "start at the first window". Remember the last
// focused window per workspace so overlays still open on the window the
// user was last working with on that workspace.
const lastFocusedByWs = new Map();   // workspace index -> Meta.Window

export function noteFocusedWindow(window) {
    if (!window) return;
    try {
        const ws = window.get_workspace();
        if (ws) lastFocusedByWs.set(ws.index(), window);
    } catch (e) { /* dead window object */ }
}

export function lastFocusedWindow(workspaceIndex) {
    const win = lastFocusedByWs.get(workspaceIndex);
    if (!win) return null;
    try {
        // A closed window's wrapper throws on method access.
        if (!win.get_workspace()) { lastFocusedByWs.delete(workspaceIndex); return null; }
    } catch (e) {
        lastFocusedByWs.delete(workspaceIndex);
        return null;
    }
    return win;
}

// ==================== APP IDENTITY / GROUPING ====================
// `appId` is the grouping key. It is stable for the lifetime of the app
// (Shell.App id) and degrades to the WM class when the tracker has not
// resolved the window yet.

export function appIdentity(window) {
    const app = Shell.WindowTracker.get_default().get_window_app(window);
    if (app) {
        const id = app.get_id();
        if (id) return { id, name: app.get_name() || id };
    }
    const wm = window.get_wm_class() || window.get_wm_class_instance() || 'unknown';
    return { id: `wm:${wm}`, name: wm };
}

export function groupKey(workspaceIndex, appId) {
    return `${workspaceIndex}::${appId}`;
}

/**
 * Same-app windows inside one workspace, in model order.
 * Returns [{ appId, appName, key, windows, indices, count }] where
 * `indices[k]` is the position of `windows[k]` inside `entry.windows`.
 */
export function groupWindowsInWorkspace(entry) {
    const out = [];
    const byId = new Map();
    entry.windows.forEach((w, i) => {
        const { id, name } = appIdentity(w);
        let g = byId.get(id);
        if (!g) {
            g = {
                appId: id, appName: name,
                key: groupKey(entry.index, id),
                windows: [], indices: [], count: 0,
            };
            byId.set(id, g);
            out.push(g);
        }
        g.windows.push(w);
        g.indices.push(i);
        g.count++;
    });
    return out;
}

// ==================== THE REPRESENTATION ====================
export function buildModel() {
    const live = Display.list_all_windows().filter(isEligible);

    const liveSet = new Set(live);
    for (const w of [...ranks.keys()])
        if (!liveSet.has(w)) ranks.delete(w);
    for (const w of live)
        if (!ranks.has(w)) ranks.set(w, nextRank++);

    // Drop remembered focus entries whose window is gone.
    for (const [idx, win] of lastFocusedByWs)
        if (!liveSet.has(win)) lastFocusedByWs.delete(idx);

    const workspaces = [];
    for (let i = 0; i < WorkspaceManager.n_workspaces; i++) {
        const workspace = WorkspaceManager.get_workspace_by_index(i);
        const windows = live
            .filter(w => w.located_on_workspace(workspace))
            .sort((a, b) => ranks.get(a) - ranks.get(b));

        const entry = {
            workspace, index: i, windows,
            // Display mode only: `true` renders one icon per window,
            // `false` renders one badged icon per app.
            direct: windows.length <= DIRECT_MODE_MAX_WINDOWS,
        };
        entry.groups = groupWindowsInWorkspace(entry);
        workspaces.push(entry);
    }
    return { workspaces, drag: intent.drag, hintWs: intent.hintWs };
}

// ==================== MODEL MUTATIONS ====================
export function placeWindow(window, workspace, index = null) {
    const fullList = buildModel().workspaces[workspace.index()].windows;
    const list = fullList.filter(w => w !== window);

    let at = index === null ? list.length : index;
    at = Math.max(0, Math.min(at, list.length));

    list.splice(at, 0, window);

    for (const w of list) ranks.set(w, nextRank++);
}

let _signals = [];
let _wsSignals = [];
let _settings = null;

function _listen(obj, signal, cb) { _signals.push([obj, obj.connect(signal, cb)]); }

function _watchWorkspaces() {
    for (const [obj, id] of _wsSignals) obj.disconnect(id);
    _wsSignals = [];
    for (let i = 0; i < WorkspaceManager.n_workspaces; i++) {
        const ws = WorkspaceManager.get_workspace_by_index(i);
        _wsSignals.push([ws, ws.connect('window-added', () => requestRender())]);
        _wsSignals.push([ws, ws.connect('window-removed', () => requestRender())]);
    }
}

export function connectModelSignals(settings = null) {
    disconnectModelSignals();
    _settings = settings;

    _listen(WorkspaceManager, 'notify::n-workspaces', () => { _watchWorkspaces(); requestRender(); });
    _listen(WorkspaceManager, 'workspace-switched', () => requestRender());
    _listen(Display, 'window-created', () => requestRender());
    _listen(Display, 'notify::focus-window', () => noteFocusedWindow(Display.focus_window));
    _listen(Main.sessionMode, 'updated', () => requestRender());
    if (settings)
        _listen(settings, 'changed', () => requestRender());

    _watchWorkspaces();
    noteFocusedWindow(Display.focus_window);   // seed with the current focus
}

export function disconnectModelSignals() {
    for (const [obj, id] of _wsSignals) obj.disconnect(id);
    for (const [obj, id] of _signals) obj.disconnect(id);
    _wsSignals = [];
    _signals = [];
    _settings = null;
}

// Window/workspace actions that are neither model state nor overlay logic.
// Shared by the DBus adapter (CycleWindowsOnWorkspace) and the top-bar
// scroll handler.

import Meta from 'gi://Meta';

import * as Model from './model.js';

/**
 * Advance focus through the flat model order — every window across every
 * workspace that is NOT in overflow mode — by `direction` (+1 next,
 * -1 previous), wrapping at both ends.
 *
 * Workspaces with `direct === false` (i.e. >5 windows) are skipped
 * entirely: the strip shows a single count button for them, so there is
 * no per-window order to walk. Windows on overflow workspaces are simply
 * not in the cycle.
 *
 * Sticky windows (`on_all_workspaces`) appear in every workspace's list;
 * they are deduplicated so a flat walk hits each one once.
 *
 * The anchor is the focused window's position in that flat list; when it
 * isn't in the cycle (focused on an overflow workspace, on a desktop, or
 * no focus at all) we start at the end the direction points from.
 *
 * Does NOT maximize — this is "scroll through the strip", not the DBus
 * "cycle" action.
 */
export function scrollModelWindows(direction = 1) {
    // const activeIdx = global.workspace_manager.get_active_workspace_index();
    // if (!model.workspaces[activeIdx]?.direct) return false;
    const model = Model.buildModel();

    const windows = [];
    const seen = new Set();
    for (const entry of model.workspaces) {
        if (!entry.direct) continue;
        for (const w of entry.windows) {
            const id = w.get_id();
            if (seen.has(id)) continue;
            seen.add(id);
            windows.push(w);
        }
    }
    if (windows.length === 0) return false;

    const n = windows.length;
    const idx = windows.indexOf(Model.Display.focus_window);
    const nextIdx = idx === -1
        ? (direction > 0 ? 0 : n - 1)
        : ((idx + direction) % n + n) % n;
    const target = windows[nextIdx];

    if (target.minimized) target.unminimize();
    const ws = target.get_workspace();
    if (ws) ws.activate_with_focus(target, global.get_current_time());
    else target.activate(global.get_current_time());
    return true;
}

/**
 * Focus the next (or previous) window on a *specific* workspace, in model
 * order. Kept workspace-scoped and maximizing for the DBus
 * CycleWindowsOnWorkspace method, whose contract is "cycle + focus at full
 * size" and is independent of the display mode.
 */
export function cycleWindowsOnWorkspace(workspaceIndex, direction = 1) {
    const entry = Model.buildModel().workspaces[workspaceIndex];
    if (!entry || entry.windows.length === 0) return false;

    const windows = entry.windows;
    const n = windows.length;
    const idx = windows.indexOf(Model.Display.focus_window);

    const nextIdx = idx === -1
        ? (direction > 0 ? 0 : n - 1)
        : ((idx + direction) % n + n) % n;
    const target = windows[nextIdx];

    if (target.minimized) target.unminimize();
    target.maximize(Meta.MaximizeFlags.BOTH);
    target.activate(global.get_current_time());
    return true;
}
// ===== FILE: ./lib/fuzzy.js =====
// Fuzzy string matching for the search overlays. Pure: no imports, no state,
// no actors. Both WindowSearchOverlay and AppSearchOverlay run their queries
// through fuzzyMatch. Returns matched character offsets so callers can bold
// them in their markup.
//
// Order of attempts per token: substring > subsequence > typo-tolerant
// word-prefix.

function subsequenceMatch(token, text) {
    const positions = [];
    let qi = 0, score = 0, consecutive = 0;
    for (let ti = 0; ti < text.length && qi < token.length; ti++) {
        if (text[ti] === token[qi]) { score += 1 + consecutive; consecutive++; qi++; positions.push(ti); }
        else consecutive = 0;
    }
    return qi < token.length ? { matched: false } : { matched: true, score, positions };
}

// Optimal-string-alignment distance (insert / delete / substitute / transpose).
function osaDistance(a, b) {
    const d = Array.from({ length: a.length + 1 }, (_, i) => [i, ...new Array(b.length).fill(0)]);
    for (let j = 0; j <= b.length; j++) d[0][j] = j;
    for (let i = 1; i <= a.length; i++) {
        for (let j = 1; j <= b.length; j++) {
            const cost = a[i - 1] === b[j - 1] ? 0 : 1;
            d[i][j] = Math.min(d[i - 1][j] + 1, d[i][j - 1] + 1, d[i - 1][j - 1] + cost);
            if (i > 1 && j > 1 && a[i - 1] === b[j - 2] && a[i - 2] === b[j - 1])
                d[i][j] = Math.min(d[i][j], d[i - 2][j - 2] + 1);
        }
    }
    return d[a.length][b.length];
}

function typoMatch(token, text) {
    if (token.length < 4) return null;
    const limit = token.length >= 6 ? 2 : 1;
    for (const m of text.matchAll(/[^\s\-_.\/:—]+/g)) {
        const word = m[0];
        for (let len = token.length - 1; len <= token.length + 1; len++) {
            if (len < 1 || len > word.length) continue;
            if (osaDistance(token, word.slice(0, len)) <= limit) {
                const positions = [];
                for (let i = 0; i < word.length; i++) positions.push(m.index + i);
                return { score: 10, positions };
            }
        }
    }
    return null;
}

export function fuzzyMatch(query, text) {
    const tokens = query.trim().toLowerCase().split(/\s+/).filter(Boolean);
    const indices = new Set();
    if (tokens.length === 0) return { matched: true, score: 0, indices };

    const lower = text.toLowerCase();
    let score = 0;
    for (const token of tokens) {
        const idx = lower.indexOf(token);
        if (idx !== -1) {
            const boundary = idx === 0 || /[\s\-_.\/]/.test(lower[idx - 1]);
            score += 60 - Math.min(idx, 40) + (boundary ? 20 : 0);
            for (let i = idx; i < idx + token.length; i++) indices.add(i);
            continue;
        }
        const sub = subsequenceMatch(token, lower);
        if (sub.matched) { score += sub.score; sub.positions.forEach(p => indices.add(p)); continue; }
        const typo = typoMatch(token, lower);
        if (!typo) return { matched: false, score: -1, indices: new Set() };
        score += typo.score;
        typo.positions.forEach(p => indices.add(p));
    }
    return { matched: true, score, indices };
}

import GLib from 'gi://GLib';
import Gio from 'gi://Gio';

let loggingEnabled = false;
let logOutput = 'journal';
let uuid = null;
let logFile = null;


/*
 * logger.js is in the extension root.
 */
const extensionDir = GLib.path_get_dirname(
    GLib.filename_from_uri(import.meta.url)[0]
);


/*
 * Initialize logging.
 *
 * output:
 *   'journal' -> GNOME Journal only
 *   'file'    -> file only
 *   'both'    -> GNOME Journal + file
 *
 * enabled:
 *   true  -> logging enabled
 *   false -> logging disabled
 */
function initLogging(
    extensionUuid,
    output = 'journal',
    enabled = true
) {
    uuid = extensionUuid;
    logOutput = output;
    loggingEnabled = enabled;

    if (output === 'file' || output === 'both') {
        logFile = GLib.build_filenamev([
            GLib.get_home_dir(),
            `${uuid}.log`,
        ]);
    } else {
        logFile = null;
    }
}


/*
 * Create a logger for a source file.
 *
 * Usage:
 *
 * const journal = createLogger(import.meta.url);
 */
function createLogger(source) {
    let sourceFile;

    try {
        sourceFile = GLib.filename_from_uri(source)[0];
    } catch (e) {
        sourceFile = source;
    }

    /*
     * Convert absolute path into a path
     * relative to the extension root.
     */
    const prefix = `${extensionDir}/`;

    if (sourceFile.startsWith(prefix)) {
        sourceFile = sourceFile.substring(prefix.length);
    }

    const relativeSource = sourceFile;


    return function journal(msg) {

        /*
         * Logging disabled?
         *
         * Do absolutely nothing.
         */
        if (!loggingEnabled)
            return;


        /*
         * Create timestamp once.
         */
        // const timestamp = GLib.DateTime
        //     .new_now_local()
        //     .format('%Y-%m-%d %H:%M:%S');


        /*
         * Create the complete message once.
         *
         * Both Journal and file receive
         * exactly this same string.
         */
        // const output = `[${timestamp}] [${relativeSource}] ${msg}`;
        const output = `[${relativeSource}] ${msg}`;


        if (
            logOutput === 'journal' ||
            logOutput === 'both'
        ) {
            writeToJournal(output);
        }


        if (
            logOutput === 'file' ||
            logOutput === 'both'
        ) {
            writeToFile(output);
        }
    };
}


function writeToJournal(output) {
    GLib.log_structured(
        uuid,
        GLib.LogLevelFlags.LEVEL_MESSAGE,
        {
            MESSAGE: output,
            SYSLOG_IDENTIFIER: uuid,
        }
    );
}


function writeToFile(output) {
    if (!logFile)
        return;

    try {
        const file = Gio.File.new_for_path(logFile);

        const stream = file.append_to(
            Gio.FileCreateFlags.NONE,
            null
        );

        const bytes =
            new TextEncoder().encode(`${output}\n`);

        stream.write_all(bytes, null);
        stream.close(null);

    } catch (e) {
        /*
         * Don't call journal() here.
         */
        console.error(
            `[${uuid}] Failed to write log file: ${e}`
        );
    }
}

    initLogging,
    createLogger,
};

// ===== FILE: ./extension.js =====
import Clutter from 'gi://Clutter';
import GLib from 'gi://GLib';
import * as ExtensionModule from 'resource:///org/gnome/shell/extensions/extension.js';
import * as Main from 'resource:///org/gnome/shell/ui/main.js';

import * as WorkspaceIndicatorModule from './lib/workspaceIndicator.js';
import * as OverlayService from './lib/overlayService.js';
import * as DBusService from './lib/dbusService.js';
import * as Interactions from './lib/interactions.js';
import * as Model from './lib/model.js';
import * as WorkspaceActions from './lib/workspaceActions.js';

import { initExternalDrag, destroyExternalDrag } from './lib/externalDrag.js'

import { initLogging, createLogger } from './logger.js';

const journal = createLogger(import.meta.url);

// Minimum time between two accepted scroll steps.
const SCROLL_THROTTLE_MS = 150;

export default class TopNotchWorkspaces extends ExtensionModule.Extension {
    constructor(metadata) {
        super(metadata);
        this._indicator = null;
        this._titleBarMoveMonitor = null;
        this._lastScrollTime = 0;
    }

    enable() {
        initLogging(this.uuid, 'file', true);
        journal(`Enabled`);

        const settings = this.getSettings();

        Model.connectModelSignals(settings);
        initExternalDrag();

        OverlayService.initOverlayService(settings);
        DBusService.initDBusService();

        // Commands the indicator can invoke. `showGroupWindows` is the
        // click path on a badged app group: the full selector, scoped to
        // that workspace-local group. The drag path never reaches here —
        // it goes to GroupFlyout instead.
        this._indicator = new WorkspaceIndicatorModule.WorkspaceIndicator(settings, {
            showAllApps: OverlayService.showAllApps,
            showAllWindows: OverlayService.showAllWindows,
            showWorkspaceWindows: OverlayService.showWorkspaceWindows,
            showGroupWindows: OverlayService.showGroupWindows,
        });
        Main.panel.addToStatusArea('workspace-indicator', this._indicator, 0, 'left');

        this._titleBarMoveMonitor = new WorkspaceIndicatorModule.TitleBarMoveMonitor();

        this._lastScrollTime = 0;
        Main.panel.connectObject('scroll-event',
            (_actor, event) => this._handleScroll(event), this);
    }

    disable() {
        Main.panel.disconnectObject(this);

        DBusService.destroyDBusService();
        OverlayService.destroyOverlayService();

        if (this._titleBarMoveMonitor) {
            this._titleBarMoveMonitor.destroy();
            this._titleBarMoveMonitor = null;
        }
        if (this._indicator) {
            this._indicator.destroy();
            this._indicator = null;
        }

        destroyExternalDrag();
        Model.disconnectModelSignals();
        Interactions.destroyTooltip();
    }

    // Top-bar scroll: step focus through the model's flat window order —
    // every window across every non-overflow workspace, deduped for sticky
    // windows. Workspaces with >5 windows (`direct === false`) are excluded
    // entirely, so their windows are not in the cycle.
    _handleScroll(event) {
        if (event.type() !== Clutter.EventType.SCROLL)
            return Clutter.EVENT_PROPAGATE;

        let step;
        switch (event.get_scroll_direction()) {
            case Clutter.ScrollDirection.UP:
            case Clutter.ScrollDirection.LEFT:
                step = -1;
                break;
            case Clutter.ScrollDirection.DOWN:
            case Clutter.ScrollDirection.RIGHT:
                step = 1;
                break;
            default:
                return Clutter.EVENT_PROPAGATE;
        }

        const now = GLib.get_monotonic_time() / 1000;
        if (now - this._lastScrollTime < SCROLL_THROTTLE_MS)
            return Clutter.EVENT_STOP;
        this._lastScrollTime = now;

        WorkspaceActions.scrollModelWindows(step);
        return Clutter.EVENT_STOP;
    }
}

import Clutter from 'gi://Clutter';

import { TimeoutDelay, SettleAnimationDuration } from './shellGlobals.js';

import { createLogger } from '../logger.js';
const journal = createLogger(import.meta.url);

/**
 * Fade in an actor from opacity 0 to 255.
 * Safe: skips if actor is invalid or not on stage.
 */
export function fadeInActor(actor, duration = TimeoutDelay, mode = Clutter.AnimationMode.EASE_OUT_QUAD) {
    if (!actor) {
        journal('[fadeInActor] called with null actor');
        return;
    }
    try {
        // If the actor isn't on stage, it's likely being destroyed – skip
        if (!actor.get_stage()) {
            journal('[fadeInActor] actor not on stage, skipping');
            return;
        }
        actor.remove_all_transitions();
        actor.opacity = 0;
        journal(`[fadeInActor] fading in ${actor} (0→255) duration ${duration}ms`);
        actor.ease({
            opacity: 255,
            duration: duration,
            mode: mode,
        });
    } catch (e) {
        journal(`[fadeInActor] error: ${e}`);
    }
}

/**
 * "Settle" an icon: scale from 0.85 → 1.0, opacity 160 → 255.
 * Safe: skips if actor is invalid or not on stage.
 */
export function settleIcon(actor, duration = SettleAnimationDuration, mode = Clutter.AnimationMode.EASE_OUT_QUAD) {
    if (!actor) {
        journal('[settleIcon] called with null actor');
        return;
    }
    try {
        const stage = actor.get_stage();
        if (!stage) {
            journal('[settleIcon] actor not on stage, skipping');
            return;
        }
        journal(`[settleIcon] actor on stage, applying animation`);
        // Remove any leftover drag styling – ignore errors
        try {
            actor.remove_style_class_name('reorder-drag-source');
        } catch (e) {
            // actor may not have the style or may be destroyed
        }
        actor.remove_all_transitions();
        actor.set_pivot_point(0.5, 0.5);
        actor.set_scale(0.85, 0.85);
        actor.opacity = 160;
        journal(`[settleIcon] animating ${actor} (scale 0.85→1.0, opacity 160→255) duration ${duration}ms`);
        actor.ease({
            scale_x: 1,
            scale_y: 1,
            opacity: 255,
            duration: duration,
            mode: mode,
        });
    } catch (e) {
        journal(`[settleIcon] error: ${e}`);
    }
}

// ==================== ACTIVE PREVIEW TRACKER ====================
// Only one icon has its hover preview / title popup open at a time; the
// CTRL poll (which swaps between them) only runs while one is active.
class ActivePreviewTracker {
    static activePreview = null;
    static _ctrlPollId = null;
    static _ctrlPressed = false;

    static register(preview) {
        if (this.activePreview && this.activePreview !== preview)
            this.activePreview.forceHidePreview();
        this.activePreview = preview;
        this._startPoll();
    }

    static unregister(preview) {
        if (this.activePreview === preview) {
            this.activePreview = null;
            this._stopPoll();
        }
    }

    static _readCtrl() {
        const [, , mods] = global.get_pointer();
        return (mods & Clutter.ModifierType.CONTROL_MASK) !== 0;
    }

    static _startPoll() {
        if (this._ctrlPollId || !this.activePreview) return;
        this._ctrlPressed = this._readCtrl();
        this._ctrlPollId = GLib.timeout_add(GLib.PRIORITY_DEFAULT, TimeoutDelay, () => {
            if (!this.activePreview) { this._ctrlPollId = null; return GLib.SOURCE_REMOVE; }
            const ctrl = this._readCtrl();
            if (ctrl !== this._ctrlPressed) {
                this._ctrlPressed = ctrl;
                this.activePreview?.onCtrlChanged(ctrl);
            }
            return GLib.SOURCE_CONTINUE;
        });
    }

    static _stopPoll() {
        if (this._ctrlPollId) { GLib.Source.remove(this._ctrlPollId); this._ctrlPollId = null; }
    }
}

// Hovering an EXTERNAL drag (files, text) over an icon for ~400 ms raises its window.
class ExternalDragAutoActivator {
    constructor(window) { this._window = window; this._timeoutId = 0; this._lastHover = 0; }

    notifyDragOver() {
        this._lastHover = GLib.get_monotonic_time();
        if (this._timeoutId) return;
        this._timeoutId = GLib.timeout_add(GLib.PRIORITY_DEFAULT, 400, () => {
            this._timeoutId = 0;
            if ((GLib.get_monotonic_time() - this._lastHover) / 1000 > 500)
                return GLib.SOURCE_REMOVE;
            const win = this._window;
            if (win.minimized) win.unminimize();
            win.get_workspace().activate_with_focus(win, global.get_current_time());
            return GLib.SOURCE_REMOVE;
        });
    }

    destroy() {
        if (this._timeoutId) { GLib.Source.remove(this._timeoutId); this._timeoutId = 0; }
    }
}

// ==================== WINDOW ICON ====================
// A plain object wrapping one St.Button. Created fresh on every render;
// it owns hover preview / title popup / context menu for its window.
class WindowIcon {
    constructor(window, settings) {
        this._window = window;
        this._settings = settings;
        this._hooks = null; // { handleDragOver, acceptDrop } from the owning thumbnail
        this._preview = null;
        this._popup = null;
        this._hoverTimeoutId = 0;
        this._cleanupId = 0;
        this._dragActivator = new ExternalDragAutoActivator(window);

        const actor = this.actor = new St.Button({
            style_class: 'window-preview-icon', reactive: true, track_hover: true,
        });
        actor._icon = this;

        const iconSize = settings.get_int('icon-size');
        this._setIcon(iconSize);

        // Forward DnD to the thumbnail behind us; the thumbnail classifies.
        // NOTE: Shell passes the draggable's actor._delegate as `source`.
        const delegate = {
            handleDragOver: (source, a, x, y, time) => {
                if (source === actor._delegate) return DragMotionResult.CONTINUE;
                if (!source?._window) this._dragActivator.notifyDragOver();
                return this._hooks?.handleDragOver(source, a, x, y, time) ?? DragMotionResult.CONTINUE;
            },
            acceptDrop: (source, a, x, y, time) =>
                this._hooks?.acceptDrop(source, a, x, y, time) ?? false,
        };
        // Draggable must connect before our press handler so DND sees presses first.
        makeWindowDraggable(actor, window, iconSize, { delegate, sourceIcon: true });

        actor.connect('notify::hover', () => this._onHoverChange());
        actor.connect('button-press-event', (a, event) => this._onButtonPressed(event));
        actor.connect('button-release-event', (a, event) => this._onButtonReleased(event));
        actor.connect('destroy', () => this._onDestroy());

        // Anything that should change how this icon looks → full re-render.
        this._winSignals = ['notify::wm-class', 'notify::mapped', 'notify::appears-focused']
            .map(sig => window.connect(sig, () => requestRender()));

        if (window.appears_focused)
            actor.add_style_class_name('focused');
    }

    setHooks(hooks) { this._hooks = hooks; }

    _setIcon(iconSize) {
        const iconActor = createWindowIconTexture(this._window, iconSize);
        this.actor.set_child(iconActor);
        const id = iconActor.connect('stage-views-changed', () => {
            const rect = new Mtk.Rectangle();
            [rect.x, rect.y] = iconActor.get_transformed_position();
            [rect.width, rect.height] = iconActor.get_transformed_size();
            this._window.set_icon_geometry(rect);
            iconActor.disconnect(id);
        });
    }

    // ---------- hover preview / title popup ----------
    _isShowing() { return !!(this._preview || this._popup); }

    onCtrlChanged(ctrl) {
        if (ctrl) { this._showPopup(); this._hidePreview(); }
        else { this._showPreview(); this._hidePopup(); }
    }

    forceHidePreview() {
        this._stopCleanup();
        ActivePreviewTracker.unregister(this);
        this._hidePreview();
        this._hidePopup();
    }

    _onHoverChange() {
        if (this._hoverTimeoutId) { GLib.Source.remove(this._hoverTimeoutId); this._hoverTimeoutId = 0; }
        if (intent.drag) return;
        if (this.actor.hover) {
            this._stopCleanup();
            if (this._isShowing()) { this.onCtrlChanged(ActivePreviewTracker._ctrlPressed); return; }
            // 30 ms debounce: don't flash a preview while the pointer passes through.
            this._hoverTimeoutId = GLib.timeout_add(GLib.PRIORITY_DEFAULT, 30, () => {
                this._hoverTimeoutId = 0;
                ActivePreviewTracker.register(this);
                if (this.actor.hover || this._isShowing()) {
                    if (ActivePreviewTracker._ctrlPressed) this._showPopup();
                    else this._showPreview();
                }
                return GLib.SOURCE_REMOVE;
            });
        } else if (this._isShowing()) {
            this._startCleanup();
        }
    }

    _onPreviewHover(isHovered) {
        if (isHovered) this._stopCleanup();
        else if (!this.actor.hover) this._startCleanup();
    }

    _startCleanup() {
        this._stopCleanup();
        this._cleanupId = GLib.timeout_add(GLib.PRIORITY_DEFAULT, TimeoutDelay, () => {
            this._cleanupId = 0;
            if (this.actor.hover || this._preview?.hover || this._popup?.hover)
                return GLib.SOURCE_REMOVE;
            ActivePreviewTracker.unregister(this);
            this._hidePreview();
            this._hidePopup();
            return GLib.SOURCE_REMOVE;
        });
    }

    _stopCleanup() {
        if (this._cleanupId) { GLib.Source.remove(this._cleanupId); this._cleanupId = 0; }
    }

    _showPreview() {
        if (this._preview) return;
        const win = this._window;
        const previewHeight = this._settings.get_int('hover-preview-height');
        const built = createClonePreviewActor(win, previewHeight, {
            wrapperStyleClass: 'hover-preview-wrapper',
            showTitle: false,
            onClose: w => { w.delete(global.get_current_time()); this._hidePreview(); },
            closeButtonSize: this._settings.get_int('close-button-size'),
            closeButtonOffsetX: 60,
            closeButtonOffsetY: 10,
            onHoverChange: h => this._onPreviewHover(h),
            onActivate: () => {
                win.get_workspace().activate_with_focus(win, 0);
                this._hidePreview();
            },
        });
        if (!built) return;

        const anchorWidth = this.actor.get_width();
        const [anchorX] = this.actor.get_transformed_position();
        const previewX = Math.max(0, anchorX + (anchorWidth - built.width) / 2);
        const previewY = screenHeight - previewHeight - 200 + 55;

        built.actor.set_position(previewX, previewY);
        this._preview = built.actor;
        Main.layoutManager.addChrome(this._preview);
        fadeInActor(this._preview);
    }

    _hidePreview() {
        if (!this._preview) return;
        const actor = this._preview;
        this._preview = null;
        Main.layoutManager.removeChrome(actor);
        actor.destroy();
    }

    _showPopup() {
        if (this._popup) return;
        const title = this._window.get_title() || 'Untitled Window';
        const label = new St.Label({ text: title, style_class: 'hover-title-popup', reactive: true, track_hover: true });
        label.set_style(`font-size: ${this._settings.get_int('title-popup-font-size')}pt;`);
        Main.layoutManager.addChrome(label);

        const [iconX] = this.actor.get_transformed_position();
        const padding = 10;
        const labelWidth = Math.min(label.width, screenWidth - 2 * padding);
        let labelX = iconX + (this.actor.width - labelWidth) / 2;
        labelX = Math.max(padding, Math.min(labelX, screenWidth - labelWidth - padding));
        label.set_position(labelX, screenHeight - 200);

        this._popup = label;
        fadeInActor(label);
        label.connect('notify::hover', () => this._onPreviewHover(label.hover));
    }

    _hidePopup() {
        if (!this._popup) return;
        const actor = this._popup;
        this._popup = null;
        Main.layoutManager.removeChrome(actor);
        actor.destroy();
    }

    // ---------- clicks ----------
    // Press: only the right-click menu. A primary press must NOT touch
    // window state, or the resulting re-render destroys this icon mid-press
    // and the drag never starts.
    _onButtonPressed(event) {
        const button = event.get_button();
        if (button === Clutter.BUTTON_SECONDARY) {
            this.forceHidePreview();
            this._openMenu();
            return Clutter.EVENT_STOP;
        }
        // Swallow primary so the thumbnail behind us doesn't also handle it.
        return button === Clutter.BUTTON_PRIMARY ? Clutter.EVENT_STOP : Clutter.EVENT_PROPAGATE;
    }

    // Release: the actual click. After a real drag the release goes to
    // DND's grab, so this only runs for genuine clicks.
    _onButtonReleased(event) {
        if (event.get_button() !== Clutter.BUTTON_PRIMARY)
            return Clutter.EVENT_PROPAGATE;

        this.forceHidePreview();
        const win = this._window;
        const winWs = win.get_workspace();
        if (winWs === WorkspaceManager.get_active_workspace()) {
            if (win.minimized) { win.unminimize(); win.activate_with_workspace(0, winWs); }
            else if (!win.appears_focused || this._isCovered(win)) win.activate_with_workspace(0, winWs);
            else win.minimize();
        } else {
            winWs.activate_with_focus(win, 0);
        }
        return Clutter.EVENT_STOP;
    }

    _openMenu() {
        const win = this._window;
        const menu = openContextMenu(this.actor, m => {
            m.addAction(`Activate ${win.title}`, () => win.get_workspace().activate_with_focus(win, 0));
            m.addAction(`Close ${win.title}`, () => win.delete(0));
            m.addAction(`Close Except ${win.title}`, () => {
                const wmClass = win.get_wm_class();
                const others = Display.get_tab_list(Meta.TabList.NORMAL, win.get_workspace())
                    .filter(w => w !== win && w.get_wm_class() === wmClass && w.get_wm_class_instance() !== 'file_progress');
                const now = global.get_current_time();
                for (const w of others) w.delete(now);
            });

            const app = getWindowApp(win);
            if (!app) return;
            m.addMenuItem(new PopupMenu.PopupSeparatorMenuItem());

            const favorites = AppFavorites.getAppFavorites();
            const isFav = favorites.isFavorite(app.get_id());
            m.addAction(isFav ? `Unpin ${app.get_name()}` : `Pin ${app.get_name()}`, () => {
                if (isFav) favorites.removeFavorite(app.get_id());
                else favorites.addFavorite(app.get_id());
            });

            const appInfo = app.get_app_info();
            const actions = appInfo?.list_actions();
            if (actions && actions.length > 0) {
                m.addMenuItem(new PopupMenu.PopupSeparatorMenuItem());
                for (const action of actions)
                    m.addAction(`${appInfo.get_action_name(action)}`, () => app.launch_action(action, 0, -1));
            }
        });
        if (menu._boxPointer)
            menu._boxPointer.translation_y = -35;
    }

    // True if any visible window stacked above `window` overlaps it, even
    // partially. Dialogs attached to the window itself don't count.
    _isCovered(window) {
        if (window.minimized) return false;
        const workspace = WorkspaceManager.get_active_workspace();
        const stacked = Display.sort_windows_by_stacking(
            Display.list_all_windows().filter(w =>
                !w.minimized &&
                (w.get_window_type() === Meta.WindowType.NORMAL || w.get_window_type() === Meta.WindowType.DIALOG) &&
                w.get_workspace() === workspace));

        const index = stacked.indexOf(window);
        if (index === -1) return false;

        const area = workspace.get_work_area_for_monitor(window.get_monitor());
        const clip = r => ({
            l: Math.max(r.x, area.x), t: Math.max(r.y, area.y),
            r: Math.min(r.x + r.width, area.x + area.width),
            b: Math.min(r.y + r.height, area.y + area.height),
        });

        const t = clip(window.get_frame_rect());
        if (t.r <= t.l || t.b <= t.t) return false;

        for (let i = index + 1; i < stacked.length; i++) {
            const other = stacked[i];
            if (other.get_transient_for() === window) continue;
            const o = clip(other.get_frame_rect());
            if (o.r <= o.l || o.b <= o.t) continue;
            if (t.l < o.r && t.r > o.l && t.t < o.b && t.b > o.t) return true;
        }
        return false;
    }

    _onDestroy() {
        this.forceHidePreview();
        if (this._hoverTimeoutId) { GLib.Source.remove(this._hoverTimeoutId); this._hoverTimeoutId = 0; }
        for (const id of this._winSignals) {
            try { this._window.disconnect(id); } catch (e) { /* window may be gone */ }
        }
        this._winSignals = [];
        this._dragActivator.destroy();
        this._hooks = null;
    }
}

// ==================== WORKSPACE INDICATOR ====================
// Immediate-mode: every state change schedules _render(), which destroys
// the whole UI and rebuilds it as a pure function of buildModel().
export class WorkspaceIndicator extends PanelMenu.Button {
    static { GObject.registerClass(this); }

    constructor(settings) {
        super(0.0, _('Workspace Indicator'));
        this.reactive = false;
        this._settings = settings;
        _instance = this;

        this._renderId = 0;
        this._renderDelay = 0;
        this._leavePollId = 0;
        this._nameHint = null;
        this._thumbsBox = null;
        this._signals = [];
        this._wsSignals = [];

        this._mainBox = new St.BoxLayout({
            style_class: 'workspace-indicator-main-box', y_expand: true, x_expand: true, reactive: true,
        });
        this.add_child(this._mainBox);

        onIntentChange(() => this._queueRender());

        this._on(WorkspaceManager, 'notify::n-workspaces', () => { this._watchWorkspaces(); this._queueRender(); });
        this._on(WorkspaceManager, 'workspace-switched', () => { closeActiveMenu(); this._queueRender(); });
        this._on(Display, 'window-created', () => this._queueRender(TimeoutDelay));
        this._on(Main.sessionMode, 'updated', () => {
            journal('[WorkspaceIndicator] session mode updated, rebuilding');
            this._queueRender();
        });
        this._on(this._settings, 'changed', () => this._queueRender());

        this._watchWorkspaces();
        this._render();
    }

    _on(obj, signal, cb) { this._signals.push([obj, obj.connect(signal, cb)]); }

    _watchWorkspaces() {
        for (const [obj, id] of this._wsSignals) obj.disconnect(id);
        this._wsSignals = [];
        for (let i = 0; i < WorkspaceManager.n_workspaces; i++) {
            const ws = WorkspaceManager.get_workspace_by_index(i);
            this._wsSignals.push([ws, ws.connect('window-added', () => this._queueRender(TimeoutDelay))]);
            this._wsSignals.push([ws, ws.connect('window-removed', () => this._queueRender())]);
        }
    }

    // ---------- scheduling ----------
    _queueRender(delay = 0) {
        if (this._renderId) {
            if (delay >= this._renderDelay) return;
            GLib.Source.remove(this._renderId);
        }
        this._renderDelay = delay;
        const cb = () => { this._renderId = 0; this._render(); return GLib.SOURCE_REMOVE; };
        this._renderId = delay
            ? GLib.timeout_add(GLib.PRIORITY_DEFAULT, delay, cb)
            : GLib.idle_add(GLib.PRIORITY_DEFAULT, cb);
    }

    // ==================== THE RENDER ====================
    _render() {
        if (intent.menuOpen) return; // re-queued when the menu closes
        const model = buildModel();
        const s = this._settings;

        // The drag source icon must survive the rebuild — Shell DND holds it.
        const keep = model.drag?.sourceIcon ?? null;
        keep?.get_parent()?.remove_child(keep);
        this._mainBox.destroy_all_children();
        this._settleActor = null;

        const iconSize = s.get_int('toolbar-icon-size');
        if (s.get_boolean('show-drun-button'))
            this._mainBox.add_child(this._toolbarButton('view-app-grid-symbolic', iconSize, 'Show apps', () => showAllApps()));
        if (s.get_boolean('show-window-button'))
            this._mainBox.add_child(this._toolbarButton('focus-windows-symbolic', iconSize, 'All windows', () => showAllWindows()));
        if (s.get_boolean('show-workspace-names')) {
            const name = new St.Label({
                style_class: 'workspace-name-label',
                y_expand: true,
                y_align: Clutter.ActorAlign.CENTER,
                text: Meta.prefs_get_workspace_name(WorkspaceManager.get_active_workspace_index()),
            });
            name.set_style(`font-size: ${s.get_int('workspace-name-font-size')}px;`);
            this._mainBox.add_child(name);
        }

        this._thumbsBox = new St.BoxLayout({
            style_class: 'workspace-indicator-class', y_expand: true, x_expand: true, reactive: true,
        });
        for (const entry of model.workspaces)
            this._thumbsBox.add_child(this._buildThumbnail(entry, model, keep));
        this._mainBox.add_child(this._thumbsBox);

        if (keep && !keep.get_parent()) keep.destroy(); // its window left; nothing to keep alive
        if (this._settleActor) settleIcon(this._settleActor);
        this._syncHint(model);
    }

    _toolbarButton(iconName, size, accessibleName, onClicked) {
        const button = new St.Button({
            style_class: 'workspace-toolbar-button',
            accessible_name: accessibleName,
            can_focus: true,
            track_hover: true,
            child: new St.Icon({ icon_name: iconName, icon_size: size }),
        });
        button.connect('clicked', onClicked);
        return button;
    }

    _buildThumbnail(entry, model, keep) {
        const { workspace, index, windows } = entry;
        const s = this._settings;
        const drag = model.drag;

        const thumb = new St.Button({ style_class: 'workspace-thumbnail', x_expand: true, y_expand: true });
        thumb.set_style(`min-width: ${s.get_int('thumbnail-min-width')}px;`);
        if (index === WorkspaceManager.get_active_workspace_index())
            thumb.add_style_class_name('active');
        thumb._wsEntry = { workspace, index };

        const box = new St.BoxLayout();
        thumb.set_child(box);
        const direct = windows.length <= DIRECT_MODE_MAX_WINDOWS;

        const hooks = {
            handleDragOver: (source) => {
                try {
                    journal(`[handleDragOver] ws=${index} source._window=${source?._window?.title ?? 'none'} drag=${!!intent.drag}`);
                    return this._handleDragOver(box, entry, direct, source);
                } catch (e) {
                    journal(`[handleDragOver] THREW: ${e.message}\n${e.stack}`, true);
                    return DragMotionResult.MOVE_DROP;
                }
            },
            acceptDrop: (source, actor) => {
                try {
                    journal(`[acceptDrop] ws=${index} source._window=${source?._window?.title ?? 'none'}`);
                    return this._acceptDrop(thumb, box, entry, direct, source, actor);
                } catch (e) {
                    journal(`[acceptDrop] THREW: ${e.message}\n${e.stack}`, true);
                    setIntent({ drag: null });
                    return false;
                }
            },
        };
        thumb._delegate = hooks;

        thumb.connect('button-press-event', (actor, event) => {
            const b = event.get_button();
            if (b === Clutter.BUTTON_PRIMARY) workspace.activate(0);
            if (b === Clutter.BUTTON_SECONDARY) this._openThumbnailMenu(thumb, workspace);
            return Clutter.EVENT_STOP;
        });

        if (!direct) {
            const overflow = new St.Button({
                style_class: 'workspace-thumbnail-collection-icon', reactive: true, track_hover: true, can_focus: true,
            });
            overflow.set_child(new St.Label({
                style_class: 'collection-icon-label',
                text: `▱ ${windows.length}`,
                y_align: Clutter.ActorAlign.CENTER,
                x_align: Clutter.ActorAlign.CENTER,
            }));
            overflow.connect('clicked', () => showWorkspaceWindows(index));
            box.add_child(overflow);
            return thumb;
        }

        // Direct mode: icons, plus the drag ghost if it targets this workspace.
        const ghostAt = drag?.target?.wsIndex === index ? drag.target.index : null;
        let n = 0, ghostPlaced = false;
        const placeGhost = () => {
            box.add_child(createGhostIcon(getWindowAppIcon(drag.window), s.get_int('icon-size')));
            ghostPlaced = true;
        };

        for (const w of windows) {
            if (w !== drag?.window) {
                if (ghostAt === n) placeGhost();
                n++;
            }
            const actor = (keep && keep._window === w) ? keep : new WindowIcon(w, s).actor;
            actor._icon.setHooks(hooks);
            // Experiment: keep the source in the layout, dimmed, rather than
            // hiding it. The ghost marks the drop target next to it.
            if (actor === keep) actor.opacity = 80;
            box.add_child(actor);
            if (w === _settleWindow) { _settleWindow = null; this._settleActor = actor; }
        }
        if (ghostAt !== null && !ghostPlaced) placeGhost();
        return thumb;
    }

    _syncHint(model) {
        const s = this._settings;
        const target = model.hintWs !== null ? this._thumbsBox?.get_child_at_index(model.hintWs) : null;
        if (!target || !s.get_boolean('show-workspace-names')) {
            this._nameHint?.hide();
            return;
        }
        if (!this._nameHint)
            this._nameHint = new FloatingTooltip({ fontSize: s.get_int('tooltip-font-size') });
        this._nameHint.showAbove(target, Meta.prefs_get_workspace_name(model.hintWs));
    }

    _openThumbnailMenu(anchor, workspace) {
        const windows = Display.get_tab_list(Meta.TabList.NORMAL, workspace);
        openContextMenu(anchor, menu => {
            menu.addAction('Close all windows on all workspaces', () => {
                const now = global.get_current_time();
                for (const w of Display.get_tab_list(Meta.TabList.NORMAL, null)) w.delete(now);
            });
            if (windows.length > 0) {
                menu.addAction(`Close all windows except workspace ${workspace.index()}`, () => {
                    const now = global.get_current_time();
                    for (const w of Display.get_tab_list(Meta.TabList.NORMAL, null).filter(w => w.get_workspace() !== workspace))
                        w.delete(now);
                });
                menu.addAction(`Close all windows on workspace ${workspace.index()}`, () => {
                    const now = global.get_current_time();
                    for (const w of windows) w.delete(now);
                });
            }
        });
    }

    // ==================== DnD ====================
    // The thumbnail is the sole classifier. `source` is the draggable's
    // _delegate (NOT the actor):
    //   source._window set   → reorder / move a window (sets drag.target)
    //   source._window unset → external drag (files, text): switch workspace on hover
    _handleDragOver(box, entry, direct, source) {
        this._ensureLeavePoll();
        setIntent({ hintWs: entry.index });

        if (!source?._window) {
            if (entry.workspace !== WorkspaceManager.get_active_workspace())
                entry.workspace.activate(global.get_current_time());
            return DragMotionResult.MOVE_DROP;
        }

        if (direct) {
            const [pointerX] = global.get_pointer();
            setDragTarget({ wsIndex: entry.index, index: this._insertionIndex(box, pointerX) });
        } else {
            setDragTarget(null); // collection mode isn't a reorder target
        }
        return DragMotionResult.MOVE_DROP;
    }

    _acceptDrop(thumb, box, entry, direct, source, actor) {
        setIntent({ hintWs: null });
        const win = source?._window ?? null;
        if (!win) return true; // external: workspace already switched on hover

        const [pointerX] = global.get_pointer();
        const index = direct ? this._insertionIndex(box, pointerX) : null; // null → end
        const sameWs = win.get_workspace() === entry.workspace;

        // Synthetic drag actor (not the real source icon): ours to discard.
        if (actor && actor !== source._actor) {
            actor.get_parent()?.remove_child(actor);
            try { actor.destroy(); } catch (e) { /* ignore */ }
        }

        if (sameWs && !direct) { setIntent({ drag: null }); return true; }

        placeWindow(win, entry.workspace, index);
        if (!sameWs) {
            const monitorIndex = Main.layoutManager.findIndexForActor(thumb);
            if (monitorIndex !== win.get_monitor()) win.move_to_monitor(monitorIndex);
            win.change_workspace(entry.workspace);
        }
        _settleWindow = win;
        setIntent({ drag: null });
        return true;
    }

    // Insertion index among the visible icons (excluding the dragged one),
    // measured in "ghost-free" coordinates: icons after the ghost are
    // shifted back by the ghost's slot width so the ghost can't make the
    // result flip-flop.
    _insertionIndex(box, pointerX) {
        const kids = box.get_children();
        const ghostPos = kids.findIndex(k => k._isGhost);
        let shift = 0;
        if (ghostPos >= 0) {
            const next = kids.slice(ghostPos + 1).find(k => k.visible);
            if (next)
                shift = next.get_transformed_position()[0] - kids[ghostPos].get_transformed_position()[0];
        }
        const dragged = intent.drag?.window;
        const rects = [];
        kids.forEach((k, i) => {
            if (!k._window || !k.visible || k._window === dragged) return;
            let [x] = k.get_transformed_position();
            const [w] = k.get_transformed_size();
            if (ghostPos >= 0 && i > ghostPos) x -= shift;
            rects.push({ item: k, x, width: w, mid: x + w / 2, right: x + w });
        });
        return computeInsertion(rects, pointerX).index;
    }

    // The Shell has no per-actor "drag left" signal and grabs suppress
    // enter/leave, so poll while a drag hovers us: when the button is
    // released or the pointer leaves every thumbnail, clear hint + ghost.
    _ensureLeavePoll() {
        if (this._leavePollId) return;
        this._leavePollId = GLib.timeout_add(GLib.PRIORITY_DEFAULT, 100, () => {
            const [px, py, mods] = global.get_pointer();
            const held = (mods & Clutter.ModifierType.BUTTON1_MASK) !== 0;
            if (!held || !this._thumbnailAt(px, py)) {
                this._leavePollId = 0;
                setIntent({ hintWs: null });
                setDragTarget(null);
                return GLib.SOURCE_REMOVE;
            }
            return GLib.SOURCE_CONTINUE;
        });
    }

    _thumbnailAt(x, y) {
        if (!this._thumbsBox) return null;
        for (const thumb of this._thumbsBox.get_children()) {
            if (!thumb.get_stage() || !thumb._wsEntry) continue;
            const [tx, ty] = thumb.get_transformed_position();
            const [tw, th] = thumb.get_transformed_size();
            if (x >= tx && x <= tx + tw && y >= ty && y <= ty + th)
                return { ...thumb._wsEntry, actor: thumb };
        }
        return null;
    }

    // ==================== LIFECYCLE ====================
    cleanupSources() {
        if (this._renderId) { GLib.Source.remove(this._renderId); this._renderId = 0; }
        if (this._leavePollId) { GLib.Source.remove(this._leavePollId); this._leavePollId = 0; }
    }

    destroy() {
        this.cleanupSources();
        onIntentChange(null);
        resetIntent();
        _instance = null;

        for (const [obj, id] of this._signals) obj.disconnect(id);
        for (const [obj, id] of this._wsSignals) obj.disconnect(id);
        this._signals = [];
        this._wsSignals = [];

        this._nameHint?.destroy();
        this._nameHint = null;
        this._mainBox?.destroy();
        Main.panel.set_offscreen_redirect(Clutter.OffscreenRedirect.ALWAYS);
        super.destroy();
    }
}

// ==================== FUZZY MATCHING ====================
// Token-based: every whitespace-separated token must appear (substring,
// falling back to fuzzy subsequence) somewhere in the text.
function subsequenceMatch(query, text) {
    let qi = 0, score = 0, consecutive = 0;
    for (let ti = 0; ti < text.length && qi < query.length; ti++) {
        if (text[ti] === query[qi]) { score += 1 + consecutive; consecutive++; qi++; }
        else consecutive = 0;
    }
    return qi < query.length ? { matched: false, score: -1 } : { matched: true, score };
}

function fuzzyMatch(query, text) {
    const trimmed = query.trim();
    if (!trimmed) return { matched: true, score: 0 };
    const lowerText = text.toLowerCase();
    let score = 0;
    for (const token of trimmed.toLowerCase().split(/\s+/).filter(t => t.length > 0)) {
        const idx = lowerText.indexOf(token);
        if (idx !== -1) { score += 50 - Math.min(idx, 40); continue; }
        const sub = subsequenceMatch(token, lowerText);
        if (!sub.matched) return { matched: false, score: -1 };
        score += sub.score;
    }
    return { matched: true, score };
}

    // -------------------- Lifecycle --------------------
    _open() {
        Main.layoutManager.addChrome(this._container, { affectsInputRegion: true, trackFullscreen: true });
        Main.layoutManager.uiGroup.set_child_above_sibling(this._container, null);
        this._modalGrab = Main.pushModal(this._container, { actionMode: Shell.ActionMode.NORMAL });
        this._entry.grab_key_focus();
        SearchOverlayBase._current = this;

        // Mutter's workspace-activate focus restoration can fire AFTER our
        // grab and steal key focus back to a window; reclaim it whenever
        // the focused window changes while we're open.
        this._focusWindowId = global.display.connect('notify::focus-window', () => {
            if (this._closed) return;
            this._entry.grab_key_focus();
            this._onFocusWindowChanged();
        });
    }

    _close() {
        if (this._closed) return;
        this._closed = true;
        this._clearPreview();
        this._onBeforeClose();

        if (this._focusWindowId) { global.display.disconnect(this._focusWindowId); this._focusWindowId = null; }
        if (this._modalGrab) { Main.popModal(this._modalGrab); this._modalGrab = null; }
        if (this._entry?.clutter_text) {
            this._entry.clutter_text.disconnect(this._entryChangedId);
            this._entry.clutter_text.disconnect(this._entryKeyPressId);
        }
        Main.layoutManager.removeChrome(this._container);
        this._container.destroy();
    }

    // -------------------- Search / results --------------------
    _onSearchChanged() {
        if (this._closed) return;
        const query = this._entry.get_text();
        const all = this._getAllResultsSorted();
        if (!query.trim()) { this._setResults(all); return; }
        const scored = [];
        for (const item of all) {
            const result = fuzzyMatch(query, this._getSearchLabel(item));
            if (result.matched) scored.push({ ...item, score: result.score });
        }
        scored.sort((a, b) => b.score - a.score);
        this._setResults(scored);
    }

    _setResults(results) {
        const previousKey = this._selectedIndex >= 0
            ? this._getRestoreKey(this._results[this._selectedIndex])
            : null;

        this._results = results;
        this._resultsBox.destroy_all_children();
        this._resultButtons = [];
        this._selectedIndex = -1;

        results.forEach((item, index) => {
            const button = this._buildResultRow(item, index);
            button.connect('clicked', () => this._activateResult(index));
            button.connect('notify::hover', () => { if (button.hover) this._selectIndex(index); });
            this._resultsBox.add_child(button);
            this._resultButtons.push(button);
        });

        if (results.length === 0) { this._clearPreview(); return; }

        const restoredIndex = previousKey !== null
            ? results.findIndex(item => this._getRestoreKey(item) === previousKey)
            : -1;
        this._selectIndex(restoredIndex >= 0 ? restoredIndex : 0);
    }

    _selectIndex(index) {
        if (this._closed) return;
        if (index < 0 || index >= this._results.length) return;
        if (this._selectedIndex >= 0 && this._resultButtons[this._selectedIndex])
            this._resultButtons[this._selectedIndex].remove_style_class_name('selected');
        this._selectedIndex = index;
        this._resultButtons[index]?.add_style_class_name('selected');
        this._updatePreview(this._results[index]);
    }

    _updatePreview(item) { this._clearPreview(); this._updatePreviewContent(item); }

    _clearPreview() {
        if (!this._previewContent) return;
        if (this._previewContent.get_parent() === this._previewBox)
            this._previewBox.remove_child(this._previewContent);
        this._previewContent.destroy();
        this._previewContent = null;
    }

    // -------------------- Keyboard --------------------
    _onKeyPress(event) {
        if (this._closed) return Clutter.EVENT_PROPAGATE;
        const symbol = event.get_key_symbol();
        const handled = this._handleExtraKeys(event, symbol);
        if (handled !== undefined) return handled;

        switch (symbol) {
            case Clutter.KEY_Escape:
                this._close();
                return Clutter.EVENT_STOP;
            case Clutter.KEY_Down:
                if (this._selectedIndex < this._results.length - 1) this._selectIndex(this._selectedIndex + 1);
                return Clutter.EVENT_STOP;
            case Clutter.KEY_Up:
                if (this._selectedIndex > 0) this._selectIndex(this._selectedIndex - 1);
                return Clutter.EVENT_STOP;
            case Clutter.KEY_Return:
            case Clutter.KEY_KP_Enter:
                if (this._selectedIndex >= 0) this._activateResult(this._selectedIndex);
                return Clutter.EVENT_STOP;
            default:
                return Clutter.EVENT_PROPAGATE;
        }
    }
}

// ==================== WINDOW SEARCH OVERLAY ====================
const NEMO_WM_CLASS = 'Nemo';
const FILE_PROGRESS_INSTANCE = 'file_progress';

export class WindowSearchOverlay extends SearchOverlayBase {
    constructor(windows, settings, focusedWindow = null) {
        super(settings);
        journal(`[WindowSearchOverlay] Opening with ${windows.length} windows`);
        this._windows = windows;
        this._focusedWindow = focusedWindow ?? global.display.focus_window;
        this._buildUI();
        this._setResults(this._getAllResultsSorted());
        this._open();
    }

    _getHintText() { return 'Search windows…'; }
    _getSearchLabel(item) { return `${item.title} — ${item.appName}`; }

    _getAppName(window) {
        const app = WindowTracker.get_window_app(window);
        return app ? app.get_name() : (window.get_wm_class() || 'Unknown');
    }

    _getAllResultsSorted() {
        const deduped = this._dedupeNemoWindows(this._windows.filter(w => w && !w.skip_taskbar));
        const items = deduped.map(w => ({
            window: w,
            title: w.get_title() || 'Untitled Window',
            appName: this._getAppName(w),
            icon: getWindowAppIcon(w),
        }));

        items.sort((a, b) => a.appName.localeCompare(b.appName) || a.title.localeCompare(b.title));

        // Move-to-front for the focused window.
        if (this._focusedWindow) {
            const i = items.findIndex(item => item.window === this._focusedWindow);
            if (i > 0) items.unshift(...items.splice(i, 1));
        }
        return items;
    }

    _dedupeNemoWindows(windows) {
        const keptByTitle = new Map();
        const result = [];

        for (const w of windows) {
            const isNemo = w.get_wm_class() === NEMO_WM_CLASS;
            const isProgress = w.get_wm_class_instance() === FILE_PROGRESS_INSTANCE;
            if (!isNemo || isProgress) { result.push(w); continue; }

            const key = w.get_title();
            const existing = keptByTitle.get(key);
            if (!existing) { keptByTitle.set(key, w); result.push(w); continue; }

            if (w.get_user_time() > existing.get_user_time()) {
                const idx = result.indexOf(existing);
                if (idx !== -1) result.splice(idx, 1);
                journal(`[WindowSearchOverlay] Closing duplicate Nemo window: "${existing.get_title()}"`);
                existing.delete(0);
                keptByTitle.set(key, w);
                result.push(w);
            } else {
                journal(`[WindowSearchOverlay] Closing duplicate Nemo window: "${w.get_title()}"`);
                w.delete(0);
            }
        }
        return result;
    }

    _buildUI() {
        const layout = this._computeLayout(Main.layoutManager.primaryMonitor);
        this._buildContainer(layout.monitor);
        this._buildEntry(layout);
        this._buildResultsAndPreview(layout);
        this._container.add_child(this._entry);
        this._container.add_child(this._resultsScroll);
        this._container.add_child(this._previewBox);
    }

    _close() {
        if (this._closed) return;
        journal(`[WindowSearchOverlay] Closing`);
        super._close();
    }

    _buildResultRow(item, index) {
        const button = new St.Button({
            style_class: 'collection-result-item',
            x_expand: true, x_align: Clutter.ActorAlign.START, track_hover: true, reactive: true,
        });
        button.set_style(`font-size: ${this._settings.get_int('collection-result-font-size')}pt;`);
        if (item.window === this._focusedWindow)
            button.add_style_class_name('collection-result-item-focused');

        const row = new St.BoxLayout({ vertical: false, x_expand: true });
        if (item.icon) {
            row.add_child(new St.Icon({
                gicon: item.icon,
                style_class: 'collection-result-icon',
                icon_size: this._settings.get_int('collection-result-icon-size'),
                y_align: Clutter.ActorAlign.CENTER,
            }));
        }
        row.add_child(new St.Label({
            text: `${item.title}  —  ${item.appName}`,
            x_expand: true, y_align: Clutter.ActorAlign.CENTER,
        }));
        button.set_child(row);

        // Same drag path as indicator icons: the indicator sees drag.window
        // and its thumbnails accept the drop.
        const draggable = makeWindowDraggable(button, item.window, this._settings.get_int('icon-size'));
        button._draggable = draggable;
        draggable.connect('drag-begin', () => this._onResultDragBegin(button));
        return button;
    }

    _updateFocusedHighlight() {
        this._results.forEach((item, index) => {
            const button = this._resultButtons[index];
            if (!button) return;
            if (item.window === this._focusedWindow) button.add_style_class_name('collection-result-item-focused');
            else button.remove_style_class_name('collection-result-item-focused');
        });
    }

    // Only adopt the live focus-window if it's one of OUR windows —
    // transient nulls / windows on other workspaces would otherwise
    // wrongly clear a correct initial highlight.
    _onFocusWindowChanged() {
        if (this._closed) return;
        const liveFocus = global.display.focus_window;
        if (!liveFocus) return;
        if (!this._results.some(item => item.window === liveFocus)) return;
        this._focusedWindow = liveFocus;
        this._updateFocusedHighlight();
    }

    _updatePreviewContent(item) {
        const window = item.window;
        const windowFrame = window.get_frame_rect();
        if (windowFrame.height === 0) return;

        const aspect = windowFrame.width / windowFrame.height;
        let targetHeight = this._previewBox.height;
        let targetWidth = targetHeight * aspect;
        if (targetWidth > this._previewBox.width) {
            targetWidth = this._previewBox.width;
            targetHeight = targetWidth / aspect;
        }

        const built = createClonePreviewActor(window, targetHeight, {
            onClose: win => this._closeWindowFromPreview(win),
            closeButtonSize: this._settings.get_int('close-button-size'),
        });
        if (!built) return;

        built.actor.set_position(
            Math.max(0, (this._previewBox.width - built.width) / 2),
            Math.max(0, (this._previewBox.height - built.height) / 2));
        this._previewBox.add_child(built.actor);
        this._previewContent = built.actor;
    }

    _closeWindowFromPreview(window) {
        if (this._closed) return;
        journal(`[WindowSearchOverlay] Closing window from preview: ${window.title}`);
        window.delete(global.get_current_time());

        const closedIndex = this._results.findIndex(item => item.window === window);
        if (closedIndex === -1) return;

        this._results.splice(closedIndex, 1);
        this._windows = this._windows.filter(w => w !== window);

        const button = this._resultButtons[closedIndex];
        if (button) {
            if (button.get_parent() === this._resultsBox) this._resultsBox.remove_child(button);
            button.destroy();
        }
        this._resultButtons.splice(closedIndex, 1);

        if (this._results.length === 0) {
            this._selectedIndex = -1;
            this._clearPreview();
            return;
        }
        this._selectedIndex = -1;
        this._selectIndex(Math.min(closedIndex, this._results.length - 1));
    }

    _activateResult(index) {
        if (this._closed) return;
        const item = this._results[index];
        if (!item) return;
        const window = item.window;
        journal(`[WindowSearchOverlay] Activating: ${window.title}`);
        if (window.minimized) window.unminimize();
        window.get_workspace().activate_with_focus(window, global.get_current_time());
        this._close();
    }

    // -------------------- DnD lifecycle --------------------
    // Shell DND installs its own stage grab during drag-begin and releases
    // it at drag-end; popModal / destroy() synchronously from inside those
    // signals can leave the grab stuck (frozen cursor). So the overlay
    // hides itself on an idle after drag-begin and destroys its actor tree
    // on an idle after drag-end.
    _onResultDragBegin(button) {
        if (this._closed) return;
        this._closed = true;
        journal(`[WindowSearchOverlay] Drag started on result — closing overlay, drag continues`);

        const draggable = button._draggable;
        const destroyLater = () => GLib.idle_add(GLib.PRIORITY_DEFAULT, () => {
            this._destroyContainerAfterDrag();
            return GLib.SOURCE_REMOVE;
        });
        if (draggable) {
            const endId = draggable.connect('drag-end', () => {
                draggable.disconnect(endId);
                destroyLater();
            });
        } else {
            destroyLater();
        }

        GLib.idle_add(GLib.PRIORITY_DEFAULT, () => {
            this._teardownVisibleOverlay();
            return GLib.SOURCE_REMOVE;
        });
    }

    _teardownVisibleOverlay() {
        if (!this._container) return;
        this._clearPreview();
        if (this._modalGrab) { Main.popModal(this._modalGrab); this._modalGrab = null; }
        if (this._focusWindowId) { global.display.disconnect(this._focusWindowId); this._focusWindowId = null; }
        if (this._entry?.clutter_text) {
            if (this._entryChangedId) { this._entry.clutter_text.disconnect(this._entryChangedId); this._entryChangedId = null; }
            if (this._entryKeyPressId) { this._entry.clutter_text.disconnect(this._entryKeyPressId); this._entryKeyPressId = null; }
        }
        Main.layoutManager.removeChrome(this._container);
        this._container.hide();
    }

    _destroyContainerAfterDrag() {
        if (this._modalGrab) { Main.popModal(this._modalGrab); this._modalGrab = null; }
        if (this._container) { this._container.destroy(); this._container = null; }
    }
}

// ==================== APP SEARCH OVERLAY ====================
const PINNED_ICON_SIZE = 96;
const PINNED_BAR_HEIGHT = 120;

export class AppSearchOverlay extends SearchOverlayBase {
    constructor(settings) {
        super(settings);
        this._favorites = AppFavorites.getAppFavorites();
        this._menuManager = null;
        this._openMenu = null;

        // Pinned-bar drag state (the overlay is its own tiny surface; this
        // ghost is local to it and unrelated to the workspace indicator).
        this._pinnedGhost = null;
        this._pinnedGhostIndex = null;
        this._pinnedSnap = null;
        this._pinnedDragHint = null;
        this._pinnedDragActive = false;
        this._justDroppedAppId = null;

        // The alphabetical list is stable across keystrokes; only
        // installed-changed / favorites-changed invalidate it.
        this._allItemsCache = null;

        this._buildUI();
        this._setResults(this._getAllResultsSorted());
        this._open();
    }

    _getHintText() { return 'Search apps… (Ctrl+P: pin/unpin)'; }
    _getSearchLabel(item) { return item.name ?? ''; }
    _getRestoreKey(item) { return item.app.get_id(); }

    _handleExtraKeys(event, symbol) {
        if (symbol === Clutter.KEY_p && (event.get_state() & Clutter.ModifierType.CONTROL_MASK)) {
            if (this._selectedIndex >= 0) this._togglePin(this._results[this._selectedIndex]);
            return Clutter.EVENT_STOP;
        }
        return undefined;
    }

    // -------------------- App enumeration --------------------
    _getAllApps() {
        const appSystem = Shell.AppSystem.get_default();
        let appInfos = [];

        if (appSystem && typeof appSystem.get_all === 'function') {
            try {
                appInfos = appSystem.get_all().filter(app => {
                    try { return app.should_show(); } catch { return false; }
                });
            } catch (e) {
                journal(`[AppSearchOverlay] Shell.AppSystem.get_all() failed: ${e.message}`);
            }
        }

        if (appInfos.length === 0) {
            try {
                const gioApps = Gio.AppInfo.get_all().filter(app => {
                    try { return app.should_show(); } catch { return false; }
                });
                for (const info of gioApps) {
                    let shellApp = null;
                    if (appSystem && typeof appSystem.lookup_app === 'function') {
                        try { shellApp = appSystem.lookup_app(info.get_id()); } catch (e) { }
                    }
                    appInfos.push(shellApp || info);
                }
            } catch (e) {
                journal(`[AppSearchOverlay] Gio.AppInfo fallback failed: ${e.message}`);
                return [];
            }
        }
        return appInfos;
    }

    // Cached. Callers must treat the returned array as read-only.
    _getAllResultsSorted() {
        if (this._allItemsCache) return this._allItemsCache;

        try {
            const apps = this._getAllApps();
            if (!apps || apps.length === 0) {
                journal('[AppSearchOverlay] No apps found');
                return (this._allItemsCache = []);
            }

            const favoriteIds = new Set();
            for (const fav of this._favorites.getFavorites()) {
                try { const id = fav.get_id(); if (id) favoriteIds.add(id); }
                catch (e) { journal(`[AppSearchOverlay] Fav error: ${e.message}`); }
            }

            const items = [];
            for (const app of apps) {
                try {
                    const id = typeof app.get_id === 'function' ? app.get_id() : String(app);
                    if (!id) continue;
                    const name = typeof app.get_name === 'function' ? (app.get_name() || '') : (String(app) || '');
                    const icon = typeof app.get_icon === 'function' ? (app.get_icon() || null) : null;
                    items.push({ app, name, icon, pinned: favoriteIds.has(id) });
                } catch (e) {
                    journal(`[AppSearchOverlay] Skipping app: ${e.message}`);
                }
            }
            items.sort((a, b) => a.name.localeCompare(b.name));
            return (this._allItemsCache = items);
        } catch (e) {
            journal(`[AppSearchOverlay] CRITICAL ERROR: ${e.message}\n${e.stack}`);
            return (this._allItemsCache = []);
        }
    }

    _launchApp(app) {
        const name = typeof app.get_name === 'function' ? app.get_name() : 'unknown';
        try {
            app.launch(global.get_current_time(), -1, 0);
            journal(`[AppSearchOverlay] ✅ Launched successfully: ${name}`);
            return true;
        } catch (e) {
            journal(`[AppSearchOverlay] ❌ Launch failed for ${name}: ${e.message}`, true);
            return false;
        }
    }

    // -------------------- UI --------------------
    _buildUI() {
        const layout = this._computeLayout(Main.layoutManager.primaryMonitor);
        const { monitor, margin, panelWidth, panelX, entryY } = layout;

        this._buildContainer(monitor);
        this._buildEntry(layout);
        this._buildResultsAndPreview(layout, { previewVertical: true });

        this._pinnedBarBox = new St.BoxLayout({
            style_class: 'dmenu-pinned-bar',
            vertical: false, x_expand: true, y_expand: false,
            height: PINNED_BAR_HEIGHT,
            x_align: Clutter.ActorAlign.START, y_align: Clutter.ActorAlign.CENTER,
            reactive: true,
        });
        this._pinnedBarBox._delegate = this._pinnedBarBox;
        this._pinnedBarBox.handleDragOver = source => this._handlePinnedDragOver(source);
        this._pinnedBarBox.acceptDrop = (source, actor) => this._acceptPinnedDrop(source, actor);
        this._pinnedBarBox.hide();

        const mainVertical = new St.BoxLayout({ vertical: true, x_expand: true, y_expand: true });

        const pinnedWrapper = new St.BoxLayout({ vertical: false, x_expand: true, y_expand: false, style: 'padding: 0 0 8px 0;' });
        pinnedWrapper.add_child(this._pinnedBarBox);

        const entryWrapper = new St.BoxLayout({ vertical: false, x_expand: true, y_expand: false });
        entryWrapper.add_child(this._entry);

        const resultsPreviewRow = new St.BoxLayout({ vertical: false, x_expand: true, y_expand: true });
        resultsPreviewRow.add_child(this._resultsScroll);
        resultsPreviewRow.add_child(this._previewBox);

        mainVertical.add_child(pinnedWrapper);
        mainVertical.add_child(entryWrapper);
        mainVertical.add_child(resultsPreviewRow);
        mainVertical.set_position(panelX, entryY);
        mainVertical.set_size(panelWidth, monitor.height - margin * 2);
        this._container.add_child(mainVertical);

        this._favoritesChangedId = this._favorites.connect('changed', () => this._onFavoritesChanged());
        this._installedChangedId = Shell.AppSystem.get_default().connect('installed-changed', () => {
            this._allItemsCache = null;
        });

        this._renderPinnedBar();
    }

    // -------------------- Pinned bar --------------------
    _renderPinnedBar() {
        this._pinnedBarBox.destroy_all_children();
        const favorites = this._favorites.getFavorites();

        if (!favorites || favorites.length === 0) { this._pinnedBarBox.hide(); return; }
        this._pinnedBarBox.show();

        for (const app of favorites) {
            const icon = app.get_icon();
            if (!icon) continue;

            const id = app.get_id();
            const name = app.get_name?.() || '';

            const button = new St.Button({
                style_class: 'dmenu-pinned-icon',
                child: new St.Icon({ gicon: icon, icon_size: PINNED_ICON_SIZE }),
                reactive: true, can_focus: true, track_hover: true,
            });
            button._appId = id;
            button._appName = name;
            button._gicon = icon;

            const tooltip = new FloatingTooltip({
                parent: this._container,
                fontSize: this._settings.get_int('tooltip-font-size'),
            });
            button._tooltip = tooltip;

            button.connect('notify::hover', () => {
                if (this._pinnedDragActive || !button.hover) tooltip.hide();
                else tooltip.showBelow(button, name);
            });
            button.connect('destroy', () => tooltip.destroy());
            button.connect('clicked', () => { this._launchApp(app); this._close(); });
            button.connect('button-press-event', (actor, event) => {
                if (event.get_button() === Clutter.BUTTON_SECONDARY) {
                    this._openContextMenuForApp(app, button);
                    return Clutter.EVENT_STOP;
                }
                return Clutter.EVENT_PROPAGATE;
            });

            const draggable = createReorderDraggable(button, PINNED_ICON_SIZE);
            draggable.connect('drag-begin', () => {
                this._pinnedDragActive = true;
                button.add_style_class_name('reorder-drag-source');
                this._hideAllPinnedTooltips();
            });
            draggable.connect('drag-end', () => {
                this._pinnedDragActive = false;
                try { button.remove_style_class_name('reorder-drag-source'); }
                catch (e) { /* button may already be destroyed after a drop */ }
                this._clearPinnedGhost();
                this._pinnedDragHint?.hide();
            });
            button._draggable = draggable;

            button._delegate = button;
            button.handleDragOver = source => source._appId ? this._handlePinnedDragOver(source) : DragMotionResult.NO_DROP;
            button.acceptDrop = (source, actor) => source._appId ? this._acceptPinnedDrop(source, actor) : false;

            this._pinnedBarBox.add_child(button);

            if (id === this._justDroppedAppId) {
                this._justDroppedAppId = null;
                settleIcon(button);
            }
        }
    }

    _hideAllPinnedTooltips() {
        for (const b of this._pinnedBarBox.get_children()) b._tooltip?.hide();
    }

    _showPinnedGhost(index, source) {
        const box = this._pinnedBarBox;
        const clamped = Math.max(0, Math.min(index, box.get_children().length));
        if (this._pinnedGhost && this._pinnedGhostIndex === clamped) return;
        this._pinnedGhostIndex = clamped;
        if (!this._pinnedGhost)
            this._pinnedGhost = createGhostIcon(source._gicon, PINNED_ICON_SIZE);
        this._pinnedGhost.get_parent()?.remove_child(this._pinnedGhost);
        box.insert_child_at_index(this._pinnedGhost, clamped);
    }

    _clearPinnedGhost() {
        if (this._pinnedGhost) {
            this._pinnedGhost.get_parent()?.remove_child(this._pinnedGhost);
            this._pinnedGhost.destroy();
        }
        this._pinnedGhost = null;
        this._pinnedGhostIndex = null;
        this._pinnedSnap = null;
    }

    _handlePinnedDragOver(source) {
        if (!source._appId) return DragMotionResult.NO_DROP;

        const [pointerX] = global.get_pointer();

        // Snapshot once per drag so the ghost can't shift what we measure.
        if (!this._pinnedSnap) {
            this._pinnedSnap = this._pinnedBarBox.get_children()
                .filter(b => b._appId && b !== source && b.get_stage())
                .map(b => {
                    const [x] = b.get_transformed_position();
                    const [width] = b.get_transformed_size();
                    return { item: b, x, width, mid: x + width / 2, right: x + width };
                });
        }
        const insertion = computeInsertion(this._pinnedSnap, pointerX);
        this._showPinnedGhost(insertion.index, source);

        if (!this._pinnedDragHint)
            this._pinnedDragHint = new FloatingTooltip({
                parent: this._container,
                fontSize: this._settings.get_int('tooltip-font-size'),
            });

        let text = 'Pin here';
        if (insertion.neighbor) {
            const name = insertion.neighbor._appName ?? '';
            text = insertion.insertBefore ? `Move before ${name}` : `Move after ${name}`;
        }
        this._pinnedDragHint.showAbove(insertion.neighbor ?? this._pinnedBarBox, text);
        return DragMotionResult.MOVE_DROP;

        } catch (e) {
            journal(`[AppSearchOverlay] Reorder failed: ${e.message}`);

            actor.get_parent()?.remove_child(actor);

    // indexed matcher), each equal-score group sorted by Shell.AppUsage.
    // No result cap. Empty query → full alphabetical list.
    _onSearchChanged() {
        if (this._closed) return;
        const query = this._entry.get_text().trim();
        const all = this._getAllResultsSorted();
        this._setResults(query ? this._searchApps(query, all) : all);
    }

    _searchApps(query, allItems) {
        const appSystem = Shell.AppSystem.get_default();
        const usage = Shell.AppUsage.get_default();

        let groups;
        try {
            if (typeof Shell.AppSystem.search === 'function') groups = Shell.AppSystem.search(query);
            else if (appSystem && typeof appSystem.search === 'function') groups = appSystem.search(query);
            else { journal('[AppSearchOverlay] Shell.AppSystem.search is unavailable'); return []; }
        } catch (e) {
            journal(`[AppSearchOverlay] Shell.AppSystem.search() threw: ${e.message}`);
            return [];
        }
        if (!groups || groups.length === 0) return [];

        // Index by Shell.App identity (lookup_app returns canonical instances).
        const itemByApp = new Map();
        for (const item of allItems)
            if (item.app) itemByApp.set(item.app, item);

        const results = [];
        for (const group of groups) {
            if (!group || group.length === 0) continue;

            const resolved = [];
            for (const id of group) {
                let app;
                try { app = appSystem.lookup_app(id); } catch (e) { app = null; }
                if (!app) continue;
                const item = itemByApp.get(app);
                if (item) resolved.push({ id, item });
            }
            if (resolved.length === 0) continue;

            try { resolved.sort((a, b) => usage.compare(a.id, b.id)); }
            catch (e) { /* keep GLib's ordering */ }

            for (const { item } of resolved) results.push(item);
        }
        return results;
    }

    }

    _updatePreviewContent(item) {
        const { app } = item;

            x_align: Clutter.ActorAlign.CENTER,
        }));
        wrapper.add_child(new St.Label({
            text: item.name, style_class: 'clone-title-overlay', x_align: Clutter.ActorAlign.CENTER,
        }));

        const appInfo = app.get_app_info?.();
        const description = appInfo?.get_description?.();
        if (description) {
            const descLabel = new St.Label({
                text: description, style_class: 'app-description-label', x_align: Clutter.ActorAlign.CENTER,

            descLabel.clutter_text.set_line_wrap(true);
            descLabel.clutter_text.set_width(this._previewBox.width - 80);
            wrapper.add_child(descLabel);
        }

        const isFav = this._favorites.isFavorite(app.get_id());
        const pinButton = new St.Button({
            style_class: 'app-preview-pin-button',
            label: isFav ? 'Unpin from favorites (Ctrl+P)' : 'Pin to favorites (Ctrl+P)',
            can_focus: true,

        pinButton.connect('clicked', () => this._togglePin(item));
        wrapper.add_child(pinButton);

        const actions = appInfo?.list_actions?.();
        if (actions && actions.length > 0) {
            const actionsBox = new St.BoxLayout({ vertical: true, x_align: Clutter.ActorAlign.CENTER });
            for (const action of actions) {

                actionsBox.add_child(actionButton);
            }

    }

    }

        this._launchApp(item.app);
        this._close();
    }

import Meta from 'gi://Meta';

import * as Main from 'resource:///org/gnome/shell/ui/main.js';
import { thumbnailAt, settleOnNextRender } from './workspaceIndicator.js';
import { placeWindow, setIntent } from './model.js';
import { WorkspaceManager } from './shellGlobals.js';

import { createLogger } from '../logger.js';
const journal = createLogger(import.meta.url);

// ==================== TITLE BAR MOVE MONITOR ====================
// Listens for native mutter window-move grabs (dragging by the title bar)
// and, if the pointer is over one of our thumbnails, moves the window
// there. Pointer updates come from Meta.CursorTracker's
// `position-invalidated`, which fires on every motion event even during a
// window-move grab.
export class TitleBarMoveMonitor {
    constructor() {
        this._currentDragWindow = null;
        this._lastSwitchedWorkspace = null;
        this._cursorTracker = this._getCursorTracker();
        this._cursorPositionId = 0;

        this._beginId = global.display.connect('grab-op-begin',
            (display, window, op) => this._onGrabOpBegin(window, op));
        this._endId = global.display.connect('grab-op-end',
            (display, window, op) => this._onGrabOpEnd(window, op));
        this._sessionModeId = Main.sessionMode.connect('updated', () => {
            if (Main.sessionMode.currentMode === 'unlock-dialog') {
                journal('[TitleBarMoveMonitor] session locked, resetting drag state');
                this.reset();
            }
        });
    }

    _getCursorTracker() {
        try {
            if (typeof Meta.CursorTracker.get_for_display === 'function')
                return Meta.CursorTracker.get_for_display(global.display);
        } catch (e) { journal(`[TitleBarMoveMonitor] get_for_display failed: ${e}`); }
        try {
            if (global.backend && typeof global.backend.get_cursor_tracker === 'function')
                return global.backend.get_cursor_tracker();
        } catch (e) { journal(`[TitleBarMoveMonitor] get_cursor_tracker failed: ${e}`); }
        return null;
    }

    reset() {
        this._currentDragWindow = null;
        this._lastSwitchedWorkspace = null;
        this._stopGrabTracking();
        setIntent({ hintWs: null });
    }

    _isMoveOp(op) { return op === Meta.GrabOp.MOVING || op === Meta.GrabOp.KEYBOARD_MOVING; }

    _onGrabOpBegin(window, op) {
        if (!this._isMoveOp(op)) return;
        this._currentDragWindow = window;
        this._lastSwitchedWorkspace = null;
        this._startGrabTracking();
    }

    _onGrabOpEnd(window, op) {
        const grabbed = this._currentDragWindow;
        this._currentDragWindow = null;
        this._lastSwitchedWorkspace = null;
        this._stopGrabTracking();
        setIntent({ hintWs: null });

        if (!grabbed || grabbed !== window || !this._isMoveOp(op)) return;

        // Final check on release in case the last signal missed the release point.
        const [x, y] = this._getPointerPosition();
        const target = thumbnailAt(x, y);
        if (target) this._moveWindowTo(target, window);
    }

    _startGrabTracking() {
        this._stopGrabTracking();
        if (!this._cursorTracker) return;
        this._cursorPositionId = this._cursorTracker.connect('position-invalidated',
            () => this._onCursorPositionInvalidated());
    }

    _stopGrabTracking() {
        if (this._cursorPositionId && this._cursorTracker) {
            this._cursorTracker.disconnect(this._cursorPositionId);
            this._cursorPositionId = 0;
        }
    }

    _onCursorPositionInvalidated() {
        if (!this._currentDragWindow) return;

        const [x, y] = this._getPointerPosition();
        const thumb = thumbnailAt(x, y);
        setIntent({ hintWs: thumb ? thumb.index : null });
        if (!thumb) return;

        const ws = thumb.workspace;
        if (ws === WorkspaceManager.get_active_workspace() || ws === this._lastSwitchedWorkspace) return;
        this._moveWindowTo(thumb, this._currentDragWindow);
    }

    _getPointerPosition() {
        if (this._cursorTracker) {
            try {
                const [point] = this._cursorTracker.get_pointer();
                return [point.x, point.y];
            } catch (e) { journal(`[TitleBarMoveMonitor] get_pointer failed: ${e}`); }
        }
        return global.get_pointer();
    }

    // Append to the target's order, move the window, switch there. The
    // indicator re-renders from the resulting workspace events — nothing
    // to patch by hand.
    _moveWindowTo(thumb, window) {
        const ws = thumb.workspace;
        const alreadyThere = window.get_workspace() === ws;

        if (!alreadyThere) {
            placeWindow(window, ws, null);
            settleOnNextRender(window);
        }
        const monitorIndex = Main.layoutManager.findIndexForActor(thumb.actor);
        if (monitorIndex !== window.get_monitor()) window.move_to_monitor(monitorIndex);
        window.change_workspace(ws);
        ws.activate(global.get_current_time());
        this._lastSwitchedWorkspace = ws;
    }

    destroy() {
        this._stopGrabTracking();
        if (this._beginId) { global.display.disconnect(this._beginId); this._beginId = null; }
        if (this._endId) { global.display.disconnect(this._endId); this._endId = null; }
        if (this._sessionModeId) { Main.sessionMode.disconnect(this._sessionModeId); this._sessionModeId = null; }
        this._currentDragWindow = null;
    }
}

// ==================== INSERTION MATH ====================
// rects: [{ item, x, right, mid }] in left-to-right order.
export function computeInsertion(rects, pointerX) {
    if (rects.length === 0)
        return { index: 0, neighbor: null, insertBefore: true };

    const first = rects[0];
    const last = rects[rects.length - 1];
    let target = last, before = false;

    if (pointerX < first.x) {
        target = first; before = true;
    } else if (pointerX < last.right) {
        for (let i = 0; i < rects.length; i++) {
            const r = rects[i], next = rects[i + 1];
            if (pointerX >= r.x && pointerX < r.right) {
                target = r; before = pointerX < r.mid; break;
            }
            if (next && pointerX >= r.right && pointerX <= next.x) {
                before = (pointerX - r.right) > (next.x - pointerX);
                target = before ? next : r;
                break;
            }
        }
    }
    const i = rects.indexOf(target);
    return { index: before ? i : i + 1, neighbor: target.item, insertBefore: before };
}

// Makes `actor` (a window icon OR an overlay result row) a drag source
// for `window`. Dragging only sets a flag on the representation; the
// indicator renders the ghost and hides the source from it.
export function makeWindowDraggable(actor, window, iconSize, { delegate = {}, sourceIcon = false } = {}) {
    actor._window = window;
    // Shell passes actor._delegate as `source` to handleDragOver/acceptDrop,
    // so the identifying fields must live on the delegate itself.
    actor._delegate = Object.assign(delegate, {
        _window: window,
        _actor: actor,
        getDragActorSource: () => actor,
        getDragActor: () => {
            const bin = new St.Bin({ style_class: 'window-preview-icon', width: iconSize, height: iconSize });
            bin.set_child(new St.Icon({ gicon: getWindowAppIcon(window), icon_size: iconSize }));
            bin.add_style_class_name('reorder-drag-source');
            bin.opacity = 178; // CSS opacity isn't honored by St; set it directly
            return bin;
        },
    });
    const draggable = createReorderDraggable(actor, iconSize);
    draggable.connect('drag-begin', () =>
        setIntent({ drag: { window, sourceIcon: sourceIcon ? actor : null, target: null } }));
    draggable.connect('drag-end', () => setIntent({ drag: null, hintWs: null }));
    return draggable;
}

export class FloatingTooltip {
    constructor({ parent = null, styleClass = 'dmenu-tooltip', fontSize = null } = {}) {
        this._parent = parent;
        this._label = new St.Label({ style_class: styleClass, visible: false });
        if (fontSize)
            this._label.set_style(`font-size: ${fontSize}pt;`);
        if (this._parent) this._parent.add_child(this._label);
        else Main.layoutManager.addChrome(this._label);
    }

    showAbove(actor, text, gap = 8) { this._show(actor, text, gap, false); }
    showBelow(actor, text, gap = 8) { this._show(actor, text, gap, true); }

    _show(actor, text, gap, preferBelow) {
        this._label.set_text(text);
        this._label.show();

        const [width, height] = this._label.get_size();
        const [actorX, actorY] = actor.get_transformed_position();
        const actorWidth = actor.get_width();
        const actorHeight = actor.get_height();

        const monitorIndex = Main.layoutManager.findIndexForActor(actor);
        const monitor = Main.layoutManager.monitors[monitorIndex] ?? Main.layoutManager.primaryMonitor;

        const edgePad = 6;
        const minX = monitor.x + edgePad;
        const maxX = monitor.x + monitor.width - width - edgePad;
        const minY = monitor.y + edgePad;
        const maxY = monitor.y + monitor.height - height - edgePad;

        let x = Math.round(actorX + (actorWidth - width) / 2);
        x = Math.max(minX, Math.min(x, maxX));

        const above = Math.round(actorY - height - gap);
        const below = Math.round(actorY + actorHeight + gap);

        let y;
        if (preferBelow) { y = below; if (y > maxY) y = above; }
        else { y = above; if (y < minY) y = below; }
        y = Math.max(minY, Math.min(y, maxY));

        // Absolute stage coords → parent-relative when parented in an overlay.
        if (this._parent) {
            const [parentX, parentY] = this._parent.get_transformed_position();
            x -= parentX;
            y -= parentY;
        }
        this._label.set_position(x, y);
    }

    hide() { this._label.hide(); }

    destroy() {
        if (this._parent) {
            if (this._label.get_parent() === this._parent)
                this._parent.remove_child(this._label);
        } else {
            Main.layoutManager.removeChrome(this._label);
        }
        this._label.destroy();
    }
}

const LAYOUT = { TITLE_HEIGHT_MIN: 40, TITLE_HEIGHT_MAX: 100, TITLE_HEIGHT_FRACTION: 0.2 };

export function createClonePreviewActor(window, targetHeight, options = {}) {
    if (!window) return null;
    const windowActor = window.get_compositor_private();
    if (!windowActor) return null;

    const windowFrame = window.get_frame_rect();
    const bufferFrame = window.get_buffer_rect();
    if (windowFrame.height === 0) return null;

    const targetWidth = targetHeight * (windowFrame.width / windowFrame.height);
    const scale = targetHeight / windowFrame.height;
    const scaledLeftShadow = (windowFrame.x - bufferFrame.x) * scale;
    const scaledTopShadow = (windowFrame.y - bufferFrame.y) * scale;
    const scaledRightShadow = ((bufferFrame.x + bufferFrame.width) - (windowFrame.x + windowFrame.width)) * scale;
    const scaledBottomShadow = ((bufferFrame.y + bufferFrame.height) - (windowFrame.y + windowFrame.height)) * scale;

    const container = new St.BoxLayout({
        style_class: 'collection-preview-inner',
        width: targetWidth,
        height: targetHeight,
        clip_to_allocation: true,
    });

    const clone = new Clutter.Clone({
        source: windowActor,
        width: targetWidth + scaledLeftShadow + scaledRightShadow,
        height: targetHeight + scaledTopShadow + scaledBottomShadow,
    });
    clone.set_position(-scaledLeftShadow, -scaledTopShadow);

    const cloneContainer = new Clutter.Actor();
    cloneContainer.add_child(clone);
    container.add_child(cloneContainer);

    if (options.showTitle !== false) {
        const titleText = window.get_title();
        const label = titleText && titleText.trim() ? titleText : 'Untitled';
        const titleHeight = Math.min(LAYOUT.TITLE_HEIGHT_MAX,
            Math.max(LAYOUT.TITLE_HEIGHT_MIN, targetHeight * LAYOUT.TITLE_HEIGHT_FRACTION));
        const title = new St.Label({
            style_class: 'clone-title-overlay',
            text: label,
            x_align: Clutter.ActorAlign.FILL,
            y_align: Clutter.ActorAlign.CENTER,
        });
        title.clutter_text.set_x_align(Clutter.ActorAlign.CENTER);
        title.clutter_text.set_y_align(Clutter.ActorAlign.CENTER);
        title.clutter_text.set_line_wrap(true);
        title.set_size(targetWidth, titleHeight);
        title.set_position(0, (targetHeight - titleHeight) / 2);
        cloneContainer.add_child(title);
    }

    if (options.onClose) {
        const closeIconSize = options.closeButtonSize ?? 32;
        const closeOffsetX = options.closeButtonOffsetX ?? (closeIconSize + 14);
        const closeOffsetY = options.closeButtonOffsetY ?? 10;
        const closeButton = new St.Button({
            style_class: 'window-close-button',
            child: new St.Icon({ icon_name: 'window-close-symbolic', icon_size: closeIconSize }),
            x_align: Clutter.ActorAlign.END,
            y_align: Clutter.ActorAlign.START,
            reactive: true,
        });
        closeButton.set_position(targetWidth - closeOffsetX, closeOffsetY);
        closeButton.connect('clicked', () => {
            options.onClose(window);
            return Clutter.EVENT_STOP;
        });
        cloneContainer.add_child(closeButton);
    }

    if (!options.wrapperStyleClass)
        return { actor: container, width: targetWidth, height: targetHeight };

    const wrapper = new St.BoxLayout({
        style_class: options.wrapperStyleClass,
        reactive: true,
        track_hover: true,
    });
    wrapper.add_child(container);

    if (options.onHoverChange)
        wrapper.connect('notify::hover', () => options.onHoverChange(wrapper.hover));

    if (options.onActivate) {
        wrapper.connect('button-press-event', (actor, event) => {
            if (event.get_button() === Clutter.BUTTON_PRIMARY) {
                options.onActivate();
                return Clutter.EVENT_STOP;
            }
            return Clutter.EVENT_PROPAGATE;
        });
    }
    return { actor: wrapper, inner: container, width: targetWidth, height: targetHeight };

import { Display, WorkspaceManager } from './shellGlobals.js';

// ==================== INTENT (transient UI flags) ====================
// drag:     { window, sourceIcon, target: {wsIndex, index} | null } | null
// hintWs:   workspace index whose name tooltip is showing
// menuOpen: a context menu is open (renders are deferred)
export const intent = { drag: null, hintWs: null, menuOpen: false };

let _onChange = () => { };
export function onIntentChange(cb) { _onChange = cb ?? (() => { }); }
export function requestRender() { _onChange(); }
export function resetIntent() { intent.drag = null; intent.hintWs = null; intent.menuOpen = false; }

export function setIntent(patch) {
    let dirty = false;
    for (const key of Object.keys(patch)) {
        if (intent[key] !== patch[key]) { intent[key] = patch[key]; dirty = true; }
    }
    if (dirty) _onChange();
}

export function setDragTarget(target) {
    const drag = intent.drag;
    if (!drag) return;
    if (drag.target?.wsIndex === target?.wsIndex && drag.target?.index === target?.index) return;
    intent.drag = { ...drag, target };
    _onChange();
}

// ==================== USER ORDER (the only persistent state) ====================
const ranks = new Map(); // Meta.Window -> number
let nextRank = 1;

function isEligible(w) {
    return !w.skip_taskbar && !w.is_override_redirect();
}

// ==================== THE REPRESENTATION ====================
// Rebuilt from scratch from Mutter's live windows on every call.
export function buildModel() {
    const live = Display.list_all_windows().filter(isEligible);

    const liveSet = new Set(live);
    for (const w of [...ranks.keys()])
        if (!liveSet.has(w)) ranks.delete(w);
    for (const w of live)
        if (!ranks.has(w)) ranks.set(w, nextRank++);

    const workspaces = [];
    for (let i = 0; i < WorkspaceManager.n_workspaces; i++) {
        const workspace = WorkspaceManager.get_workspace_by_index(i);
        const windows = live
            .filter(w => w.located_on_workspace(workspace))
            .sort((a, b) => ranks.get(a) - ranks.get(b));
        workspaces.push({ workspace, index: i, windows });
    }
    return { workspaces, drag: intent.drag, hintWs: intent.hintWs };
}

// Put `window` at `index` (null = end) in `workspace`'s order. This only
// rewrites ranks; the caller moves the window itself. There's no
// suppression, no pending map: the next buildModel() just sees the result.
export function placeWindow(window, workspace, index = null) {
    const list = buildModel().workspaces[workspace.index()].windows.filter(w => w !== window);
    const at = index === null ? list.length : Math.max(0, Math.min(index, list.length));
    list.splice(at, 0, window);
    for (const w of list) ranks.set(w, nextRank++);
}

// ==================== OVERLAY SERVICE ====================
// The single place any search overlay is opened or closed, and the only
// place this extension talks to DBus. Only one overlay exists at a time
// (each pushes a full-screen modal grab), so every show closes whatever
// was open first. WindowSearchOverlay serves both "all windows" and "one
// workspace", so instances are tagged with `_overlayContext` for toggles.

const BUS_NAME = 'io.github.blueray453.TopNotchWorkspaces';
const BUS_PATH = '/io/github/blueray453/TopNotchWorkspaces/Overlay';

// dbus-send --print-reply=literal --session --dest=io.github.blueray453.TopNotchWorkspaces /io/github/blueray453/TopNotchWorkspaces/Overlay io.github.blueray453.TopNotchWorkspaces.Overlay.ShowWorkspaceWindows uint32:3

const MR_DBUS_IFACE = `
<node>
   <interface name="io.github.blueray453.TopNotchWorkspaces.Overlay">
      <method name="ShowAllWindows"/>
      <method name="ShowAllApps"/>
      <method name="ShowWorkspaceWindows">
         <arg type="u" direction="in" name="workspace_num"/>
      </method>
      <method name="ToggleAllWindows"/>
      <method name="ToggleAllApps"/>
      <method name="ToggleWorkspaceWindows">
         <arg type="u" direction="in" name="workspace_num"/>
      </method>
      <method name="CloseOverlay"/>
   </interface>
</node>`;

let _settings = null;
let _ownerId = 0;
let _exported = null;

// -------------------- Show / close --------------------
export function showWindowOverlay(windows, context = null, focusedWindow = null) {
    if (!_settings) {
        journal('[overlayService] showWindowOverlay before init — ignoring', true);
        return false;
    }
    SearchOverlayBase.closeCurrent();
    const overlay = new WindowSearchOverlay(windows, _settings, focusedWindow);
    overlay._overlayContext = context;
    return true;
}

export function showAllWindows() {
    return showWindowOverlay(
        Display.get_tab_list(Meta.TabList.NORMAL, null),
        { type: 'all-windows' },
        Display.focus_window);
}

export function showAllApps() {
    if (!_settings) {
        journal('[overlayService] showAllApps before init — ignoring', true);
        return false;
    }
    SearchOverlayBase.closeCurrent();
    new AppSearchOverlay(_settings);
    return true;
}

export function toggleAllApps() {
    if (SearchOverlayBase.current instanceof AppSearchOverlay) {
        SearchOverlayBase.closeCurrent();
        return false;
    }
    return showAllApps();
}

export function toggleAllWindows() {
    const current = SearchOverlayBase.current;
    if (current instanceof WindowSearchOverlay && current._overlayContext?.type === 'all-windows') {
        SearchOverlayBase.closeCurrent();
        return false;
    }
    return showAllWindows();
}

// One workspace's windows, plus a workspace switch. Window list comes from
// the model (user order); "focused" comes from Mutter's own per-workspace
// MRU list, read BEFORE activating (workspace.activate() doesn't reliably
// refocus on its own).
export function showWorkspaceWindows(workspaceIndex) {
    const workspace = WorkspaceManager.get_workspace_by_index(workspaceIndex);
    if (!workspace)
        throw new Error(`Workspace ${workspaceIndex} not found`);

    const windows = buildModel().workspaces[workspaceIndex].windows;
    const focusedWindow = Display.get_tab_list(Meta.TabList.NORMAL, workspace)[0] ?? null;

    workspace.activate(global.get_current_time());
    const opened = showWindowOverlay(windows, { type: 'workspace', workspaceIndex }, focusedWindow);
    journal(`[overlayService] showWorkspaceWindows ${workspaceIndex} (${windows.length} windows)`);
    return opened;
}

// Only recognizes an overlay opened via showWorkspaceWindows() — one opened
// from the workspace's own overflow button is untagged on purpose... see
// showWorkspaceWindows: both paths now go through it, so both tag.
export function toggleWorkspaceWindows(workspaceIndex) {
    const current = SearchOverlayBase.current;
    if (current instanceof WindowSearchOverlay &&
        current._overlayContext?.type === 'workspace' &&
        current._overlayContext.workspaceIndex === workspaceIndex) {
        SearchOverlayBase.closeCurrent();
        return false;
    }
    return showWorkspaceWindows(workspaceIndex);
}

export function closeOverlay() { return SearchOverlayBase.closeCurrent(); }

// -------------------- DBus --------------------
const dbusObject = {
    CloseOverlay() { closeOverlay(); },
    ShowAllApps() { showAllApps(); },
    ShowAllWindows() { showAllWindows(); },
    ShowWorkspaceWindows(n) { showWorkspaceWindows(n); },
    ToggleAllApps() { toggleAllApps(); },
    ToggleAllWindows() { toggleAllWindows(); },
    ToggleWorkspaceWindows(n) { toggleWorkspaceWindows(n); },
};

function _onBusAcquired(connection) {
    try {
        _exported = Gio.DBusExportedObject.wrapJSObject(MR_DBUS_IFACE, dbusObject);
        _exported.export(connection, BUS_PATH);
        journal(`[overlayService] Exported on ${BUS_PATH}`);
    } catch (e) {
        journal(`[overlayService] Failed to export: ${e.message}`, true);
    }
}

function _unexport() {
    if (!_exported) return;
    try { _exported.flush(); _exported.unexport(); } catch (e) { /* not exported */ }
    _exported = null;
}

// -------------------- Lifecycle --------------------
export function initOverlayService(settings) {
    _settings = settings;
    _ownerId = Gio.bus_own_name(
        Gio.BusType.SESSION, BUS_NAME, Gio.BusNameOwnerFlags.NONE,
        connection => _onBusAcquired(connection),
        (connection, name) => journal(`[overlayService] ${name}: name acquired`),
        (connection, name) => { journal(`[overlayService] ${name}: name lost`, true); _unexport(); });
}

export function destroyOverlayService() {
    closeOverlay(); // never leave a modal grab behind
    _unexport();
    if (_ownerId) { Gio.bus_unown_name(_ownerId); _ownerId = 0; }
    _settings = null;
}

import GLib from 'gi://GLib';
import Gio from 'gi://Gio';

let loggingEnabled = false;
let logOutput = 'journal';
let uuid = null;
let logFile = null;


/*
 * logger.js is in the extension root.
 */
const extensionDir = GLib.path_get_dirname(
    GLib.filename_from_uri(import.meta.url)[0]
);


/*
 * Initialize logging.
 *
 * output:
 *   'journal' -> GNOME Journal only
 *   'file'    -> file only
 *   'both'    -> GNOME Journal + file
 *
 * enabled:
 *   true  -> logging enabled
 *   false -> logging disabled
 */
function initLogging(
    extensionUuid,
    output = 'journal',
    enabled = true
) {
    uuid = extensionUuid;
    logOutput = output;
    loggingEnabled = enabled;

    if (output === 'file' || output === 'both') {
        logFile = GLib.build_filenamev([
            GLib.get_home_dir(),
            `${uuid}.log`,
        ]);
    } else {
        logFile = null;
    }
}


/*
 * Create a logger for a source file.
 *
 * Usage:
 *
 * const journal = createLogger(import.meta.url);
 */
function createLogger(source) {
    let sourceFile;

    try {
        sourceFile = GLib.filename_from_uri(source)[0];
    } catch (e) {
        sourceFile = source;
    }

    /*
     * Convert absolute path into a path
     * relative to the extension root.
     */
    const prefix = `${extensionDir}/`;

    if (sourceFile.startsWith(prefix)) {
        sourceFile = sourceFile.substring(prefix.length);
    }

    const relativeSource = sourceFile;


    return function journal(msg) {

        /*
         * Logging disabled?
         *
         * Do absolutely nothing.
         */
        if (!loggingEnabled)
            return;


        /*
         * Create timestamp once.
         */
        // const timestamp = GLib.DateTime
        //     .new_now_local()
        //     .format('%Y-%m-%d %H:%M:%S');


        /*
         * Create the complete message once.
         *
         * Both Journal and file receive
         * exactly this same string.
         */
        // const output = `[${timestamp}] [${relativeSource}] ${msg}`;
        const output = `[${relativeSource}] ${msg}`;


        if (
            logOutput === 'journal' ||
            logOutput === 'both'
        ) {
            writeToJournal(output);
        }


        if (
            logOutput === 'file' ||
            logOutput === 'both'
        ) {
            writeToFile(output);
        }
    };
}


function writeToJournal(output) {
    GLib.log_structured(
        uuid,
        GLib.LogLevelFlags.LEVEL_MESSAGE,
        {
            MESSAGE: output,
            SYSLOG_IDENTIFIER: uuid,
        }
    );
}


function writeToFile(output) {
    if (!logFile)
        return;

    try {
        const file = Gio.File.new_for_path(logFile);

        const stream = file.append_to(
            Gio.FileCreateFlags.NONE,
            null
        );

        const bytes =
            new TextEncoder().encode(`${output}\n`);

        stream.write_all(bytes, null);
        stream.close(null);

    } catch (e) {
        /*
         * Don't call journal() here.
         */
        console.error(
            `[${uuid}] Failed to write log file: ${e}`
        );
    }
}


export {
    initLogging,
    createLogger,
};

import { Extension } from 'resource:///org/gnome/shell/extensions/extension.js';
import * as Main from 'resource:///org/gnome/shell/ui/main.js';
import { WorkspaceIndicator } from './lib/workspaceIndicator.js';
import { TitleBarMoveMonitor } from './lib/titleBarMoveMonitor.js';
import { initOverlayService, destroyOverlayService } from './lib/overlayService.js';

import { initLogging, createLogger } from './logger.js';

const journal = createLogger(import.meta.url);

export default class TopNotchWorkspaces extends Extension {
    constructor(metadata) {
        super(metadata);
        this._indicator = null;
        this._titleBarMoveMonitor = null;
    }

    enable() {
        initLogging(this.uuid, 'file', false);
        journal(`Enabled`);

        const settings = this.getSettings();

        // Before the indicator — its toolbar buttons call into the service.
        initOverlayService(settings);

        this._indicator = new WorkspaceIndicator(settings);
        Main.panel.addToStatusArea('workspace-indicator', this._indicator, 0, 'left');

        this._titleBarMoveMonitor = new TitleBarMoveMonitor();
    }

    disable() {
        destroyOverlayService();

        if (this._titleBarMoveMonitor) {
            this._titleBarMoveMonitor.destroy();
            this._titleBarMoveMonitor = null;
        }
        if (this._indicator) {
            this._indicator.destroy();
            this._indicator = null;
        }
    }
}
